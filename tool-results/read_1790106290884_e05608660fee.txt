   343→function ThemePreview({
   344→  value,
   345→}: {
   346→  value: "light" | "dark" | "system";
   347→}) {
   348→  if (value === "system") {
   349→    return (
   350→      <div className="grid h-20 grid-cols-2 overflow-hidden rounded-xl border">
   351→        <div className="bg-gradient-to-br from-white to-neutral-200 p-2">
   352→          <div className="h-1.5 w-8 rounded-full bg-neutral-400" />
   353→          <div className="mt-1 h-1 w-10 rounded-full bg-neutral-300" />
   354→          <div className="mt-2 h-3 w-3 rounded-full bg-rose-400" />
   355→        </div>
   356→        <div className="bg-gradient-to-br from-neutral-800 to-black p-2">
   357→          <div className="h-1.5 w-8 rounded-full bg-neutral-500" />
   358→          <div className="mt-1 h-1 w-10 rounded-full bg-neutral-600" />
   359→          <div className="mt-2 h-3 w-3 rounded-full bg-rose-400" />
   360→        </div>
   361→      </div>
   362→    );
   363→  }
   364→  const isDark = value === "dark";
   365→  return (
   366→    <div
   367→      className={cn(
   368→        "h-20 rounded-xl border p-2",
   369→        isDark
   370→          ? "bg-gradient-to-br from-neutral-800 to-black"
   371→          : "bg-gradient-to-br from-white to-neutral-200",
   372→      )}
   373→    >
   374→      <div
   375→        className={cn(
   376→          "h-1.5 w-8 rounded-full",
   377→          isDark ? "bg-neutral-500" : "bg-neutral-400",
   378→        )}
   379→      />
   380→      <div
   381→        className={cn(
   382→          "mt-1 h-1 w-10 rounded-full",
   383→          isDark ? "bg-neutral-600" : "bg-neutral-300",
   384→        )}
   385→      />
   386→      <div className="mt-2 h-3 w-3 rounded-full bg-[var(--gold)]" />
   387→    </div>
   388→  );
   389→}
