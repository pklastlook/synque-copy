   555→   555→function AddQueryDialog({ onCreated }: { onCreated: () => void }) {
   556→   556→  const [open, setOpen] = React.useState(false);
   557→   557→  const [name, setName] = React.useState("");
   558→   558→  const [keywords, setKeywords] = React.useState("");
   559→   559→  const [type, setType] = React.useState<QueryType>("brand");
   560→   560→  const [color, setColor] = React.useState(SWATCHES[0]);
   561→   561→  const [saving, setSaving] = React.useState(false);
   562→   562→
   563→   563→  function reset() {
   564→   564→    setName("");
   565→   565→    setKeywords("");
   566→   566→    setType("brand");
   567→   567→    setColor(SWATCHES[0]);
   568→   568→  }
   569→   569→
   570→   570→  async function handleSubmit() {
   571→   571→    if (!name.trim() || !keywords.trim()) {
   572→   572→      toast.error("Name and keywords are required");
   573→   573→      return;
   574→   574→    }
   575→   575→    setSaving(true);
   576→   576→    try {
   577→   577→      const res = await fetch("/api/listening", {
   578→   578→        method: "POST",
   579→   579→        headers: { "Content-Type": "application/json" },
   580→   580→        body: JSON.stringify({ name: name.trim(), keywords: keywords.trim(), type, color }),
   581→   581→      });
   582→   582→      const json = await res.json();
   583→   583→      if (json.ok) {
   584→   584→        toast.success("Listening query created", { description: name.trim() });
   585→   585→        setOpen(false);
   586→   586→        reset();
   587→   587→        onCreated();
   588→   588→      } else {
   589→   589→        toast.error(json.error ?? "Failed to create query");
   590→   590→      }
   591→   591→    } catch {
   592→   592→      toast.error("Network error");
   593→   593→    } finally {
   594→   594→      setSaving(false);
   595→   595→    }
   596→   596→  }
   597→   597→
   598→   598→  return (
   599→   599→    <>
   600→   600→      <button
   601→   601→        onClick={() => setOpen(true)}
   602→   602→        className="liquid-pill glow-ring flex items-center gap-2 rounded-2xl px-4 py-2.5 text-sm font-semibold"
   603→   603→      >
   604→   604→        <Plus className="h-4 w-4" />
   605→   605→        Add listening query
   606→   606→      </button>
   607→   607→      <Dialog
   608→   608→        open={open}
   609→   609→        onOpenChange={(o) => {
   610→   610→          setOpen(o);
   611→   611→          if (!o) reset();
   612→   612→        }}
   613→   613→      >
   614→   614→        <DialogContent className="rounded-3xl sm:max-w-md">
   615→   615→          <DialogHeader>
   616→   616→            <DialogTitle className="flex items-center gap-2">
   617→   617→              <Radar className="h-5 w-5 text-primary" />
   618→   618→              New listening query
   619→   619→            </DialogTitle>
   620→   620→            <DialogDescription>
   621→   621→              Track brand mentions, competitor activity, or trending keywords across the web
   622→   622→              &amp; social platforms.
   623→   623→            </DialogDescription>
   624→   624→          </DialogHeader>
   625→   625→          <div className="space-y-4 py-2">
   626→   626→            <div className="space-y-2">
   627→   627→              <Label htmlFor="lq-name">Query name</Label>
   628→   628→              <Input
   629→   629→                id="lq-name"
   630→   630→                placeholder="e.g. SocialFlow Brand"
   631→   631→                value={name}
   632→   632→                onChange={(e) => setName(e.target.value)}
   633→   633→              />
   634→   634→            </div>
   635→   635→            <div className="space-y-2">

   643→   643→              />
   644→   644→              <p className="text-[11px] text-muted-foreground">
