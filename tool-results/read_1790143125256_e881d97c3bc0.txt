   555→function AddQueryDialog({ onCreated }: { onCreated: () => void }) {
   556→  const [open, setOpen] = React.useState(false);
   557→  const [name, setName] = React.useState("");
   558→  const [keywords, setKeywords] = React.useState("");
   559→  const [type, setType] = React.useState<QueryType>("brand");
   560→  const [color, setColor] = React.useState(SWATCHES[0]);
   561→  const [saving, setSaving] = React.useState(false);
   562→
   563→  function reset() {
   564→    setName("");
   565→    setKeywords("");
   566→    setType("brand");
   567→    setColor(SWATCHES[0]);
   568→  }
   569→
   570→  async function handleSubmit() {
   571→    if (!name.trim() || !keywords.trim()) {
   572→      toast.error("Name and keywords are required");
   573→      return;
   574→    }
   575→    setSaving(true);
   576→    try {
   577→      const res = await fetch("/api/listening", {
   578→        method: "POST",
   579→        headers: { "Content-Type": "application/json" },
   580→        body: JSON.stringify({ name: name.trim(), keywords: keywords.trim(), type, color }),
   581→      });
   582→      const json = await res.json();
   583→      if (json.ok) {
   584→        toast.success("Listening query created", { description: name.trim() });
   585→        setOpen(false);
   586→        reset();
   587→        onCreated();
   588→      } else {
   589→        toast.error(json.error ?? "Failed to create query");
   590→      }
   591→    } catch {
   592→      toast.error("Network error");
   593→    } finally {
   594→      setSaving(false);
   595→    }
   596→  }
   597→
   598→  return (
   599→    <>
   600→      <button
   601→        onClick={() => setOpen(true)}
   602→        className="liquid-pill glow-ring flex items-center gap-2 rounded-2xl px-4 py-2.5 text-sm font-semibold"
   603→      >
   604→        <Plus className="h-4 w-4" />
   605→        Add listening query
   606→      </button>

   609→        onOpenChange={(o) => {
   610→          setOpen(o);
   611→          if (!o) reset();
   612→        }}
   613→      >
   614→        <DialogContent className="rounded-3xl sm:max-w-md">
   615→          <DialogHeader>
   616→            <DialogTitle className="flex items-center gap-2">
   617→              <Radar className="h-5 w-5 text-primary" />
   618→              New listening query

   623→            </DialogDescription>

   650→              <Select
   651→                value={type}
   652→                onValueChange={(v) => setType(v as QueryType)}

   654→                <SelectTrigger className="w-full">

   668→                {SWATCHES.map((sw) => (
   669→                  <button
   670→                    key={sw}
   671→                    type="button"
   672→                    onClick={() => setColor(sw)}
   673→                    className={cn(
   674→                      "h-8 w-8 rounded-full transition",
   675→                      color === sw
   676→                        ? "ring-2 ring-offset-2 ring-offset-background"
   677→                        : "opacity-80 hover:opacity-100",
   678→                    )}
   679→                    style={{ background: sw, boxShadow: color === sw ? `0 0 0 2px ${sw}` : undefined }}
   680→                    aria-label={`Color ${sw}`}
   681→                  />
   682→                ))}
   683→              </div>
   684→            </div>
   685→          </div>
   686→          <DialogFooter>
   687→            <Button variant="ghost" onClick={() => setOpen(false)} disabled={saving}>
   688→              Cancel
   689→            </Button>
   690→            <Button onClick={handleSubmit} disabled={saving} className="gap-1.5">
   691→              {saving ? "Creating…" : "Create query"}
   692→              {!saving && <Plus className="h-3.5 w-3.5" />}
   693→            </Button>
   694→          </DialogFooter>
   695→        </DialogContent>
   696→      </Dialog>
   697→    </>
