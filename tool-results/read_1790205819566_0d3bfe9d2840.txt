    79→function SplashScreen({ onComplete }: { onComplete: () => void }) {
    80→  const [progress, setProgress] = React.useState(0);
    81→  const [done, setDone] = React.useState(false);
    82→
    83→  React.useEffect(() => {
    84→    let p = 0;
    85→    const iv = setInterval(() => {
    86→      p += Math.random() * 14 + 6;
    87→      if (p >= 100) {
    88→        p = 100;
    89→        clearInterval(iv);
    90→        setProgress(100);
    91→        setTimeout(() => setDone(true), 350);
    92→        setTimeout(onComplete, 1100);

    94→        setProgress(Math.floor(p));

   634→  const { signup } = useAuth();
   635→  const [step, setStep] = React.useState<"details" | "payment" | "activating" | "done">("details");
   636→  const [form, setForm] = React.useState({
   637→    name: "", email: "", company: "", teamSize: "5-10", phone: "", password: "",
   638→    cardName: "", cardNumber: "", expiry: "", cvc: "",

   649→  const validateDetails = () => {
   650→    const e: Record<string, string> = {};
   651→    if (!form.name.trim()) e.name = "Full name is required";
   652→    if (!form.email.trim()) e.email = "Email is required";
   653→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = "Invalid email";
   654→    if (!form.company.trim()) e.company = "Company is required";
   655→    if (!form.password.trim()) e.password = "Password is required";
   656→    else if (form.password.length < 6) e.password = "Min 6 characters";
   657→    setErrors(e);
   658→    return Object.keys(e).length === 0;
   659→  };
   660→
   661→  const validatePayment = () => {
   662→    const e: Record<string, string> = {};
   663→    if (!form.cardName.trim()) e.cardName = "Name on card required";
   664→    if (!form.cardNumber.trim()) e.cardNumber = "Card number required";
   665→    else if (form.cardNumber.replace(/\s/g, "").length < 15) e.cardNumber = "Invalid card number";
   666→    if (!form.expiry.trim()) e.expiry = "Expiry required";
