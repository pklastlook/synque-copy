   100→function useInView<T extends HTMLElement>(opts?: IntersectionObserverInit) {
   101→  const ref = React.useRef<T>(null);
   102→  const [inView, setInView] = React.useState(false);
   103→  React.useEffect(() => {
   104→    const el = ref.current;
   105→    if (!el) return;
   106→    if (typeof IntersectionObserver === "undefined") { setInView(true); return; }
   107→    const obs = new IntersectionObserver(
   108→      ([entry]) => {
   109→        if (entry.isIntersecting) {
   110→          setInView(true);
   111→          obs.unobserve(entry.target);
   112→        }
   113→      },
   114→      { threshold: 0.15, rootMargin: "0px 0px -10% 0px", ...opts },
   115→    );
   116→    obs.observe(el);
   117→    return () => obs.disconnect();
   118→  }, []);
   119→  return { ref, inView };
   120→}

   131→  const { ref, inView } = useInView<HTMLDivElement>();
   132→  return (
   133→    <div

   135→      className={cn("transition-all duration-700 ease-out will-change-transform", className)}

   137→        transitionDelay: `${delay}ms`,
   138→        opacity: inView ? 1 : 0,
   139→        transform: inView ? "translateY(0)" : `translateY(${y}px)`,
   140→      }}
   141→    >
   142→      {children}
   143→    </div>
   144→  );
   145→}

   147→/* CSS particle field — pure CSS animation, JS only generates the dots */
   148→function ParticleField({
   149→  count = 28,
   150→  colors = ["#7DD3FC", "#22D3EE", "#5EEAD4", "#A78BFA"],
   151→  size = [2, 4],
   152→}: { count?: number; colors?: string[]; size?: [number, number] }) {
   153→  const dots = React.useMemo(() => {
   154→    const seed = 1337;
   155→    const rand = (i: number) => {
   156→      const x = Math.sin(seed + i * 12.9898) * 43758.5453;
   157→      return x - Math.floor(x);
   158→    };
   159→    return Array.from({ length: count }).map((_, i) => {
   160→      const s = size[0] + rand(i) * (size[1] - size[0]);
   161→      return {
   162→        left: rand(i * 7 + 1) * 100,
   163→        top: rand(i * 11 + 3) * 100,
   164→        size: s,
   165→        delay: rand(i * 5 + 7) * -20,
   166→        dur: 14 + rand(i * 3 + 9) * 18,
   167→        color: colors[i % colors.length],
   168→        drift: (rand(i * 13 + 2) - 0.5) * 60,
   169→      };
   170→    });
   171→  }, [count, colors, size]);
   172→  return (
   173→    <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   174→      {dots.map((d, i) => (
   175→        <span
   176→          key={i}
   177→          className="absolute rounded-full"
   178→          style={{
   179→            left: `${d.left}%`,
   180→            top: `${d.top}%`,
   181→            width: d.size,
   182→            height: d.size,
   183→            background: d.color,
   184→            boxShadow: `0 0 ${d.size * 3}px ${d.color}`,
   185→            animation: `sfParticleFloat ${d.dur}s linear ${d.delay}s infinite`,
   186→            ["--drift" as string]: `${d.drift}px`,
   187→            opacity: 0.5,
   188→          }}
   189→        />
   190→      ))}
   191→    </div>
   192→  );
   193→}

   196→function ConfettiField({ count = 36 }: { count?: number }) {
   197→  const pieces = React.useMemo(() => {
   198→    const colors = ["#22D3EE", "#5EEAD4", "#A78BFA", "#FBBF24", "#F472B6", "#34D399"];
   199→    const seed = 9001;
   200→    const rand = (i: number) => {
   201→      const x = Math.sin(seed + i * 7.91) * 43758.5453;
   202→      return x - Math.floor(x);
   203→    };
   204→    return Array.from({ length: count }).map((_, i) => ({
   205→      left: rand(i + 1) * 100,
   206→      delay: rand(i + 11) * -8,
   207→      dur: 5 + rand(i + 21) * 6,
   208→      size: 6 + rand(i + 31) * 8,
   209→      color: colors[i % colors.length],
   210→      rot: rand(i + 41) * 360,
   211→    }));
   212→  }, [count]);
   213→  return (
   214→    <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   215→      {pieces.map((p, i) => (
   216→        <span
   217→          key={i}
   218→          className="absolute top-0 block"
   219→          style={{
   220→            left: `${p.left}%`,
   221→            width: p.size,
   222→            height: p.size * 0.5,
   223→            background: p.color,

   225→            transform: `rotate(${p.rot}deg)`,
   226→            animation: `sfConfettiFall ${p.dur}s linear ${p.delay}s infinite`,
   227→            opacity: 0.85,
   228→          }}
   229→        />
   230→      ))}
   231→    </div>
   232→  );
