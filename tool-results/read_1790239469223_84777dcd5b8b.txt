   404→   404→              {stats.total === 0 && (

   406→   406→                  onClick={() => setShowPasteForm(true)}

   427→   427→        </div>

   429→   429→    </div>
   430→   430→  );
   431→   431→}

   433→   433→// ============ Paste Form ============
   434→   434→function PasteForm({ userId, onSaved }: { userId: string | null; onSaved: () => void }) {
   435→   435→  const [source, setSource] = React.useState("group");
   436→   436→  const [senderName, setSenderName] = React.useState("");
   437→   437→  const [groupName, setGroupName] = React.useState("");
   438→   438→  const [content, setContent] = React.useState("");
   439→   439→  const [mediaUrl, setMediaUrl] = React.useState("");
   440→   440→  const [saving, setSaving] = React.useState(false);
   441→   441→
   442→   442→  async function handleSubmit() {
   443→   443→    if (!content.trim()) {
   444→   444→      toast.error("Paste some content first");
   445→   445→      return;
   446→   446→    }
   447→   447→    if (!senderName.trim()) {
   448→   448→      toast.error("Enter the sender's name");
   449→   449→      return;
   450→   450→    }
   451→   451→    setSaving(true);
   452→   452→    try {
   453→   453→      const res = await fetch("/api/whatsapp", {
   454→   454→        method: "POST",
   455→   455→        headers: { "Content-Type": "application/json" },
   456→   456→        body: JSON.stringify({
   457→   457→          source,
   458→   458→          senderName: senderName.trim(),
   459→   459→          groupName: groupName.trim() || null,
   460→   460→          content: content.trim(),
   461→   461→          mediaUrl: mediaUrl.trim() || null,
   462→   462→          savedById: userId,
   463→   463→        }),
   464→   464→      });
