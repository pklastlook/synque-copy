    63→// ============ Main view ============
    64→export function ReportsView() {
    65→  const [dateRange, setDateRange] = React.useState<DateRange>("30d");

    67→  const [generating, setGenerating] = React.useState(false);
    68→  const [reportTitle, setReportTitle] = React.useState("");
    69→  const [reportType, setReportType] = React.useState("performance");
    70→  const { data: savedReports, reload } = useApi<ReportData[]>("/api/reports");
    71→
    72→  const days = DATE_RANGES.find(d => d.key === dateRange)?.days ?? 30;
    73→  const analyticsUrl = `/api/analytics?days=${days}&accountId=all`;
    74→  const { data: analytics, loading: analyticsLoading } = useApi<AnalyticsData>(analyticsUrl);
    75→
    76→  function getToken(): string | null {
    77→    try { return JSON.parse(localStorage.getItem("socialflow-auth") ?? "{}")?.state?.token ?? null; }
    78→    catch { return null; }
    79→  }
    80→
    81→  async function generateReport(templateId: string) {
    82→    const template = TEMPLATES.find(t => t.id === templateId);
    83→    if (!template) return;
    84→    setGenerating(true);
    85→    setReportType(templateId);
    86→    const title = `${template.name} — ${new Date().toLocaleDateString(undefined, { month: "short", day: "numeric", year: "numeric" })}`;
    87→    setReportTitle(title);
    88→
    89→    try {
    90→      const token = getToken();
    91→      const res = await fetch("/api/reports", {
    92→        method: "POST",
    93→        headers: {
    94→          "Content-Type": "application/json",
    95→          ...(token ? { "Authorization": `Bearer ${token}` } : {}),
    96→        },
    97→        body: JSON.stringify({
    98→          name: title,
    99→          type: templateId,
   100→          description: `${DATE_RANGES.find(d => d.key === dateRange)?.label} · Last ${days} days`,
   101→          config: JSON.stringify({ days, range: dateRange }),
   102→          createdBy: null,
   103→        }),
   104→      });
   105→      if (res.ok) {
   106→        toast.success("Report generated", { description: "Scroll down to view the full report" });
   107→        setShowReport(true);
   108→        reload();
   109→      } else {
   110→        toast.error("Failed to generate report");
   111→      }
   112→    } catch {
   113→      toast.error("Network error");
   114→    } finally {
   115→      setGenerating(false);
   116→    }
   117→  }
   118→
   119→  async function downloadPDF(r: ReportData) {
   120→    try {
   121→      const token = getToken();
   122→      const res = await fetch(`/api/reports/${r.id}/pdf`, {
   123→        headers: token ? { "Authorization": `Bearer ${token}` } : {},
   124→      });
   125→      if (!res.ok) throw new Error("PDF generation failed");
   126→      const blob = await res.blob();
   127→      const url = URL.createObjectURL(blob);
   128→      const a = document.createElement("a");
   129→      a.href = url;
   130→      a.download = `${r.name.replace(/[^a-z0-9]/gi, "_")}.pdf`;
   131→      a.click();
   132→      URL.revokeObjectURL(url);
   133→      toast.success("PDF downloaded");
   134→    } catch { toast.error("Failed to generate PDF"); }
   135→  }
   136→
   137→  async function deleteReport(id: string) {
   138→    if (!confirm("Delete this report?")) return;
   139→    const token = getToken();
   140→    await fetch(`/api/reports/${id}`, {
   141→      method: "DELETE",
   142→      headers: token ? { "Authorization": `Bearer ${token}` } : {},
   143→    });
   144→    toast.success("Report deleted");
   145→    reload();
   146→  }

   148→  function printReport() {

   151→
   152→  const allReports = savedReports ?? [];

   154→  // Calculate totals

   161→      acc.engagement += a.engagement;

   163→      acc.likes += a.likes;
   164→      acc.comments += a.comments;
   165→      acc.shares += a.shares;
   166→      acc.saves += a.saves;

   168→    }, { followers: 0, impressions: 0, reach: 0, engagement: 0, posts: 0, likes: 0, comments: 0, shares: 0, saves: 0 });
   169→  }, [analytics]);
   170→
   171→  return (

   175→        then view, print, or download as PDF.

   209→          >
   210→            <div className="grid h-12 w-12 place-items-center rounded-xl" style={{ background: `${t.color}14`, color: t.color }}>

   213→            <div className="mt-3 text-sm font-bold">{t.name}</div>

   234→                <button onClick={() => downloadPDF({ id: allReports[0]?.id ?? "", name: reportTitle, type: reportType, description: "", config: "", schedule: null, createdBy: null, lastGenerated: new Date().toISOString(), createdAt: new Date().toISOString() })} className="flex items-center gap-1.5 rounded-lg bg-primary/10 px-3 py-1.5 text-xs font-semibold text-primary transition hover:bg-primary/20">

   241→          {/* KPI Grid */}
   242→          <div className="mt-4 grid grid-cols-2 gap-3 lg:grid-cols-4">

   244→            <ReportKPI label="Impressions" value={fmt(totals.impressions)} icon={<Eye className="h-4 w-4" />} color="oklch(0.62 0.22 300)" />

   249→          {/* Charts */}

   381→                      <Eye className="h-2.5 w-2.5" /> View

   387→                      <Trash2 className="h-3 w-3" />
   388→                    </button>

   390→                </div>

   412→      <div className="flex items-center gap-2" style={{ color }}>

   421→function PrintLayout({ title, dateRange, analytics, totals }: {
   422→  title: string; dateRange: string; analytics: AnalyticsData | null;
   423→  totals: { followers: number; impressions: number; reach: number; engagement: number; posts: number; likes: number; comments: number; shares: number; saves: number };
