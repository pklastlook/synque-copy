   109→   109→function fmtSize(bytes: number) {
   110→   110→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
   111→   111→  if (bytes >= 1_048_576) return `${(bytes / 1_048_576).toFixed(1)} MB`;
   112→   112→  if (bytes >= 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   113→   113→  return `${bytes} B`;
   114→   114→}
   115→   115→
   116→   116→function fmtBytesBig(bytes: number) {
   117→   117→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
   118→   118→  return `${(bytes / 1_048_576).toFixed(1)} MB`;
   119→   119→}

   142→   142→function parseTags(tags: string | null): string[] {
