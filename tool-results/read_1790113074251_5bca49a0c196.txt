   109→function fmtSize(bytes: number) {
   110→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
   111→  if (bytes >= 1_048_576) return `${(bytes / 1_048_576).toFixed(1)} MB`;
   112→  if (bytes >= 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   113→  return `${bytes} B`;
   114→}
   115→
   116→function fmtBytesBig(bytes: number) {
   117→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
   118→  return `${(bytes / 1_048_576).toFixed(1)} MB`;
   119→}

   142→function parseTags(tags: string | null): string[] {
   143→  if (!tags) return [];

   581→function UploadDialog({
   582→  open,
   583→  onOpenChange,
   584→  folders,
   585→  onUploaded,
   586→}: {
   587→  open: boolean;
   588→  onOpenChange: (v: boolean) => void;
   589→  folders: string[];
   590→  onUploaded: () => void;
   591→}) {
   592→  const [url, setUrl] = React.useState("");
   593→  const [name, setName] = React.useState("");
   594→  const [tags, setTags] = React.useState("");
   595→  const [folder, setFolder] = React.useState<string>("root");
   596→  const [type, setType] = React.useState<"image" | "video" | "audio">("image");
   597→  const [submitting, setSubmitting] = React.useState(false);
   598→
   599→  const reset = () => {
   600→    setUrl("");
   601→    setName("");
   602→    setTags("");
   603→    setFolder("root");
   604→    setType("image");
   605→  };
   606→
   607→  const handleSubmit = async (override?: Partial<typeof PRESET_LIBRARY[number]> & { url?: string; name?: string }) => {
   608→    const finalUrl = override?.url ?? url;
   609→    const finalName = override?.name ?? name;
   610→    if (!finalUrl || !finalName) {
   611→      toast.error("URL and name are required");
   612→      return;
   613→    }
   614→    setSubmitting(true);
   615→    try {
   616→      const res = await fetch("/api/media", {
   617→        method: "POST",
   618→        headers: { "Content-Type": "application/json" },
   619→        body: JSON.stringify({
   620→          name: finalName,
   621→          url: finalUrl,
   622→          type,
   623→          tags: (override?.tags ?? tags) || null,
   624→          folder,
   625→          uploadedById: "u-admin",
   626→        }),
   627→      });
   628→      if (!res.ok) throw new Error("Upload failed");
   629→      toast.success(`"${finalName}" added`);
   630→      reset();
   631→      onUploaded();
   632→    } catch {
   633→      toast.error("Could not upload asset");
   634→    } finally {
   635→      setSubmitting(false);
   636→    }
   637→  };
   638→
   639→  return (
   640→    <Dialog open={open} onOpenChange={onOpenChange}>
   641→      <DialogContent className="sm:max-w-lg rounded-3xl">
   642→        <DialogHeader>
   643→          <DialogTitle className="flex items-center gap-2">
   644→            <span className="grid h-7 w-7 place-items-center rounded-lg bg-primary text-primary-foreground">
   645→              <Upload className="h-4 w-4" />
   646→            </span>
   647→            Add media
   648→          </DialogTitle>
   649→          <DialogDescription>
   650→            Paste a URL or pick from the preset library below.
   651→          </DialogDescription>
   652→        </DialogHeader>
   653→
   654→        <div className="space-y-4">
   655→          <div className="space-y-2">
   656→            <Label htmlFor="m-url">Asset URL</Label>
   657→            <Input
   658→              id="m-url"
   659→              placeholder="https://example.com/image.jpg"
   660→              value={url}
   661→              onChange={(e) => setUrl(e.target.value)}
   662→            />
   663→          </div>
   664→          <div className="grid grid-cols-2 gap-3">
   665→            <div className="space-y-2">
   666→              <Label htmlFor="m-name">Name</Label>
   667→              <Input
   668→                id="m-name"
   669→                placeholder="Photo name"
   670→                value={name}
   671→                onChange={(e) => setName(e.target.value)}
   672→              />
   673→            </div>
   674→            <div className="space-y-2">
   675→              <Label htmlFor="m-tags">Tags</Label>
   676→              <Input
   677→                id="m-tags"
   678→                placeholder="brand, product"
   679→                value={tags}
   680→                onChange={(e) => setTags(e.target.value)}
   681→              />
   682→            </div>
   683→          </div>
   684→          <div className="grid grid-cols-2 gap-3">
   685→            <div className="space-y-2">
   686→              <Label>Type</Label>
   687→              <Select
   688→                value={type}
   689→                onValueChange={(v) => setType(v as "image" | "video" | "audio")}
   690→              >
   691→                <SelectTrigger className="w-full">
   692→                  <SelectValue />
   693→                </SelectTrigger>
   694→                <SelectContent>
   695→                  <SelectItem value="image">Image</SelectItem>
   696→                  <SelectItem value="video">Video</SelectItem>
   697→                  <SelectItem value="audio">Audio</SelectItem>
   698→                </SelectContent>
   699→              </Select>
   700→            </div>
   701→            <div className="space-y-2">
   702→              <Label>Folder</Label>
   703→              <Select value={folder} onValueChange={setFolder}>
   704→                <SelectTrigger className="w-full">
   705→                  <SelectValue />
   706→                </SelectTrigger>
   707→                <SelectContent>
   708→                  <SelectItem value="root">root</SelectItem>
   709→                  {folders
   710→                    .filter((f) => f !== "root")
   711→                    .map((f) => (
   712→                      <SelectItem key={f} value={f}>
   713→                        {f}
   714→                      </SelectItem>
   715→                    ))}
   716→                </SelectContent>
   717→              </Select>
   718→            </div>
   719→          </div>
   720→
   721→          {/* Preset library */}
   722→          <div className="space-y-2">
   723→            <Label>Or pick from library</Label>
   724→            <div className="grid grid-cols-3 gap-2">
   725→              {PRESET_LIBRARY.map((p) => (
   726→                <button
   727→                  key={p.url}
   728→                  type="button"
   729→                  onClick={() => handleSubmit(p)}
   730→                  disabled={submitting}
   731→                  className="group relative aspect-square overflow-hidden rounded-xl border border-border/40 transition hover:-translate-y-0.5 hover:border-primary/40"
   732→                  title={p.name}
   733→                >
   734→                  <img
   735→                    src={p.url}
   736→                    alt={p.name}
   737→                    className="h-full w-full object-cover transition group-hover:scale-105"
   738→                  />
   739→                  <span className="absolute inset-0 bg-black/0 transition group-hover:bg-black/30" />
   740→                  <span className="absolute bottom-1 left-1 right-1 truncate rounded-md bg-white/90 px-1.5 py-0.5 text-[9px] font-medium opacity-0 transition group-hover:opacity-100">
   741→                    {p.name}
   742→                  </span>
   743→                </button>
   744→              ))}
   745→            </div>
   746→          </div>
   747→        </div>
   748→
   749→        <DialogFooter>
   750→          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={submitting}>
   751→            Cancel
   752→          </Button>
   753→          <button
   754→            onClick={() => handleSubmit()}
   755→            disabled={submitting}
   756→            className="liquid-pill flex items-center gap-2 rounded-xl px-4 py-2 text-sm font-semibold disabled:opacity-60"
   757→          >
   758→            <Plus className="h-4 w-4" />
   759→            {submitting ? "Adding…" : "Add asset"}
   760→          </button>
   761→        </DialogFooter>
   762→      </DialogContent>
   763→    </Dialog>
   764→  );
   765→}
