   476→/* ============================== 3. Team performance table ============================== */
   477→
   478→type SortKey = "engagement" | "posts" | "followers" | "accounts" | "tasks";
   479→type SortDir = "asc" | "desc";
   480→
   481→function TeamPerformanceSection({ members }: { members: TeamMember[] }) {
   482→  const [sortKey, setSortKey] = React.useState<SortKey>("engagement");
   483→  const [sortDir, setSortDir] = React.useState<SortDir>("desc");
   484→  const [selected, setSelected] = React.useState<TeamMember | null>(null);
   485→
   486→  const sorted = React.useMemo(() => {
   487→    const copy = [...members];
   488→    copy.sort((a, b) => {
   489→      const va =
   490→        sortKey === "tasks"
   491→          ? a.tasksDone
   492→          : sortKey === "engagement"
   493→            ? a.engagement
   494→            : sortKey === "posts"
   495→              ? a.postsCount
   496→              : sortKey === "followers"
   497→                ? a.followers
   498→                : a.accountsCount;
   499→      const vb =
   500→        sortKey === "tasks"
   501→          ? b.tasksDone
   502→          : sortKey === "engagement"
   503→            ? b.engagement
   504→            : sortKey === "posts"
   505→              ? b.postsCount
   506→              : sortKey === "followers"
   507→                ? b.followers
   508→                : b.accountsCount;
   509→      return sortDir === "asc" ? va - vb : vb - va;
   510→    });
   511→    return copy;
   512→  }, [members, sortKey, sortDir]);
   513→
   514→  const toggleSort = (k: SortKey) => {
   515→    if (k === sortKey) {
   516→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   517→    } else {
   518→      setSortKey(k);
   519→      setSortDir("desc");
   520→    }
   521→  };
   522→
   523→  return (
   524→    <GlassCard className="crystal-in overflow-hidden p-0" style={{ animationDelay: "0.5s" }}>
   525→      <div className="p-5 pb-3">
   526→        <SectionTitle
   527→          title="Team performance"
   528→          desc="Per-member output across accounts, posts, tasks & engagement"
   529→          action={
   530→            <div className="flex items-center gap-2 text-[11px] text-muted-foreground">
   531→              <TrendingUp className="h-3.5 w-3.5 text-primary" />
   532→              sorted by{" "}
   533→              <span className="font-medium text-foreground">{sortKey}</span>
   534→            </div>
   535→          }
   536→        />
   537→      </div>
   538→
   539→      <div className="overflow-x-auto">
   540→        <div className="min-w-[820px]">
   541→          {/* Header */}
   542→          <div className="grid grid-cols-[2.2fr_0.8fr_0.8fr_1fr_1fr_1fr_auto] items-center gap-3 border-b border-border/40 px-5 py-2.5 text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   543→            <SortHeader
   544→              label="Member"
   545→              active={false}
   546→              dir={sortDir}
   547→              onClick={() => toggleSort("engagement")}
   548→            />
   549→            <SortHeader
   550→              label="Accounts"
   551→              active={sortKey === "accounts"}
   552→              dir={sortDir}
   553→              onClick={() => toggleSort("accounts")}
   554→              align="right"
   555→            />
   556→            <SortHeader
   557→              label="Posts"
   558→              active={sortKey === "posts"}
   559→              dir={sortDir}
   560→              onClick={() => toggleSort("posts")}
   561→              align="right"
   562→            />
   563→            <SortHeader
   564→              label="Tasks"
   565→              active={sortKey === "tasks"}
   566→              dir={sortDir}
   567→              onClick={() => toggleSort("tasks")}
   568→              align="right"
   569→            />
   570→            <SortHeader
   571→              label="Engagement"
   572→              active={sortKey === "engagement"}
   573→              dir={sortDir}
   574→              onClick={() => toggleSort("engagement")}
   575→              align="right"
   576→            />
   577→            <SortHeader
   578→              label="Followers"
   579→              active={sortKey === "followers"}
   580→              dir={sortDir}
   581→              onClick={() => toggleSort("followers")}
   582→              align="right"
   583→            />
   584→            <span className="text-right">Profile</span>
   585→          </div>
   586→
   587→          {/* Rows */}
   588→          <div className="max-h-[640px] overflow-y-auto scroll-liquid">
   589→            {sorted.map((m) => (
   590→              <TeamRow
   591→                key={m.id}
   592→                member={m}
   593→                onProfile={() => setSelected(m)}
   594→              />
   595→            ))}
   596→            {sorted.length === 0 && (
   597→              <div className="py-12 text-center text-sm text-muted-foreground">
   598→                No team members yet.
   599→              </div>
   600→            )}
   601→          </div>
   602→        </div>
   603→      </div>
   604→
   605→      <MemberProfileDialog
   606→        member={selected}
   607→        onClose={() => setSelected(null)}
   608→      />
   609→    </GlassCard>
   610→  );
   611→}
   612→
   613→function SortHeader({
   614→  label,
   615→  active,
   616→  dir,
   617→  onClick,
   618→  align = "left",
   619→}: {
   620→  label: string;
   621→  active: boolean;
   622→  dir: SortDir;
   623→  onClick: () => void;
   624→  align?: "left" | "right";
   625→}) {
   626→  return (
   627→    <button
   628→      onClick={onClick}
   629→      className={cn(
   630→        "flex items-center gap-1 transition hover:text-foreground",
   631→        align === "right" ? "justify-end" : "justify-start",
   632→        active && "text-foreground",
   633→      )}
   634→    >
   635→      {label}
