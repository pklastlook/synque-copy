   634→   634→  const { signup } = useAuth();

   650→   650→    const e: Record<string, string> = {};

   655→   655→    if (!form.password.trim()) e.password = "Password is required";
   656→   656→    else if (form.password.length < 6) e.password = "Min 6 characters";
   657→   657→    setErrors(e);
   658→   658→    return Object.keys(e).length === 0;
   659→   659→  };

   662→   662→    const e: Record<string, string> = {};

   669→   669→    else if (form.cvc.length < 3) e.cvc = "Invalid CVC";
   670→   670→    setErrors(e);
   671→   671→    return Object.keys(e).length === 0;
   672→   672→  };

   682→   682→    } else {
   683→   683→      setStep("payment");

   685→   685→    }
   686→   686→  };
   687→   687→
   688→   688→  return (
   689→   689→    <Dialog open={open} onOpenChange={onOpenChange}>
   690→   690→      <DialogContent className="glass-strong max-h-[92vh] max-w-lg overflow-y-auto">

   693→   693→          <StepDot n={1} active={step === "details"} done={step !== "details"} />
   694→   694→          <div className={cn("h-px flex-1 transition", step !== "details" ? "bg-[var(--gold)]" : "bg-border")} />
   695→   695→          <StepDot n={2} active={step === "payment" || step === "activating"} done={step === "activating" || step === "done"} />
   696→   696→          <div className={cn("h-px flex-1 transition", step === "done" ? "bg-[var(--gold)]" : "bg-border")} />
   697→   697→          <StepDot n={3} active={step === "done"} done={step === "done"} />
   698→   698→        </div>
   699→   699→
   700→   700→        {step === "details" && (
