   132→/** Build a tiny sparkline dataset deterministically from the account id. */
   133→function buildSpark(id: string, growth: number) {
   134→  let h = 0;
   135→  for (let i = 0; i < id.length; i++) h = id.charCodeAt(i) + ((h << 5) - h);
   136→  const points = 12;
   137→  const base = 50 + (Math.abs(h) % 30);
   138→  const arr: { i: number; v: number }[] = [];
   139→  for (let i = 0; i < points; i++) {
   140→    const wave = Math.sin((i + h) * 0.7) * 12 + Math.cos((i + h) * 0.4) * 6;
   141→    const trend = (i / points) * (growth > 0 ? 30 : -10);
   142→    arr.push({ i, v: Math.max(2, base + wave + trend) });
   143→  }
   144→  return arr;
   145→}

   293→interface ConnectAccountDialogProps {

   299→function ConnectAccountDialog({ open, onOpenChange, onConnected }: ConnectAccountDialogProps) {
   300→  const { user } = useAuth();
   301→  const [step, setStep] = React.useState<Step>("platform");
   302→  const [platform, setPlatform] = React.useState<PlatformId | null>(null);
   303→  const [form, setForm] = React.useState({
   304→    handle: "",
   305→    displayName: "",
   306→    loginEmail: "",
   307→    password: "",
   308→    apiToken: "",
   309→    businessAccount: true,

   311→  const [errors, setErrors] = React.useState<Record<string, string>>({});
   312→
   313→  // Reset when closed
   314→  React.useEffect(() => {
   315→    if (!open) {
   316→      const t = setTimeout(() => {
   317→        setStep("platform");
   318→        setPlatform(null);
   319→        setForm({
   320→          handle: "",
   321→          displayName: "",
   322→          loginEmail: "",
   323→          password: "",
   324→          apiToken: "",
   325→          businessAccount: true,
   326→        });
   327→        setErrors({});
   328→      }, 300);
   329→      return () => clearTimeout(t);
   330→    }
   331→  }, [open]);
   332→
   333→  const p = platform ? PLATFORMS[platform] : null;
   334→
   335→  const pickPlatform = (id: PlatformId) => {
   336→    setPlatform(id);
   337→    setStep("credentials");
   338→    // prefill handle hint
   339→    const plat = PLATFORMS[id];
   340→    setForm((f) => ({ ...f, handle: f.handle || `@yourbrand` , displayName: f.displayName || "Your Brand" }));
   341→    void plat;
   342→  };
   343→
   344→  const validate = () => {
   345→    const e: Record<string, string> = {};
   346→    if (!form.handle.trim()) e.handle = "Handle is required";
   347→    else if (!form.handle.startsWith("@")) e.handle = "Handle must start with @";
   348→    if (!form.displayName.trim()) e.displayName = "Display name is required";
   349→    if (!form.loginEmail.trim()) e.loginEmail = "Login email is required";
   350→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.loginEmail)) e.loginEmail = "Invalid email format";
   351→    if (!form.password.trim()) e.password = "Password is required";
   352→    else if (form.password.length < 6) e.password = "Password must be at least 6 characters";
   353→    setErrors(e);
   354→    return Object.keys(e).length === 0;
   355→  };
   356→
   357→  const connect = async () => {
   358→    if (!validate() || !platform) return;
   359→    setStep("connecting");
   360→    // Simulate the OAuth/credential verification + account fetch
   361→    await new Promise((r) => setTimeout(r, 2200));
   362→    // Persist the new account via API
   363→    try {
   364→      const res = await fetch("/api/accounts", {
   365→        method: "POST",
   366→        headers: { "Content-Type": "application/json" },
   367→        body: JSON.stringify({
   368→          platform,
   369→          handle: form.handle,
   370→          displayName: form.displayName,
   371→          loginEmail: form.loginEmail,
   372→          businessAccount: form.businessAccount,
   373→          ownerId: user?.id ?? "u-admin",
   374→        }),
   375→      });
   376→      if (res.ok) {
   377→        setStep("success");
   378→        setTimeout(() => onConnected(), 1400);
   379→      } else {
   380→        toast.error("Failed to connect account");
   381→        setStep("credentials");
   382→      }
   383→    } catch {
   384→      toast.error("Network error");
   385→      setStep("credentials");
   386→    }
   387→  };
   388→
   389→  return (
   390→    <Dialog open={open} onOpenChange={onOpenChange}>
   391→      <DialogContent className="glass-strong max-h-[92vh] overflow-y-auto sm:max-w-lg">
   392→        {/* Step indicator */}
   393→        <div className="flex items-center gap-2 pb-1">
   394→          <StepDot active={step !== "platform" || true} done={step !== "platform"} label="1" />
   395→          <div className={cn("h-px flex-1 transition", step !== "platform" ? "bg-primary" : "bg-border")} />
   396→          <StepDot active={step === "credentials" || step === "connecting" || step === "success"} done={step === "connecting" || step === "success"} label="2" />
   397→          <div className={cn("h-px flex-1 transition", step === "success" ? "bg-primary" : "bg-border")} />
   398→          <StepDot active={step === "success"} done={step === "success"} label="3" />
   399→        </div>
   400→
   401→        {/* STEP 1 — platform picker */}
   402→        {step === "platform" && (
   403→          <>
   404→            <DialogHeader>
   405→              <DialogTitle className="flex items-center gap-2 text-lg">
   406→                <Plus className="h-5 w-5 text-primary" />
   407→                Connect a new account
   408→              </DialogTitle>
   409→              <DialogDescription>
   410→                Choose a platform. You&apos;ll enter your credentials on the next step.
   411→              </DialogDescription>
   412→            </DialogHeader>
   413→            <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3">
   414→              {PLATFORM_LIST.map((p) => (
   415→                <button
   416→                  key={p.id}
   417→                  onClick={() => pickPlatform(p.id)}
   418→                  className="lift-hover group flex flex-col items-center gap-2 rounded-2xl border border-border/40 bg-muted/30 p-4 transition hover:border-primary/40 hover:bg-primary/5"
   419→                >
   420→                  <div className="transition group-hover:scale-110">
   421→                    <PlatformBadge platform={p.id} size={44} />
   422→                  </div>
   423→                  <span className="text-sm font-medium">{p.name}</span>
   424→                </button>
   425→              ))}
   426→            </div>
   427→          </>
   428→        )}
   429→
   430→        {/* STEP 2 — credentials */}
   431→        {step === "credentials" && p && (
   432→          <>
   433→            <DialogHeader>
   434→              <DialogTitle className="flex items-center gap-2.5 text-lg">
