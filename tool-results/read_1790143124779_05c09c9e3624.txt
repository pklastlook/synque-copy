   196→  const [genOverride, setGenOverride] = React.useState<Record<string, string>>({});
   197→
   198→  const reports = data ?? [];
   199→  const selected = reports.find((r) => r.id === selectedId) ?? null;
   200→  const selectedGenerated = selected
   201→    ? genOverride[selected.id] ?? selected.lastGenerated
   202→    : null;
   203→
   204→  const totals = React.useMemo(() => {
   205→    const scheduled = reports.filter((r) => r.schedule).length;
   206→    const types = new Set(reports.map((r) => r.type)).size;
   207→    const now = new Date();
   208→    const dayMs = 86_400_000;
   209→    const weekAgo = new Date(now.getTime() - 7 * dayMs);
   210→    const generatedThisWeek = reports.filter(
   211→      (r) => (genOverride[r.id] ?? r.lastGenerated) && new Date(genOverride[r.id] ?? r.lastGenerated!) >= weekAgo,
   212→    ).length;
   213→    return {
   214→      total: reports.length,
   215→      scheduled,
   216→      types,
   217→      generatedThisWeek,
   218→    };

   230→        method: "POST",

   233→          name: tpl.name,
   234→          type: tpl.type,
   235→          description: tpl.desc,
   236→          config: JSON.stringify(config),
   237→          schedule: tpl.schedule ?? null,
   238→        }),

   570→}: {
   571→  report: Report;
   572→  lastGenerated: string | null;
   573→}) {
   574→  const meta = TYPE_META[report.type];
   575→  const Icon = meta.icon;
   576→
   577→  let config: ReportConfig = {};
   578→  try {
   579→    config = JSON.parse(report.config) as ReportConfig;
   580→  } catch {
   581→    config = {};
   582→  }
   583→
   584→  // Build a deterministic mock series from the report id.
   585→  const series = React.useMemo(() => {
   586→    let h = 0;
   587→    for (let i = 0; i < report.id.length; i++) h = report.id.charCodeAt(i) + ((h << 5) - h);
   588→    const days = config.dateRange === "7d" ? 7 : config.dateRange === "14d" ? 14 : config.dateRange === "90d" ? 30 : 30;
   589→    const out: { date: string; value: number }[] = [];
   590→    const now = Date.now();
   591→    for (let i = days - 1; i >= 0; i--) {
   592→      const wave = Math.sin((i + h) * 0.6) * 18 + Math.cos((i + h) * 0.3) * 9;
   593→      const trend = ((days - i) / days) * 30;
   594→      out.push({
   595→        date: new Date(now - i * 86_400_000).toISOString().slice(0, 10),
   596→        value: Math.max(20, Math.round(80 + wave + trend + (Math.abs(h) % 30))),
   597→      });
   598→    }
   599→    return out;
   600→  }, [report.id, config.dateRange]);
   601→
   602→  const metricRows = (config.metrics && config.metrics.length > 0
   603→    ? config.metrics
   604→    : ["impressions", "reach", "engagement", "followers"]
   605→  ).slice(0, 5);
   606→
   607→  return (
   608→    <GlassCard className="animate-rise overflow-hidden p-5">
   609→      <SectionTitle
   610→        title={report.name}
   611→        desc={`${meta.label} report · ${config.dateRange ?? "30d"} range · ${lastGenerated ? `Generated ${timeAgo(lastGenerated)}` : "Not generated yet"}`}
   612→        action={
   613→          <div
   614→            className="flex items-center gap-2 rounded-full px-2.5 py-1 text-[11px] font-semibold text-white shadow"
   615→            style={{ background: meta.color }}
   616→          >
   617→            <Icon className="h-3.5 w-3.5" />
   618→            {meta.label}
   619→          </div>
   620→        }
   621→      />
   622→
   623→      <div className="grid gap-4 lg:grid-cols-3">
   624→        <div className="lg:col-span-2">
   625→          <div className="rounded-2xl border border-border/40 bg-background/40 p-4">
   626→            <div className="mb-2 flex items-center justify-between">
   627→              <span className="text-xs font-medium text-muted-foreground">
   628→                {metricRows[0] ?? "value"} trend
   629→              </span>
   630→              <span className="text-[11px] font-semibold text-emerald-500">
   631→                +{((series[series.length - 1]?.value ?? 0) - (series[0]?.value ?? 0)) / Math.max(1, series[0]?.value ?? 1) > 0 ? "+" : ""}
   632→                {Math.round(
   633→                  (((series[series.length - 1]?.value ?? 0) - (series[0]?.value ?? 0)) /
   634→                    Math.max(1, series[0]?.value ?? 1)) *
   635→                    100,
   636→                )}
   637→                %
   638→              </span>
   639→            </div>
   640→            <div className="h-[200px] w-full">
   641→              <ResponsiveContainer width="100%" height="100%">
   642→                <AreaChart data={series} margin={{ left: -16, right: 4, top: 4, bottom: 0 }}>
   643→                  <defs>
   644→                    <linearGradient id="prevGrad" x1="0" y1="0" x2="0" y2="1">
   645→                      <stop offset="0%" stopColor={meta.color} stopOpacity={0.45} />
   646→                      <stop offset="100%" stopColor={meta.color} stopOpacity={0} />
   647→                    </linearGradient>
   648→                  </defs>
   649→                  <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   650→                  <XAxis
   651→                    dataKey="date"
   652→                    tickFormatter={(d) =>
   653→                      new Date(d).toLocaleDateString(undefined, { day: "numeric", month: "short" })
   654→                    }
   655→                    tick={{ fontSize: 10, fill: "var(--muted-foreground)" }}
   656→                    axisLine={false}
   657→                    tickLine={false}
   658→                    interval={Math.max(0, Math.floor(series.length / 6))}
   659→                  />
   660→                  <YAxis
   661→                    tickFormatter={(v) => fmt(Number(v))}
   662→                    tick={{ fontSize: 10, fill: "var(--muted-foreground)" }}
   663→                    axisLine={false}
   664→                    tickLine={false}
   665→                    width={44}
   666→                  />
   667→                  <Tooltip
   668→                    contentStyle={TOOLTIP_STYLE}
   669→                    labelFormatter={(d) =>
   670→                      new Date(d).toLocaleDateString(undefined, {
   671→                        weekday: "short",
   672→                        month: "short",
   673→                        day: "numeric",
   674→                      })
   675→                    }
   676→                    formatter={(v: number) => [fmt(v), "Value"]}
   677→                  />
   678→                  <Area
   679→                    type="monotone"
   680→                    dataKey="value"
   681→                    stroke={meta.color}
   682→                    strokeWidth={2.5}
   683→                    fill="url(#prevGrad)"
   684→                  />
   685→                </AreaChart>
   686→              </ResponsiveContainer>
   687→            </div>
   688→          </div>
   689→        </div>
   690→
   691→        <div className="rounded-2xl border border-border/40 bg-background/40 p-4">
   692→          <div className="mb-2 flex items-center gap-2 text-xs font-medium text-muted-foreground">
   693→            <CheckCircle2 className="h-3.5 w-3.5 text-primary" /> Metrics snapshot
   694→          </div>
   695→          <div className="space-y-1.5">
   696→            {metricRows.map((m, i) => {
   697→              const base = 40 + (i * 7 + (report.id.charCodeAt(0) % 5)) * 12;
   698→              const prev = Math.max(10, base - 8);
   699→              const delta = Math.round(((base - prev) / prev) * 100);
   700→              return (
   701→                <div
   702→                  key={m}
   703→                  className="flex items-center justify-between rounded-xl bg-muted/30 px-3 py-2 text-xs"
   704→                >
   705→                  <span className="capitalize text-muted-foreground">{m}</span>
   706→                  <div className="flex items-center gap-2">
   707→                    <span className="font-semibold tabular-nums">{fmt(base * 1000)}</span>
   708→                    <span
   709→                      className={cn(
   710→                        "rounded-full px-1.5 py-0.5 text-[10px] font-semibold",
   711→                        delta >= 0
   712→                          ? "bg-emerald-500/15 text-emerald-500"
   713→                          : "bg-rose-500/15 text-rose-500",
   714→                      )}
   715→                    >
   716→                      {delta >= 0 ? "+" : ""}
   717→                      {delta}%
   718→                    </span>
   719→                  </div>
   720→                </div>
   721→              );
   722→            })}
   723→          </div>
   724→        </div>
   725→      </div>
   726→    </GlassCard>
   727→  );
   728→}

   730→function CreateReportDialog({
   731→  open,
   732→  onOpenChange,
   733→  accounts,
   734→  onCreated,
   735→}: {
   736→  open: boolean;
   737→  onOpenChange: (v: boolean) => void;
   738→  accounts: AccountOption[];
   739→  onCreated: () => void;
   740→}) {
   741→  const [name, setName] = React.useState("");
   742→  const [type, setType] = React.useState<ReportType>("performance");
   743→  const [description, setDescription] = React.useState("");
   744→  const [metrics, setMetrics] = React.useState<string[]>(["impressions", "engagement"]);
   745→  const [dateRange, setDateRange] = React.useState<string>("30d");
   746→  const [schedule, setSchedule] = React.useState<string>("none");
   747→  const [accountsFilter, setAccountsFilter] = React.useState<string>("all");
   748→  const [submitting, setSubmitting] = React.useState(false);
   749→
   750→  const toggleMetric = (m: string) => {
   751→    setMetrics((prev) =>
   752→      prev.includes(m) ? prev.filter((x) => x !== m) : [...prev, m],
   753→    );
   754→  };
   755→
   756→  const reset = () => {
   757→    setName("");
   758→    setType("performance");
   759→    setDescription("");
   760→    setMetrics(["impressions", "engagement"]);
   761→    setDateRange("30d");
   762→    setSchedule("none");
   763→    setAccountsFilter("all");
   764→  };
   765→
   766→  const submit = async () => {
   767→    if (!name.trim()) {
   768→      toast.error("Report name is required");
   769→      return;
   770→    }
   771→    if (metrics.length === 0) {
   772→      toast.error("Pick at least one metric");
   773→      return;
   774→    }
   775→    setSubmitting(true);
   776→    const config: ReportConfig = {
   777→      metrics,
   778→      dateRange,
   779→      accounts: accountsFilter,
   780→    };
   781→    try {
   782→      const res = await fetch("/api/reports", {
   783→        method: "POST",
   784→        headers: { "Content-Type": "application/json" },
   785→        body: JSON.stringify({
   786→          name: name.trim(),
   787→          type,
   788→          description: description.trim() || null,
   789→          config: JSON.stringify(config),
   790→          schedule: schedule === "none" ? null : schedule,
   791→        }),
   792→      });
   793→      if (!res.ok) throw new Error("Failed");
   794→      toast.success("Report created");
   795→      reset();
   796→      onCreated();
   797→    } catch {
   798→      toast.error("Could not create report");
   799→    } finally {
   800→      setSubmitting(false);
   801→    }
   802→  };
   803→
   804→  return (
   805→    <Dialog open={open} onOpenChange={onOpenChange}>
   806→      <DialogContent className="max-w-xl rounded-3xl">
   807→        <DialogHeader>
   808→          <DialogTitle className="flex items-center gap-2">
   809→            <Wand2 className="h-4 w-4 text-primary" /> New report
   810→          </DialogTitle>
   811→          <DialogDescription>
   812→            Configure the metrics, range, and schedule for your custom report.
   813→          </DialogDescription>
   814→        </DialogHeader>
   815→
   816→        <div className="max-h-[60vh] space-y-4 overflow-y-auto scroll-liquid pr-1">
   817→          <div className="space-y-2">
   818→            <Label htmlFor="r-name">Report name</Label>
   819→            <Input
   820→              id="r-name"
   821→              placeholder="e.g. Q3 Instagram performance"
   822→              value={name}
   823→              onChange={(e) => setName(e.target.value)}
   824→            />
   825→          </div>
   826→
   827→          <div className="grid grid-cols-2 gap-3">
   828→            <div className="space-y-2">
   829→              <Label>Type</Label>
   830→              <Select value={type} onValueChange={(v) => setType(v as ReportType)}>
   831→                <SelectTrigger className="w-full">
   832→                  <SelectValue />
   833→                </SelectTrigger>
   834→                <SelectContent>
   835→                  {(Object.keys(TYPE_META) as ReportType[]).map((t) => {
   836→                    const M = TYPE_META[t];
   837→                    return (
   838→                      <SelectItem key={t} value={t}>
   839→                        {M.label}
   840→                      </SelectItem>
   841→                    );
   842→                  })}
   843→                </SelectContent>
   844→              </Select>
   845→            </div>
   846→            <div className="space-y-2">
   847→              <Label>Date range</Label>
   848→              <Select value={dateRange} onValueChange={setDateRange}>
   849→                <SelectTrigger className="w-full">
   850→                  <SelectValue />
   851→                </SelectTrigger>
   852→                <SelectContent>
   853→                  {DATE_RANGES.map((d) => (
   854→                    <SelectItem key={d} value={d}>
   855→                      Last {d}
   856→                    </SelectItem>
   857→                  ))}
   858→                </SelectContent>
   859→              </Select>
   860→            </div>
   861→          </div>
   862→
   863→          <div className="space-y-2">
   864→            <Label>Description</Label>
   865→            <Textarea
   866→              rows={2}
   867→              placeholder="What does this report show?"
   868→              value={description}
   869→              onChange={(e) => setDescription(e.target.value)}
   870→            />
   871→          </div>
   872→
   873→          <div className="space-y-2">
   874→            <Label>Metrics</Label>
   875→            <div className="flex flex-wrap gap-2">
   876→              {METRIC_OPTIONS.map((m) => {
   877→                const active = metrics.includes(m);
   878→                return (
   879→                  <button
   880→                    key={m}
   881→                    type="button"
   882→                    onClick={() => toggleMetric(m)}
   883→                    className={cn(
   884→                      "rounded-full border px-3 py-1.5 text-xs font-medium transition-all",
   885→                      active
   886→                        ? "border-primary bg-primary text-primary-foreground shadow-sm"
   887→                        : "border-border bg-background/40 text-muted-foreground hover:border-primary/40 hover:text-foreground",
   888→                    )}
   889→                  >
   890→                    {m}
   891→                  </button>
   892→                );
   893→              })}
   894→            </div>
   895→          </div>
   896→
   897→          <div className="grid grid-cols-2 gap-3">
   898→            <div className="space-y-2">
   899→              <Label>Schedule</Label>
   900→              <Select value={schedule} onValueChange={setSchedule}>
   901→                <SelectTrigger className="w-full">
   902→                  <SelectValue />
   903→                </SelectTrigger>
   904→                <SelectContent>
   905→                  {SCHEDULES.map((s) => (
   906→                    <SelectItem key={s} value={s}>
   907→                      {s === "none" ? "No schedule" : s.charAt(0).toUpperCase() + s.slice(1)}
   908→                    </SelectItem>
   909→                  ))}
   910→                </SelectContent>
   911→              </Select>
   912→            </div>
   913→            <div className="space-y-2">
   914→              <Label>Accounts</Label>
   915→              <Select value={accountsFilter} onValueChange={setAccountsFilter}>
   916→                <SelectTrigger className="w-full">
   917→                  <SelectValue />
   918→                </SelectTrigger>
   919→                <SelectContent>
   920→                  <SelectItem value="all">All accounts</SelectItem>
   921→                  {accounts.map((a) => (
   922→                    <SelectItem key={a.id} value={a.id}>
   923→                      {a.displayName}
   924→                    </SelectItem>
   925→                  ))}
   926→                </SelectContent>
   927→              </Select>
   928→            </div>
   929→          </div>
   930→        </div>
   931→
   932→        <DialogFooter className="pt-2">
   933→          <button
   934→            type="button"
   935→            onClick={() => onOpenChange(false)}
   936→            className="rounded-xl bg-muted/50 px-4 py-2 text-sm font-medium text-muted-foreground transition hover:bg-muted hover:text-foreground"
   937→          >
   938→            Cancel
   939→          </button>
   940→          <button
   941→            type="button"
   942→            onClick={submit}
   943→            disabled={submitting}
   944→            className="liquid-pill flex items-center gap-2 rounded-xl px-4 py-2 text-sm font-semibold disabled:opacity-60"
   945→          >
   946→            <Plus className="h-4 w-4" />
   947→            {submitting ? "Creating…" : "Create report"}
   948→          </button>
   949→        </DialogFooter>
   950→      </DialogContent>
   951→    </Dialog>
   952→  );
   953→}
