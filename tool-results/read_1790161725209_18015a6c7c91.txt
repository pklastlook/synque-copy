   378→function MailboxInbox({ mailboxId, onClose }: { mailboxId: string; onClose: () => void }) {
   379→  const { data: messages, reload } = useApi<MessageInfo[]>(`/api/mail/messages?mailboxId=${mailboxId}`);
   380→  const [folder, setFolder] = React.useState("inbox");
   381→  const [selected, setSelected] = React.useState<string | null>(null);
   382→  const [composing, setComposing] = React.useState(false);
   383→
   384→  const all = messages ?? [];
   385→  const filtered = folder === "all" ? all : all.filter((m) => m.folder === folder);
   386→  const selectedMsg = filtered.find((m) => m.id === selected) ?? null;
   387→  const unread = all.filter((m) => m.folder === "inbox" && !m.isRead).length;
   388→
   389→  const markRead = async (id: string) => {
   390→    await fetch("/api/mail/messages", { method: "PATCH", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ id, isRead: true }) });
   391→    reload();
   392→  };
   393→
   394→  const FOLDERS = [
   395→    { id: "inbox", label: "Inbox", icon: InboxIcon, count: unread },
   396→    { id: "starred", label: "Starred", icon: Star, count: all.filter((m) => m.folder === "starred").length },
   397→    { id: "sent", label: "Sent", icon: Send, count: all.filter((m) => m.folder === "sent").length },
   398→  ];
   399→
   400→  return (
   401→    <GlassCard strong className="overflow-hidden">
   402→      <div className="flex items-center gap-3 border-b border-border/40 p-4">
   403→        <button onClick={onClose} className="grid h-9 w-9 place-items-center rounded-xl bg-muted/50 hover:bg-muted" aria-label="Back">
   404→          <ArrowLeft className="h-4 w-4" />
   405→        </button>
   406→        <div className="min-w-0 flex-1">
   407→          <div className="truncate text-sm font-semibold">Mailbox inbox</div>
   408→          <div className="text-[11px] text-muted-foreground">{unread} unread · {all.length} total</div>
   409→        </div>
   410→        <button onClick={() => setComposing(true)} className="liquid-pill glow-ring flex items-center gap-1.5 rounded-xl px-3 py-2 text-xs font-semibold">
   411→          <PenSquare className="h-3.5 w-3.5" /> Compose
   412→        </button>
   413→      </div>
   414→
   415→      <div className="grid gap-0 lg:grid-cols-[200px_320px_1fr]">
   416→        {/* Folders */}
   417→        <div className="hidden border-r border-border/40 p-3 lg:block">
   418→          {FOLDERS.map((f) => {
   419→            const Icon = f.icon;
   420→            const active = folder === f.id;
   421→            return (
   422→              <button
   423→                key={f.id}
   424→                onClick={() => { setFolder(f.id); setSelected(null); }}
   425→                className={cn(
   426→                  "flex w-full items-center gap-2 rounded-xl px-3 py-2 text-sm transition",
   427→                  active ? "bg-primary/15 text-primary" : "text-muted-foreground hover:bg-muted/50 hover:text-foreground"
   428→                )}
   429→              >
   430→                <Icon className="h-4 w-4" />
   431→                <span className="flex-1 text-left">{f.label}</span>
   432→                {f.count > 0 && (
   433→                  <span className={cn("rounded-full px-1.5 py-0.5 text-[10px] font-bold", active ? "bg-primary text-primary-foreground" : "bg-muted text-muted-foreground")}>
   434→                    {f.count}
   435→                  </span>
   436→                )}
   437→              </button>
   438→            );
   439→          })}
   440→        </div>
   441→
   442→        {/* Message list */}
   443→        <div className="scroll-liquid max-h-[500px] overflow-y-auto border-r border-border/40 lg:max-h-[600px]">
   444→          {filtered.length === 0 ? (
   445→            <div className="grid place-items-center py-12 text-center text-sm text-muted-foreground">
   446→              <InboxIcon className="mb-2 h-8 w-8 opacity-40" />
   447→              No messages in {folder}
   448→            </div>
   449→          ) : (
   450→            filtered.map((m) => (
   451→              <button
   452→                key={m.id}
   453→                onClick={() => { setSelected(m.id); if (!m.isRead) markRead(m.id); }}
   454→                className={cn(
   455→                  "flex w-full flex-col gap-1 border-b border-border/30 p-3 text-left transition hover:bg-muted/40",
   456→                  selected === m.id && "bg-primary/10",
   457→                  !m.isRead && m.direction === "in" && "bg-primary/5"
   458→                )}
   459→              >
   460→                <div className="flex items-center gap-2">
   461→                  {!m.isRead && m.direction === "in" && <span className="h-1.5 w-1.5 shrink-0 rounded-full bg-primary" />}
   462→                  {m.isStarred && <Star className="h-3 w-3 shrink-0 fill-amber-400 text-amber-400" />}
   463→                  <span className="truncate text-sm font-medium">{m.direction === "out" ? `To: ${m.toAddr}` : m.fromAddr}</span>
   464→                  <span className="ml-auto shrink-0 text-[10px] text-muted-foreground">{timeAgo(m.date)}</span>
   465→                </div>
   466→                <div className="truncate text-xs font-medium">{m.subject}</div>
   467→                <div className="truncate text-[11px] text-muted-foreground">{m.body.replace(/\n/g, " ").slice(0, 80)}</div>
   468→              </button>
   469→            ))
