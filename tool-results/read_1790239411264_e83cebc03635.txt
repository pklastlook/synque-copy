   106→   106→  function getToken(): string | null {

   108→   108→      return JSON.parse(localStorage.getItem("socialflow-auth") ?? "{}")?.state?.token ?? null;
   109→   109→    } catch { return null; }

   112→   112→  async function generateReport() {
   113→   113→    if (!reportName.trim()) {
   114→   114→      toast.error("Enter a report name");
   115→   115→      return;
   116→   116→    }

   118→   118→    try {
   119→   119→      const token = getToken();
   120→   120→      const days = period === "custom" ? customDays : PERIOD_META[period].days;
   121→   121→      const config = JSON.stringify({
   122→   122→        period, scope, days,
   123→   123→        memberId: scope === "individual" ? selectedMemberId : null,
   124→   124→        accountId: scope === "account" ? selectedAccountId : null,
   125→   125→      });

   127→   127→      const res = await fetch("/api/reports", {
   128→   128→        method: "POST",
   129→   129→        headers: {
   130→   130→          "Content-Type": "application/json",
   131→   131→          ...(token ? { "Authorization": `Bearer ${token}` } : {}),
   132→   132→        },
   133→   133→        body: JSON.stringify({
   134→   134→          name: reportName.trim(),
   135→   135→          type: reportType,
   136→   136→          description: `${PERIOD_META[period].label} · ${SCOPE_META[scope].label} · Last ${days} days`,
   137→   137→          config,
   138→   138→          createdBy: null,
   139→   139→        }),
   140→   140→      });
   141→   141→
   142→   142→      if (res.ok) {
   143→   143→        const json = await res.json();
   144→   144→        toast.success("Report generated");
   145→   145→
   146→   146→        // Fetch analytics data for preview
   147→   147→        const analyticsRes = await fetch(`/api/analytics?days=${days}&accountId=all`, {
   148→   148→          headers: token ? { "Authorization": `Bearer ${token}` } : {},
   149→   149→        });
   150→   150→        const analyticsJson = await analyticsRes.json();
   151→   151→        setPreviewData(analyticsJson);
   152→   152→        setPreviewReport({
   153→   153→          id: json.id,
   154→   154→          name: reportName.trim(),
   155→   155→          type: reportType,
   156→   156→          description: `${PERIOD_META[period].label} · ${SCOPE_META[scope].label} · Last ${days} days`,
   157→   157→          config, schedule: null, createdBy: null,
   158→   158→          lastGenerated: new Date().toISOString(), createdAt: new Date().toISOString(),
   159→   159→        });
   160→   160→        setShowGenerator(false);
   161→   161→        setReportName("");
   162→   162→        reload();

   164→   164→        toast.error("Failed to generate report");
   165→   165→      }
   166→   166→    } catch {
   167→   167→      toast.error("Network error");
   168→   168→    } finally {
   169→   169→      setGenerating(false);
   170→   170→    }
   171→   171→  }

   189→   189→  async function deleteReport(id: string) {

   192→   192→    await fetch(`/api/reports/${id}`, {
   193→   193→      method: "DELETE",
   194→   194→      headers: token ? { "Authorization": `Bearer ${token}` } : {},

   205→   205→        headers: token ? { "Authorization": `Bearer ${token}` } : {},

   212→   212→      a.download = `${r.name.replace(/[^a-z0-9]/gi, "_")}.pdf`;
   213→   213→      a.click();

   215→   215→      toast.success("PDF downloaded");
   216→   216→    } catch {
   217→   217→      toast.error("Failed to generate PDF");
   218→   218→    }
   219→   219→  }
