   130→export function ComposerView() {
   131→  const { composerSeed, clearComposerSeed } = useNav();
   132→  const { user } = useAuth();
   133→  const currentUserId = user?.id ?? "u-admin";
   134→  const { data: accounts } = useApi<AccountData[]>("/api/accounts");
   135→
   136→  const [state, setState] = React.useState<ComposerState>(DEFAULT_STATE);
   137→  const [generating, setGenerating] = React.useState(false);
   138→  const [publishing, setPublishing] = React.useState(false);
   139→  const [search, setSearch] = React.useState("");
   140→  const [uploadOpen, setUploadOpen] = React.useState(false);
   141→  const [urlInput, setUrlInput] = React.useState("");
   142→
   143→  const initRef = React.useRef(false);
   144→
   145→  // Preselect 4 diverse accounts on first load
   146→  React.useEffect(() => {
   147→    if (initRef.current || !accounts || accounts.length === 0) return;
   148→    initRef.current = true;
   149→    const presets = ["ac-ig-1", "ac-x-1", "ac-li-1", "ac-tt-1"].filter((id) =>
   150→      accounts.some((a) => a.id === id)
   151→    );
   152→    if (presets.length) {
   153→      setState((s) => ({ ...s, selectedAccountIds: presets }));
   154→    }
   155→  }, [accounts]);
   156→
   157→  // Composer seed prefill — set caption + jump to step 2 (keeps existing behavior)
   158→  React.useEffect(() => {
   159→    if (composerSeed) {
   160→      setState((s) => ({ ...s, caption: composerSeed, step: 2 }));
   161→      clearComposerSeed();
   162→    }
   163→  }, [composerSeed, clearComposerSeed]);
   164→
   165→  const update = React.useCallback((patch: Partial<ComposerState>) => {
   166→    setState((s) => ({ ...s, ...patch }));
   167→  }, []);
   168→
   169→  const accountsList = accounts ?? [];
   170→  const selectedAccounts = accountsList.filter((a) =>
   171→    state.selectedAccountIds.includes(a.id)
   172→  );
   173→  const filteredAccounts = accountsList.filter((a) => {
   174→    const q = search.trim().toLowerCase();
   175→    if (!q) return true;
   176→    return (
   177→      a.displayName.toLowerCase().includes(q) ||
   178→      a.handle.toLowerCase().includes(q) ||
   179→      PLATFORMS[a.platform].name.toLowerCase().includes(q)
   180→    );
   181→  });
   182→
   183→  function toggleAccount(id: string) {
   184→    setState((s) => {
   185→      const next = new Set(s.selectedAccountIds);
   186→      if (next.has(id)) next.delete(id);
   187→      else next.add(id);
   188→      return { ...s, selectedAccountIds: Array.from(next) };
   189→    });
   190→  }
   191→
   192→  const selectAll = () =>
   193→    setState((s) => ({ ...s, selectedAccountIds: accountsList.map((a) => a.id) }));
   194→  const clearAll = () => setState((s) => ({ ...s, selectedAccountIds: [] }));
   195→
   196→  const goToStep = (step: Step) => setState((s) => ({ ...s, step }));
   197→  const next = () =>
   198→    setState((s) => ({ ...s, step: Math.min(5, (s.step as number) + 1) as Step }));
   199→  const back = () =>
   200→    setState((s) => ({ ...s, step: Math.max(1, (s.step as number) - 1) as Step }));
   201→
   202→  async function generate(regenerateOnly?: string) {
   203→    if (!state.caption.trim()) {
   204→      toast.error("Write a caption first");
   205→      return;
   206→    }
   207→    if (selectedAccounts.length === 0) {
   208→      toast.error("Select at least one account");
   209→      return;
   210→    }
   211→    setGenerating(true);
   212→    const targets = selectedAccounts.map((a) => ({
   213→      accountId: a.id,
   214→      platform: a.platform,
   215→      handle: a.handle,
   216→    }));
   217→    const bodyTargets = regenerateOnly
   218→      ? targets.filter((t) => t.accountId === regenerateOnly)
   219→      : targets;
   220→    try {
   221→      const res = await fetch("/api/ai/generate", {
   222→        method: "POST",
   223→        headers: { "Content-Type": "application/json" },
   224→        body: JSON.stringify({
   225→          caption: state.caption,
   226→          tone: state.tone,
   227→          targets: bodyTargets,
   228→          mediaHint: state.mediaUrl ? "attached visual" : "no media",
   229→        }),
   230→      });
   231→      const json = await res.json().catch(() => ({}));
   232→      const incoming = json.variants ?? json.fallback?.variants;
   233→      if (incoming && typeof incoming === "object") {
   234→        const normalized: Record<string, Variant> = {};
   235→        for (const k of Object.keys(incoming)) {
   236→          normalized[k] = {
   237→            content: incoming[k]?.content ?? "",
   238→            hashtags: incoming[k]?.hashtags ?? "",
   239→          };
   240→        }
   241→        if (regenerateOnly) {
   242→          const nv = normalized[regenerateOnly];
   243→          if (nv) {
   244→            setState((s) => ({
   245→              ...s,
   246→              variants: { ...s.variants, [regenerateOnly]: nv },
   247→            }));
   248→            toast.success("Regenerated variant");
   249→          } else {
   250→            toast.error("Variant not found in AI response");
   251→          }
   252→        } else {
   253→          setState((s) => ({ ...s, variants: normalized }));
   254→          toast.success(
   255→            `Generated ${Object.keys(normalized).length} platform-native captions`
   256→          );
   257→        }
   258→      } else {
   259→        toast.error(json.error ?? "Generation failed");
   260→      }
   261→    } catch {
   262→      toast.error("Network error generating captions");
   263→    } finally {
   264→      setGenerating(false);
   265→    }
   266→  }
   267→
   268→  async function publish() {
   269→    if (selectedAccounts.length === 0) {
   270→      toast.error("Select accounts");
   271→      return;
   272→    }
   273→    setPublishing(true);
   274→    try {
   275→      const v: Record<string, { content: string; hashtags?: string }> = {};
   276→      for (const a of selectedAccounts) {
   277→        const ex = state.variants[a.id];
   278→        v[a.id] = ex ?? { content: state.caption, hashtags: "" };
   279→      }
   280→      const scheduledAt =
   281→        state.scheduleEnabled && state.scheduledAt
   282→          ? new Date(state.scheduledAt).toISOString()
   283→          : null;
   284→      const res = await fetch("/api/posts", {
   285→        method: "POST",
   286→        headers: { "Content-Type": "application/json" },
   287→        body: JSON.stringify({
   288→          title: state.caption.slice(0, 50).trim() || "Untitled",
   289→          caption: state.caption,
   290→          mediaUrl: state.mediaUrl || null,
   291→          tone: state.tone,
   292→          scheduledAt,
   293→          accountIds: selectedAccounts.map((a) => a.id),
   294→          variants: v,
   295→          authorId: currentUserId,
   296→        }),
   297→      });
   298→      if (res.ok) {
   299→        toast.success(
   300→          state.scheduleEnabled && scheduledAt
   301→            ? "Post scheduled successfully"
   302→            : "Post published to all accounts"
   303→        );
   304→        setState(DEFAULT_STATE);
   305→      } else {
   306→        toast.error("Failed to publish");
   307→      }
   308→    } catch {
   309→      toast.error("Network error");
   310→    } finally {
   311→      setPublishing(false);
   312→    }
   313→  }
   314→
   315→  function submitUrl() {
   316→    if (!urlInput.trim()) {
   317→      toast.error("Enter an image URL");
   318→      return;
   319→    }
   320→    update({ mediaUrl: urlInput.trim() });
   321→    setUrlInput("");
   322→    setUploadOpen(false);
   323→    toast.success("Media URL added");
   324→  }
   325→
   326→  const canContinue =
   327→    state.step === 1
   328→      ? selectedAccounts.length > 0
   329→      : state.step === 2
   330→      ? state.caption.trim().length > 0
   331→      : state.step === 3
   332→      ? !!state.mediaUrl
   333→      : state.step === 4
   334→      ? true
   335→      : false;
   336→
   337→  return (
   338→    <div className="space-y-5 pb-28">
   339→      <StepIndicator step={state.step} onStep={goToStep} />
   340→
   341→      <div className="mx-auto w-full max-w-3xl space-y-5">
   342→        {state.step === 1 && (
   343→          <Step1
   344→            accounts={filteredAccounts}
   345→            selectedIds={state.selectedAccountIds}
   346→            onToggle={toggleAccount}
   347→            onSelectAll={selectAll}
   348→            onClear={clearAll}
   349→            search={search}
   350→            onSearch={setSearch}
   351→            selectedAccounts={selectedAccounts}
   352→          />
   353→        )}
   354→        {state.step === 2 && (
   355→          <Step2
   356→            caption={state.caption}
   357→            onCaption={(caption) => update({ caption })}
   358→            tone={state.tone}
   359→            onTone={(tone) => update({ tone })}
   360→            onGenerate={() => generate()}
   361→            generating={generating}
   362→            hasVariants={Object.keys(state.variants).length > 0}
   363→            selectedCount={selectedAccounts.length}
   364→          />
   365→        )}
   366→        {state.step === 3 && (
   367→          <Step3
   368→            mediaUrl={state.mediaUrl}
   369→            mediaType={state.mediaType}
   370→            altText={state.altText}
   371→            onMedia={(mediaUrl) => update({ mediaUrl })}
   372→            onMediaType={(mediaType) => update({ mediaType })}
   373→            onAltText={(altText) => update({ altText })}
   374→            uploadOpen={uploadOpen}
   375→            onUploadOpen={setUploadOpen}
   376→            urlInput={urlInput}
   377→            onUrlInput={setUrlInput}
   378→            onUrlSubmit={submitUrl}
   379→          />
   380→        )}
   381→        {state.step === 4 && (
   382→          <Step4
   383→            accounts={selectedAccounts}
   384→            variants={state.variants}
   385→            onVariantChange={(id, v) =>
   386→              setState((s) => ({ ...s, variants: { ...s.variants, [id]: v } }))
   387→            }
   388→            mediaUrl={state.mediaUrl}
   389→            caption={state.caption}
   390→            onRegenerate={(id) => generate(id)}
   391→            onRegenerateAll={() => generate()}
   392→            generating={generating}
   393→          />
   394→        )}
   395→        {state.step === 5 && (
   396→          <Step5
   397→            accounts={selectedAccounts}
   398→            mediaUrl={state.mediaUrl}
   399→            mediaType={state.mediaType}
   400→            tone={state.tone}
   401→            caption={state.caption}
   402→            scheduleEnabled={state.scheduleEnabled}
   403→            onScheduleEnabled={(scheduleEnabled) => update({ scheduleEnabled })}
   404→            scheduledAt={state.scheduledAt}
   405→            onScheduledAt={(scheduledAt) => update({ scheduledAt })}
   406→            hasVariants={Object.keys(state.variants).length > 0}
   407→          />
   408→        )}
   409→      </div>
   410→
   411→      <StepFooter
   412→        step={state.step}
   413→        onBack={back}
   414→        onNext={next}
   415→        canContinue={canContinue}
   416→        onPublish={publish}
   417→        publishing={publishing}
   418→        scheduleEnabled={state.scheduleEnabled}
   419→      />
   420→    </div>
   421→  );
   422→}

   424→/* ============================================================
   425→   Step indicator — sticky, responsive (collapses on mobile)
   426→   ============================================================ */
   427→function StepIndicator({
   428→  step,
   429→  onStep,
   430→}: {
   431→  step: Step;
   432→  onStep: (s: Step) => void;
   433→}) {
   434→  const current = STEPS.find((s) => s.id === step) ?? STEPS[0];
   435→  return (
   436→    <div className="sticky top-0 z-30 -mx-4 px-4 py-2 sm:-mx-6 sm:px-6 lg:-mx-8 lg:px-8">
   437→      <div className="glass-strong glass-sheen rounded-2xl px-4 py-3 sm:px-5">
   438→        {/* Mobile — compact */}
   439→        <div className="flex items-center justify-between gap-3 sm:hidden">
   440→          <div className="flex items-center gap-2.5">
   441→            <span className="grid h-8 w-8 place-items-center rounded-full bg-primary text-xs font-bold text-primary-foreground glow-pulse">
   442→              {step}
   443→            </span>
   444→            <div className="min-w-0">
   445→              <div className="text-[10px] uppercase tracking-wider text-muted-foreground">
   446→                Step {step} of {STEPS.length}
   447→              </div>
   448→              <div className="truncate text-sm font-semibold">{current.label}</div>
   449→            </div>
   450→          </div>
   451→          <div className="flex gap-1">
   452→            {STEPS.map((s) => (
   453→              <span
   454→                key={s.id}
   455→                className={cn(
   456→                  "h-1.5 rounded-full transition-all",
   457→                  s.id === step
   458→                    ? "w-5 bg-primary"
   459→                    : s.id < step
   460→                    ? "w-1.5 bg-primary/60"
   461→                    : "w-1.5 bg-muted-foreground/30"
   462→                )}
   463→              />
   464→            ))}
   465→          </div>
   466→        </div>
   467→
   468→        {/* Desktop — full stepper */}
   469→        <div className="hidden items-center sm:flex">
   470→          {STEPS.map((s, i) => {
   471→            const done = s.id < step;
   472→            const active = s.id === step;
   473→            const clickable = s.id <= step;
   474→            return (
   475→              <React.Fragment key={s.id}>
   476→                <button
   477→                  type="button"
   478→                  disabled={!clickable}
   479→                  onClick={() => clickable && onStep(s.id)}
   480→                  className={cn(
   481→                    "group flex shrink-0 items-center gap-2.5 transition",
   482→                    clickable ? "cursor-pointer" : "cursor-default"
   483→                  )}
   484→                >
   485→                  <span
   486→                    className={cn(
   487→                      "grid h-9 w-9 place-items-center rounded-full border text-sm font-semibold transition-all",
   488→                      active
   489→                        ? "border-primary bg-primary text-primary-foreground glow-pulse"
   490→                        : done
   491→                        ? "border-primary/40 bg-primary/15 text-primary"
   492→                        : "border-border bg-muted/40 text-muted-foreground"
   493→                    )}
   494→                  >
   495→                    {done ? <Check className="h-4 w-4" /> : s.id}
   496→                  </span>
   497→                  <div className="text-left">
   498→                    <div
   499→                      className={cn(
   500→                        "text-xs font-semibold leading-tight",
   501→                        active
   502→                          ? "text-foreground"
   503→                          : done
   504→                          ? "text-foreground/80"
   505→                          : "text-muted-foreground"
   506→                      )}
   507→                    >
   508→                      {s.label}
   509→                    </div>
   510→                  </div>
   511→                </button>
   512→                {i < STEPS.length - 1 && (
   513→                  <span
   514→                    className={cn(
   515→                      "mx-2 h-0.5 flex-1 rounded-full transition-all",
   516→                      s.id < step ? "bg-primary/50" : "bg-border"
   517→                    )}
   518→                  />
   519→                )}
   520→              </React.Fragment>
   521→            );
   522→          })}
   523→        </div>
   524→      </div>
   525→    </div>
   526→  );
   527→}

   532→function Step1({
   533→  accounts,
   534→  selectedIds,
   535→  onToggle,
   536→  onSelectAll,
   537→  onClear,
   538→  search,
   539→  onSearch,
   540→  selectedAccounts,
   541→}: {
   542→  accounts: AccountData[];
   543→  selectedIds: string[];
   544→  onToggle: (id: string) => void;
   545→  onSelectAll: () => void;
   546→  onClear: () => void;
   547→  search: string;
   548→  onSearch: (v: string) => void;
   549→  selectedAccounts: AccountData[];
   550→}) {
   551→  const combinedReach = selectedAccounts.reduce((s, a) => s + a.followers, 0);
   552→  return (
   553→    <GlassCard
   554→      strong
   555→      className="crystal-in glass-sheen relative overflow-hidden p-5 sm:p-7"
   556→      style={{ animationDelay: "0.05s" }}
   557→    >
   558→      <div className="pointer-events-none absolute -right-12 -top-12 h-44 w-44 rounded-full bg-primary/20 blur-3xl" />
   559→      <SectionTitle
   560→        title="Where should we publish?"
   561→        desc="Pick the destination accounts. We'll adapt your message for each platform."
   562→        action={
   563→          <div className="flex items-center gap-2 text-xs">
   564→            <button
   565→              onClick={onSelectAll}
   566→              className="font-medium text-primary hover:underline"
   567→            >
   568→              Select all
   569→            </button>
   570→            <span className="text-muted-foreground/60">·</span>

   573→              className="text-muted-foreground hover:text-foreground"
   574→            >

   579→      />
   580→
   581→      {/* Search */}
   582→      <div className="relative mb-4">
   583→        <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   584→        <input
   585→          value={search}
   586→          onChange={(e) => onSearch(e.target.value)}
   587→          placeholder="Search by name, handle, or platform…"
   588→          className="w-full rounded-xl border border-border/50 bg-muted/30 py-2.5 pl-9 pr-3 text-sm outline-none transition focus:border-primary/50 focus:bg-background/60"
   589→        />
   590→      </div>
   591→
   592→      <div className="grid gap-4 lg:grid-cols-[1fr_240px]">
   593→        {/* Account grid */}
