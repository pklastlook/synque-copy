   130→   130→export function ComposerView() {
   131→   131→  const { composerSeed, clearComposerSeed } = useNav();
   132→   132→  const { user } = useAuth();
   133→   133→  const currentUserId = user?.id ?? "u-admin";
   134→   134→  const { data: accounts } = useApi<AccountData[]>("/api/accounts");
   135→   135→
   136→   136→  const [state, setState] = React.useState<ComposerState>(DEFAULT_STATE);
   137→   137→  const [generating, setGenerating] = React.useState(false);
   138→   138→  const [publishing, setPublishing] = React.useState(false);
   139→   139→  const [search, setSearch] = React.useState("");
   140→   140→  const [uploadOpen, setUploadOpen] = React.useState(false);
   141→   141→  const [urlInput, setUrlInput] = React.useState("");
   142→   142→
   143→   143→  const initRef = React.useRef(false);
   144→   144→
   145→   145→  // Preselect 4 diverse accounts on first load
   146→   146→  React.useEffect(() => {
   147→   147→    if (initRef.current || !accounts || accounts.length === 0) return;
   148→   148→    initRef.current = true;
   149→   149→    const presets = ["ac-ig-1", "ac-x-1", "ac-li-1", "ac-tt-1"].filter((id) =>
   150→   150→      accounts.some((a) => a.id === id)
   151→   151→    );
   152→   152→    if (presets.length) {
   153→   153→      setState((s) => ({ ...s, selectedAccountIds: presets }));
   154→   154→    }
   155→   155→  }, [accounts]);
   156→   156→
   157→   157→  // Composer seed prefill — set caption + jump to step 2 (keeps existing behavior)
   158→   158→  React.useEffect(() => {
   159→   159→    if (composerSeed) {
   160→   160→      setState((s) => ({ ...s, caption: composerSeed, step: 2 }));
   161→   161→      clearComposerSeed();
   162→   162→    }
   163→   163→  }, [composerSeed, clearComposerSeed]);
   164→   164→
   165→   165→  const update = React.useCallback((patch: Partial<ComposerState>) => {
   166→   166→    setState((s) => ({ ...s, ...patch }));
   167→   167→  }, []);
   168→   168→
   169→   169→  const accountsList = accounts ?? [];
   170→   170→  const selectedAccounts = accountsList.filter((a) =>
   171→   171→    state.selectedAccountIds.includes(a.id)
   172→   172→  );
   173→   173→  const filteredAccounts = accountsList.filter((a) => {
   174→   174→    const q = search.trim().toLowerCase();
   175→   175→    if (!q) return true;
   176→   176→    return (
   177→   177→      a.displayName.toLowerCase().includes(q) ||
   178→   178→      a.handle.toLowerCase().includes(q) ||
   179→   179→      PLATFORMS[a.platform].name.toLowerCase().includes(q)
   180→   180→    );
   181→   181→  });
   182→   182→
   183→   183→  function toggleAccount(id: string) {
   184→   184→    setState((s) => {
   185→   185→      const next = new Set(s.selectedAccountIds);
   186→   186→      if (next.has(id)) next.delete(id);
   187→   187→      else next.add(id);
   188→   188→      return { ...s, selectedAccountIds: Array.from(next) };
   189→   189→    });
   190→   190→  }
   191→   191→
   192→   192→  const selectAll = () =>
   193→   193→    setState((s) => ({ ...s, selectedAccountIds: accountsList.map((a) => a.id) }));
   194→   194→  const clearAll = () => setState((s) => ({ ...s, selectedAccountIds: [] }));
   195→   195→
   196→   196→  const goToStep = (step: Step) => setState((s) => ({ ...s, step }));
   197→   197→  const next = () =>
   198→   198→    setState((s) => ({ ...s, step: Math.min(5, (s.step as number) + 1) as Step }));
   199→   199→  const back = () =>
   200→   200→    setState((s) => ({ ...s, step: Math.max(1, (s.step as number) - 1) as Step }));
   201→   201→
   202→   202→  async function generate(regenerateOnly?: string) {
   203→   203→    if (!state.caption.trim()) {
   204→   204→      toast.error("Write a caption first");
   205→   205→      return;
   206→   206→    }
   207→   207→    if (selectedAccounts.length === 0) {
   208→   208→      toast.error("Select at least one account");
   209→   209→      return;
   210→   210→    }
   211→   211→    setGenerating(true);
   212→   212→    const targets = selectedAccounts.map((a) => ({
   213→   213→      accountId: a.id,
   214→   214→      platform: a.platform,
   215→   215→      handle: a.handle,
   216→   216→    }));
   217→   217→    const bodyTargets = regenerateOnly
   218→   218→      ? targets.filter((t) => t.accountId === regenerateOnly)
   219→   219→      : targets;
   220→   220→    try {
   221→   221→      const res = await fetch("/api/ai/generate", {
   222→   222→        method: "POST",
   223→   223→        headers: { "Content-Type": "application/json" },
   224→   224→        body: JSON.stringify({
   225→   225→          caption: state.caption,
   226→   226→          tone: state.tone,
   227→   227→          targets: bodyTargets,
   228→   228→          mediaHint: state.mediaUrl ? "attached visual" : "no media",
   229→   229→        }),
   230→   230→      });
   231→   231→      const json = await res.json().catch(() => ({}));
   232→   232→      const incoming = json.variants ?? json.fallback?.variants;
   233→   233→      if (incoming && typeof incoming === "object") {
   234→   234→        const normalized: Record<string, Variant> = {};
   235→   235→        for (const k of Object.keys(incoming)) {
   236→   236→          normalized[k] = {
   237→   237→            content: incoming[k]?.content ?? "",
   238→   238→            hashtags: incoming[k]?.hashtags ?? "",
   239→   239→          };
   240→   240→        }
   241→   241→        if (regenerateOnly) {
   242→   242→          const nv = normalized[regenerateOnly];
   243→   243→          if (nv) {
   244→   244→            setState((s) => ({
   245→   245→              ...s,
   246→   246→              variants: { ...s.variants, [regenerateOnly]: nv },
   247→   247→            }));
   248→   248→            toast.success("Regenerated variant");
   249→   249→          } else {
   250→   250→            toast.error("Variant not found in AI response");
   251→   251→          }
   252→   252→        } else {
   253→   253→          setState((s) => ({ ...s, variants: normalized }));
   254→   254→          toast.success(
   255→   255→            `Generated ${Object.keys(normalized).length} platform-native captions`
   256→   256→          );
   257→   257→        }
   258→   258→      } else {
   259→   259→        toast.error(json.error ?? "Generation failed");
   260→   260→      }
   261→   261→    } catch {
   262→   262→      toast.error("Network error generating captions");
   263→   263→    } finally {
   264→   264→      setGenerating(false);
   265→   265→    }
   266→   266→  }
   267→   267→
   268→   268→  async function publish() {
   269→   269→    if (selectedAccounts.length === 0) {
   270→   270→      toast.error("Select accounts");
   271→   271→      return;
   272→   272→    }
   273→   273→    setPublishing(true);
   274→   274→    try {
   275→   275→      const v: Record<string, { content: string; hashtags?: string }> = {};
   276→   276→      for (const a of selectedAccounts) {
   277→   277→        const ex = state.variants[a.id];
   278→   278→        v[a.id] = ex ?? { content: state.caption, hashtags: "" };
   279→   279→      }
   280→   280→      const scheduledAt =
   281→   281→        state.scheduleEnabled && state.scheduledAt
   282→   282→          ? new Date(state.scheduledAt).toISOString()
   283→   283→          : null;
   284→   284→      const res = await fetch("/api/posts", {
   285→   285→        method: "POST",
   286→   286→        headers: { "Content-Type": "application/json" },
   287→   287→        body: JSON.stringify({
   288→   288→          title: state.caption.slice(0, 50).trim() || "Untitled",
   289→   289→          caption: state.caption,
   290→   290→          mediaUrl: state.mediaUrl || null,
   291→   291→          tone: state.tone,
   292→   292→          scheduledAt,
   293→   293→          accountIds: selectedAccounts.map((a) => a.id),
   294→   294→          variants: v,
   295→   295→          authorId: currentUserId,
   296→   296→        }),
   297→   297→      });
   298→   298→      if (res.ok) {
   299→   299→        toast.success(
   300→   300→          state.scheduleEnabled && scheduledAt
   301→   301→            ? "Post scheduled successfully"
   302→   302→            : "Post published to all accounts"
   303→   303→        );
   304→   304→        setState(DEFAULT_STATE);
   305→   305→      } else {
   306→   306→        toast.error("Failed to publish");
   307→   307→      }
   308→   308→    } catch {
   309→   309→      toast.error("Network error");
   310→   310→    } finally {
   311→   311→      setPublishing(false);
   312→   312→    }
   313→   313→  }
   314→   314→
   315→   315→  function submitUrl() {
   316→   316→    if (!urlInput.trim()) {
   317→   317→      toast.error("Enter an image URL");
   318→   318→      return;
   319→   319→    }
   320→   320→    update({ mediaUrl: urlInput.trim() });
   321→   321→    setUrlInput("");
   322→   322→    setUploadOpen(false);
   323→   323→    toast.success("Media URL added");
   324→   324→  }
   325→   325→
   326→   326→  const canContinue =
   327→   327→    state.step === 1
   328→   328→      ? selectedAccounts.length > 0
   329→   329→      : state.step === 2
   330→   330→      ? state.caption.trim().length > 0
   331→   331→      : state.step === 3
   332→   332→      ? !!state.mediaUrl
   333→   333→      : state.step === 4
   334→   334→      ? true
   335→   335→      : false;
   336→   336→
   337→   337→  return (
   338→   338→    <div className="space-y-5 pb-28">
   339→   339→      <StepIndicator step={state.step} onStep={goToStep} />
   340→   340→
   341→   341→      <div className="mx-auto w-full max-w-3xl space-y-5">
   342→   342→        {state.step === 1 && (
   343→   343→          <Step1
   344→   344→            accounts={filteredAccounts}
   345→   345→            selectedIds={state.selectedAccountIds}
   346→   346→            onToggle={toggleAccount}
   347→   347→            onSelectAll={selectAll}
   348→   348→            onClear={clearAll}
   349→   349→            search={search}
   350→   350→            onSearch={setSearch}
   351→   351→            selectedAccounts={selectedAccounts}
   352→   352→          />
   353→   353→        )}
   354→   354→        {state.step === 2 && (
   355→   355→          <Step2
   356→   356→            caption={state.caption}
   357→   357→            onCaption={(caption) => update({ caption })}
   358→   358→            tone={state.tone}
   359→   359→            onTone={(tone) => update({ tone })}
   360→   360→            onGenerate={() => generate()}
   361→   361→            generating={generating}
   362→   362→            hasVariants={Object.keys(state.variants).length > 0}
   363→   363→            selectedCount={selectedAccounts.length}
   364→   364→          />
   365→   365→        )}
   366→   366→        {state.step === 3 && (
   367→   367→          <Step3
   368→   368→            mediaUrl={state.mediaUrl}
   369→   369→            mediaType={state.mediaType}
   370→   370→            altText={state.altText}
   371→   371→            onMedia={(mediaUrl) => update({ mediaUrl })}
   372→   372→            onMediaType={(mediaType) => update({ mediaType })}
   373→   373→            onAltText={(altText) => update({ altText })}
   374→   374→            uploadOpen={uploadOpen}
   375→   375→            onUploadOpen={setUploadOpen}
   376→   376→            urlInput={urlInput}
   377→   377→            onUrlInput={setUrlInput}
   378→   378→            onUrlSubmit={submitUrl}
   379→   379→          />
   380→   380→        )}
   381→   381→        {state.step === 4 && (
   382→   382→          <Step4
   383→   383→            accounts={selectedAccounts}
   384→   384→            variants={state.variants}
   385→   385→            onVariantChange={(id, v) =>
   386→   386→              setState((s) => ({ ...s, variants: { ...s.variants, [id]: v } }))
   387→   387→            }
   388→   388→            mediaUrl={state.mediaUrl}
   389→   389→            caption={state.caption}
   390→   390→            onRegenerate={(id) => generate(id)}
   391→   391→            onRegenerateAll={() => generate()}
   392→   392→            generating={generating}
   393→   393→          />
   394→   394→        )}
   395→   395→        {state.step === 5 && (
   396→   396→          <Step5
   397→   397→            accounts={selectedAccounts}
   398→   398→            mediaUrl={state.mediaUrl}
   399→   399→            mediaType={state.mediaType}
   400→   400→            tone={state.tone}
   401→   401→            caption={state.caption}
   402→   402→            scheduleEnabled={state.scheduleEnabled}
   403→   403→            onScheduleEnabled={(scheduleEnabled) => update({ scheduleEnabled })}
   404→   404→            scheduledAt={state.scheduledAt}
   405→   405→            onScheduledAt={(scheduledAt) => update({ scheduledAt })}
   406→   406→            hasVariants={Object.keys(state.variants).length > 0}
   407→   407→          />
   408→   408→        )}
   409→   409→      </div>
   410→   410→
   411→   411→      <StepFooter
   412→   412→        step={state.step}
   413→   413→        onBack={back}
   414→   414→        onNext={next}
   415→   415→        canContinue={canContinue}
   416→   416→        onPublish={publish}
   417→   417→        publishing={publishing}
   418→   418→        scheduleEnabled={state.scheduleEnabled}
   419→   419→      />
   420→   420→    </div>
   421→   421→  );
   422→   422→}

   424→   424→/* ============================================================
   425→   425→   Step indicator — sticky, responsive (collapses on mobile)
   426→   426→   ============================================================ */
   427→   427→function StepIndicator({
   428→   428→  step,
   429→   429→  onStep,
   430→   430→}: {
   431→   431→  step: Step;
   432→   432→  onStep: (s: Step) => void;
   433→   433→}) {
   434→   434→  const current = STEPS.find((s) => s.id === step) ?? STEPS[0];
   435→   435→  return (
   436→   436→    <div className="sticky top-0 z-30 -mx-4 px-4 py-2 sm:-mx-6 sm:px-6 lg:-mx-8 lg:px-8">
   437→   437→      <div className="glass-strong glass-sheen rounded-2xl px-4 py-3 sm:px-5">
   438→   438→        {/* Mobile — compact */}
   439→   439→        <div className="flex items-center justify-between gap-3 sm:hidden">
   440→   440→          <div className="flex items-center gap-2.5">
   441→   441→            <span className="grid h-8 w-8 place-items-center rounded-full bg-primary text-xs font-bold text-primary-foreground glow-pulse">
   442→   442→              {step}
   443→   443→            </span>
   444→   444→            <div className="min-w-0">
   445→   445→              <div className="text-[10px] uppercase tracking-wider text-muted-foreground">
   446→   446→                Step {step} of {STEPS.length}
   447→   447→              </div>
   448→   448→              <div className="truncate text-sm font-semibold">{current.label}</div>
   449→   449→            </div>
   450→   450→          </div>
   451→   451→          <div className="flex gap-1">
   452→   452→            {STEPS.map((s) => (
   453→   453→              <span
   454→   454→                key={s.id}
   455→   455→                className={cn(
   456→   456→                  "h-1.5 rounded-full transition-all",
   457→   457→                  s.id === step
   458→   458→                    ? "w-5 bg-primary"
   459→   459→                    : s.id < step
   460→   460→                    ? "w-1.5 bg-primary/60"
   461→   461→                    : "w-1.5 bg-muted-foreground/30"
   462→   462→                )}
   463→   463→              />
   464→   464→            ))}
   465→   465→          </div>
   466→   466→        </div>
   467→   467→
   468→   468→        {/* Desktop — full stepper */}
   469→   469→        <div className="hidden items-center sm:flex">
   470→   470→          {STEPS.map((s, i) => {
   471→   471→            const done = s.id < step;
   472→   472→            const active = s.id === step;
   473→   473→            const clickable = s.id <= step;
   474→   474→            return (
   475→   475→              <React.Fragment key={s.id}>
   476→   476→                <button
   477→   477→                  type="button"
   478→   478→                  disabled={!clickable}
   479→   479→                  onClick={() => clickable && onStep(s.id)}
   480→   480→                  className={cn(
   481→   481→                    "group flex shrink-0 items-center gap-2.5 transition",
   482→   482→                    clickable ? "cursor-pointer" : "cursor-default"
   483→   483→                  )}
   484→   484→                >
