   295→   295→function ProfileSection() {
   296→   296→  const { user } = useAuth();
   297→   297→  const { data, reload } = useApi<ApiUser[]>("/api/users");
   298→   298→  const [name, setName] = React.useState("");
   299→   299→  const [email, setEmail] = React.useState("");
   300→   300→  const [title, setTitle] = React.useState("");
   301→   301→  const [department, setDepartment] = React.useState<Department>("Leadership");
   302→   302→  const [bio, setBio] = React.useState("");
   303→   303→  const [timezone, setTimezone] = React.useState("America/Los_Angeles");
   304→   304→  const [language, setLanguage] = React.useState("en-US");
   305→   305→  const [saving, setSaving] = React.useState(false);
   306→   306→
   307→   307→  // Load real user data — prefer department from /api/users (the auth-store's
   308→   308→  // AuthUser shape doesn't carry department).
   309→   309→  React.useEffect(() => {
   310→   310→    if (!user) return;
   311→   311→    setName(user.name);
   312→   312→    setEmail(user.email);
   313→   313→    setTitle(user.title ?? "");
   314→   314→    const dbUser = (data ?? []).find((u) => u.id === user.id);
   315→   315→    const dept = dbUser?.department as Department | undefined;
   316→   316→    if (dept && (DEPARTMENTS as string[]).includes(dept)) {
   317→   317→      setDepartment(dept);
   318→   318→    } else {
   319→   319→      setDepartment("Leadership");
   320→   320→    }
   321→   321→  }, [user, data]);
   322→   322→
   323→   323→  const save = async () => {
   324→   324→    if (!user) return;
   325→   325→    setSaving(true);
   326→   326→    try {
   327→   327→      await fetch("/api/users", {
   328→   328→        method: "PATCH",
   329→   329→        headers: { "Content-Type": "application/json" },
   330→   330→        body: JSON.stringify({ id: user.id, name, email, title, department }),
   331→   331→      });
   332→   332→      toast.success("Profile saved", { description: "Your changes have been synced." });
   333→   333→      reload();
   334→   334→    } catch {
   335→   335→      toast.error("Failed to save profile");
   336→   336→    } finally {
   337→   337→      setSaving(false);
   338→   338→    }
   339→   339→  };
   340→   340→
   341→   341→  return (
   342→   342→    <GlassCard className="crystal-in p-5">
   343→   343→      <SectionTitle
   344→   344→        title="Profile"
   345→   345→        desc="Your personal information visible to your team."
   346→   346→        action={
   347→   347→          <button
   348→   348→            onClick={save}
   349→   349→            disabled={saving}
   350→   350→            className="liquid-pill flex items-center gap-1.5 rounded-xl px-3.5 py-1.5 text-xs font-semibold disabled:opacity-60"
   351→   351→          >
   352→   352→            <Save className="h-3.5 w-3.5" /> {saving ? "Saving…" : "Save changes"}
   353→   353→          </button>
   354→   354→        }
   355→   355→      />
   356→   356→
   357→   357→      {/* Avatar */}
   358→   358→      <div className="mb-5 flex items-center gap-4 rounded-2xl bg-muted/30 p-4">
   359→   359→        <NameAvatar name={name} color="rose" className="h-20 w-20 text-lg" />
   360→   360→        <div className="flex flex-col gap-2">
   361→   361→          <div className="text-sm font-semibold">{name}</div>
   362→   362→          <div className="text-xs text-muted-foreground">PNG / JPG up to 4MB · square recommended</div>
   363→   363→          <div className="flex gap-2">
   364→   364→            <button
   365→   365→              onClick={() => toast.info("Avatar upload — configure a storage provider in Integrations")}
   366→   366→              className="liquid-pill flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-xs font-medium"
   367→   367→            >
   368→   368→              <Camera className="h-3.5 w-3.5" /> Change
   369→   369→            </button>
   370→   370→            <button
   371→   371→              onClick={() => toast("Remove current avatar?")}
   372→   372→              className="rounded-lg border border-border bg-background px-3 py-1.5 text-xs font-medium text-muted-foreground transition hover:bg-muted/60"
   373→   373→            >
   374→   374→              Remove
   375→   375→            </button>
   376→   376→          </div>
   377→   377→        </div>
   378→   378→      </div>
   379→   379→
   380→   380→      {/* Fields */}
   381→   381→      <div className="grid gap-4 sm:grid-cols-2">
   382→   382→        <Field label="Full name">
   383→   383→          <Input value={name} onChange={(e) => setName(e.target.value)} />
   384→   384→        </Field>

   387→   387→        </Field>
