   109→function platformProfileUrl(platform: PlatformId, handle: string): string | null {
   110→  const h = handle.replace(/^@/, "");
   111→  switch (platform) {
   112→    case "instagram":
   113→      return `https://instagram.com/${h}`;
   114→    case "facebook":
   115→      return `https://facebook.com/${h}`;
   116→    case "x":
   117→      return `https://x.com/${h}`;
   118→    case "linkedin":
   119→      return `https://linkedin.com/company/${h}`;
   120→    case "tiktok":
   121→      return `https://tiktok.com/@${h}`;
   122→    case "youtube":
   123→      return `https://youtube.com/@${h}`;
   124→    case "threads":
   125→      return `https://threads.net/@${h}`;
   126→    default:
   127→      return null;
   128→  }
   129→}

   532→
   533→async function patchCompetitor(
   534→  id: string,
   535→  patch: Partial<Pick<Competitor, "name" | "handle" | "platform" | "notes" | "status">>,
   536→  reload: () => void,
   537→) {
   538→  try {
   539→    const res = await fetch(`/api/competitors/${id}`, {
   540→      method: "PATCH",
   541→      headers: { "Content-Type": "application/json" },
   542→      body: JSON.stringify(patch),
   543→    });
   544→    if (!res.ok) throw new Error("Failed");
   545→    toast.success("Competitor updated");
   546→    reload();
   547→  } catch {
   548→    toast.error("Could not update competitor");
   549→  }
   550→}
   551→
   552→async function deleteCompetitor(
   553→  id: string,
   554→  name: string,
   555→  reload: () => void,
   556→) {
   557→  try {
   558→    const res = await fetch(`/api/competitors/${id}`, { method: "DELETE" });
   559→    if (!res.ok) throw new Error("Failed");
   560→    toast.success("Competitor removed", { description: name });
   561→    reload();
   562→  } catch {
   563→    toast.error("Could not delete competitor");
   564→  }
   565→}

   567→function AddCompetitorForm({
   568→  onCancel,
   569→  onDone,
   570→}: {
   571→  onCancel: () => void;
   572→  onDone: () => void;
   573→}) {
   574→  const [name, setName] = React.useState("");
   575→  const [handle, setHandle] = React.useState("");
   576→  const [platform, setPlatform] = React.useState<PlatformId>("instagram");
   577→  const [notes, setNotes] = React.useState("");
   578→  const [busy, setBusy] = React.useState(false);
   579→
   580→  const submit = async () => {
   581→    if (!name.trim() || !handle.trim()) {
   582→      toast.error("Name and handle are required");
   583→      return;
   584→    }
   585→    setBusy(true);
   586→    try {
   587→      const res = await fetch("/api/competitors", {
   588→        method: "POST",
   589→        headers: { "Content-Type": "application/json" },
   590→        body: JSON.stringify({
   591→          name: name.trim(),
   592→          handle: handle.trim().replace(/^@/, ""),
   593→          platform,
   594→          notes: notes.trim() || null,
   595→        }),
   596→      });
   597→      if (!res.ok) throw new Error("Create failed");
   598→      toast.success("Competitor added");
   599→      setName("");
   600→      setHandle("");
   601→      setNotes("");
   602→      onDone();
   603→    } catch {
   604→      toast.error("Could not add competitor");
   605→    } finally {
   606→      setBusy(false);
   607→    }

   953→function EditCompetitorDialog({
   954→  competitor,
   955→  onClose,
   956→  onSaved,
   957→}: {
   958→  competitor: Competitor | null;
   959→  onClose: () => void;
   960→  onSaved: () => void;
   961→}) {
   962→  const [name, setName] = React.useState("");
   963→  const [handle, setHandle] = React.useState("");
   964→  const [platform, setPlatform] = React.useState<PlatformId>("instagram");
   965→  const [notes, setNotes] = React.useState("");
   966→  const [busy, setBusy] = React.useState(false);
   967→
   968→  React.useEffect(() => {
   969→    if (competitor) {
   970→      setName(competitor.name);
   971→      setHandle(competitor.handle);
   972→      setPlatform(competitor.platform);
   973→      setNotes(competitor.notes ?? "");
   974→    }
   975→  }, [competitor]);
   976→
   977→  const submit = async () => {
   978→    if (!competitor) return;
   979→    if (!name.trim() || !handle.trim()) {
   980→      toast.error("Name and handle are required");
   981→      return;
   982→    }
   983→    setBusy(true);
   984→    try {
   985→      const res = await fetch(`/api/competitors/${competitor.id}`, {
   986→        method: "PATCH",
   987→        headers: { "Content-Type": "application/json" },
   988→        body: JSON.stringify({
   989→          name: name.trim(),
   990→          handle: handle.trim().replace(/^@/, ""),
   991→          platform,
   992→          notes: notes.trim() || null,
   993→        }),
   994→      });
   995→      if (!res.ok) throw new Error("Update failed");
   996→      toast.success("Competitor updated");
   997→      onSaved();
   998→    } catch {
   999→      toast.error("Could not update competitor");
  1000→    } finally {
  1001→      setBusy(false);
  1002→    }
  1003→  };
  1004→
  1005→  return (
  1006→    <Dialog open={!!competitor} onOpenChange={(o) => !o && onClose()}>
  1007→      <DialogContent className="rounded-3xl sm:max-w-md">
  1008→        <DialogHeader>
  1009→          <DialogTitle className="flex items-center gap-2">
  1010→            <Pencil className="h-5 w-5 text-primary" />
  1011→            Edit competitor
  1012→          </DialogTitle>
  1013→          <DialogDescription>
  1014→            Update the rival&apos;s profile details. Tracking status and notes can be changed here.
  1015→          </DialogDescription>
  1016→        </DialogHeader>
  1017→        <div className="space-y-4 py-2">
  1018→          <div className="space-y-2">
  1019→            <Label htmlFor="ce-name">Name</Label>
  1020→            <Input
  1021→              id="ce-name"
  1022→              value={name}
  1023→              onChange={(e) => setName(e.target.value)}
  1024→            />
  1025→          </div>
  1026→          <div className="grid grid-cols-2 gap-3">
  1027→            <div className="space-y-2">
  1028→              <Label htmlFor="ce-handle">Handle</Label>
  1029→              <Input
  1030→                id="ce-handle"
  1031→                value={handle}
  1032→                onChange={(e) => setHandle(e.target.value)}
  1033→              />
  1034→            </div>
  1035→            <div className="space-y-2">
  1036→              <Label>Platform</Label>
  1037→              <Select
  1038→                value={platform}
  1039→                onValueChange={(v) => setPlatform(v as PlatformId)}
  1040→              >
  1041→                <SelectTrigger className="w-full">
  1042→                  <SelectValue />
  1043→                </SelectTrigger>
  1044→                <SelectContent>
  1045→                  {PLATFORM_LIST.map((p) => (
  1046→                    <SelectItem key={p.id} value={p.id}>
  1047→                      {p.name}
  1048→                    </SelectItem>
  1049→                  ))}
  1050→                </SelectContent>
  1051→              </Select>
  1052→            </div>
  1053→          </div>
  1054→          <div className="space-y-2">
  1055→            <Label htmlFor="ce-notes">Notes</Label>
  1056→            <Textarea
  1057→              id="ce-notes"
  1058→              value={notes}
  1059→              onChange={(e) => setNotes(e.target.value)}
  1060→              className="min-h-[80px] resize-none"
  1061→            />
  1062→          </div>
  1063→        </div>
  1064→        <DialogFooter>
  1065→          <Button variant="ghost" onClick={onClose} disabled={busy}>
  1066→            Cancel
  1067→          </Button>
  1068→          <button
  1069→            onClick={submit}
  1070→            disabled={busy}
  1071→            className="liquid-pill flex items-center gap-2 rounded-xl px-4 py-2 text-sm font-semibold disabled:opacity-60"
  1072→          >
  1073→            <Pencil className="h-4 w-4" />
  1074→            {busy ? "Saving…" : "Save changes"}
  1075→          </button>
