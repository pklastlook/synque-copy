   295→function ProfileSection() {
   296→  const { user } = useAuth();
   297→  const { data, reload } = useApi<ApiUser[]>("/api/users");
   298→  const [name, setName] = React.useState("");
   299→  const [email, setEmail] = React.useState("");
   300→  const [title, setTitle] = React.useState("");
   301→  const [department, setDepartment] = React.useState<Department>("Leadership");
   302→  const [bio, setBio] = React.useState("");
   303→  const [timezone, setTimezone] = React.useState("America/Los_Angeles");
   304→  const [language, setLanguage] = React.useState("en-US");
   305→  const [saving, setSaving] = React.useState(false);
   306→
   307→  // Load real user data — prefer department from /api/users (the auth-store's
   308→  // AuthUser shape doesn't carry department).
   309→  React.useEffect(() => {
   310→    if (!user) return;
   311→    setName(user.name);
   312→    setEmail(user.email);
   313→    setTitle(user.title ?? "");
   314→    const dbUser = (data ?? []).find((u) => u.id === user.id);
   315→    const dept = dbUser?.department as Department | undefined;
   316→    if (dept && (DEPARTMENTS as string[]).includes(dept)) {
   317→      setDepartment(dept);
   318→    } else {
   319→      setDepartment("Leadership");
   320→    }
   321→  }, [user, data]);
   322→
   323→  const save = async () => {
   324→    if (!user) return;
   325→    setSaving(true);
   326→    try {
   327→      await fetch("/api/users", {
   328→        method: "PATCH",
   329→        headers: { "Content-Type": "application/json" },
   330→        body: JSON.stringify({ id: user.id, name, email, title, department }),
   331→      });
   332→      toast.success("Profile saved", { description: "Your changes have been synced." });
   333→      reload();
   334→    } catch {
   335→      toast.error("Failed to save profile");

   337→      setSaving(false);

   463→function WorkspaceSection() {

   472→  const [autoVariants, setAutoVariants] = React.useState(true);
   473→  const [tone, setTone] = React.useState("Confident");
   474→  const [brandVoice, setBrandVoice] = React.useState("");
   475→  const [colors, setColors] = React.useState<string[]>(["#1B2A57", "#D4A24E", "#5BC0BE"]);
   476→  const [saving, setSaving] = React.useState(false);
   477→
   478→  // Load real workspace data from the database
   479→  React.useEffect(() => {
   480→    if (!ws) return;
   481→    setName(ws.name);
   482→    setTz(ws.timezone);
   483→    setCadence(ws.cadence);
   484→    setStart(ws.startTime);
   485→    setEnd(ws.endTime);
   486→    setAutoVariants(ws.autoVariants);
   487→    setTone(ws.tone);
   488→    setBrandVoice(ws.brandVoice ?? "");

   491→
   492→  const updateColor = (i: number, v: string) =>
   493→    setColors((prev) => prev.map((c, idx) => (idx === i ? v : c)));
   494→
   495→  const save = async () => {
   496→    if (!user?.id) {
   497→      toast.error("Sign in to save workspace settings");
   498→      return;
   499→    }
   500→    setSaving(true);
   501→    try {
   502→      const res = await fetch("/api/workspace", {
   503→        method: "PUT",
   504→        headers: { "Content-Type": "application/json" },
   505→        body: JSON.stringify({
   506→          ownerId: user.id,
   507→          name,
   508→          timezone: tz,
   509→          cadence,
   510→          startTime: start,
   511→          endTime: end,
   512→          autoVariants,
   513→          tone,
   514→          brandVoice,
   515→          colors,
   516→        }),
   517→      });
   518→      if (res.ok) {
   519→        toast.success("Workspace saved", { description: "Defaults updated for all team members." });
   520→        reload();
   521→      } else {
   522→        toast.error("Failed to save workspace");
   523→      }
   524→    } catch {
   525→      toast.error("Network error");
   526→    } finally {
   527→      setSaving(false);
   528→    }
   529→  };
   530→
   531→  return (
   532→    <GlassCard className="crystal-in p-5">
   533→      <SectionTitle
   534→        title="Workspace"
   535→        desc="Defaults applied across compose, schedule, and AI generation."
   536→        action={
   537→          <button onClick={save} disabled={saving} className="liquid-pill flex items-center gap-1.5 rounded-xl px-3.5 py-1.5 text-xs font-semibold disabled:opacity-50">
   538→            <Save className="h-3.5 w-3.5" /> {saving ? "Saving…" : "Save changes"}
   539→          </button>
   540→        }
   541→      />

   818→  name: string;

   829→function TeamRolesSection() {
   830→  const { data, loading } = useApi<ApiUser[]>("/api/users");
   831→  const [members, setMembers] = React.useState<TeamMember[]>([]);
   832→  const [inviteOpen, setInviteOpen] = React.useState(false);
   833→  const [inviteName, setInviteName] = React.useState("");
   834→  const [inviteEmail, setInviteEmail] = React.useState("");
   835→  const [inviteRole, setInviteRole] = React.useState<Role>("member");
   836→  const [invitePassword, setInvitePassword] = React.useState("");
   837→  const [inviteLoading, setInviteLoading] = React.useState(false);
   838→  const [saving, setSaving] = React.useState(false);
   839→
   840→  // Load real users from database
   841→  React.useEffect(() => {
   842→    if (!data) return;
   843→    const next: TeamMember[] = data.map((u) => ({
   844→      id: u.id,
   845→      name: u.name,
   846→      email: u.email,
   847→      color: (u.color as TeamMember["color"]) ?? "blue",
   848→      role: u.role,
   849→      title: u.title ?? "",
   850→      department: (u.department as Department) ?? "General",
   851→      status: u.status,
   852→      permissions: u.permissions ?? [],
   853→    }));

   856→

   858→    setMembers((prev) => prev.map((m) => (m.id === id ? { ...m, ...patch } : m)));

   860→  const removeMember = async (id: string, name: string) => {
   861→    try {
   862→      const res = await fetch("/api/users", {
   863→        method: "DELETE",
   864→        headers: { "Content-Type": "application/json" },
   865→        body: JSON.stringify({ id }),
   866→      });

   868→        setMembers((prev) => prev.filter((m) => m.id !== id));
   869→        toast.success(`Removed ${name}`);
   870→      } else {
   871→        toast.error("Failed to remove member");
   872→      }
   873→    } catch {
   874→      toast.error("Network error");
   875→    }
   876→  };
   877→
   878→  const save = async () => {
   879→    setSaving(true);
   880→    try {
   881→      await Promise.all(
   882→        members.map((m) =>
   883→          fetch("/api/users", {
   884→            method: "PATCH",
   885→            headers: { "Content-Type": "application/json" },
   886→            body: JSON.stringify({
   887→              id: m.id,
   888→              role: m.role,
   889→              title: m.title,
   890→              department: m.department,
   891→              status: m.status,
   892→            }),
   893→          })

   900→      setSaving(false);
   901→    }

   911→      return;
   912→    }

   915→      const res = await fetch("/api/auth/signup", {
   916→        method: "POST",
   917→        headers: { "Content-Type": "application/json" },
   918→        body: JSON.stringify({ name: inviteName.trim(), email: inviteEmail.trim(), password: invitePassword }),
   919→      });
   920→      const data = await res.json();
   921→      if (res.ok && data.token) {
   922→        // Set the new user's role via PATCH
   923→        if (inviteRole !== "member") {
   924→          await fetch("/api/users", {
   925→            method: "PATCH",
   926→            headers: { "Content-Type": "application/json" },
   927→            body: JSON.stringify({ id: data.user.id, role: inviteRole, title: inviteRole === "admin" ? "Administrator" : "Team Member" }),
   928→          }).catch(() => {});

   931→        setInviteName(""); setInviteEmail(""); setInviteRole("member"); setInvitePassword("");

   935→      } else {
   936→        toast.error(data.error || "Failed to create user");
