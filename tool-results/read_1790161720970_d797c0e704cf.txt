   117→function fmtRange(start: string | null, end: string | null) {
   118→  const s = start && isValid(parseISO(start)) ? parseISO(start) : null;
   119→  const e = end && isValid(parseISO(end)) ? parseISO(end) : null;
   120→  if (!s && !e) return null;
   121→  if (s && !e) return format(s, "MMM d, yyyy");
   122→  if (!s && e) return `→ ${format(e, "MMM d, yyyy")}`;
   123→  return `${format(s as Date, "MMM d")} → ${format(e as Date, "MMM d, yyyy")}`;
   124→}

   336→function CampaignCard({
   337→  campaign,
   338→  onViewPosts,
   339→  onEdit,
   340→  onDuplicate,
   341→  onDelete,
   342→}: {
   343→  campaign: Campaign;
   344→  onViewPosts: () => void;
   345→  onEdit: () => void;
   346→  onDuplicate: () => void;
   347→  onDelete: () => void;
   348→}) {
   349→  const status = STATUS_META[campaign.status];
   350→  const range = fmtRange(campaign.startDate, campaign.endDate);
   351→  const objective = campaign.objective
   352→    ? OBJECTIVES.find((o) => o.value === campaign.objective)
   353→    : null;
   354→
   355→  return (
   356→    <GlassCard hover className="relative flex overflow-hidden p-5">
   357→      {/* Colored left accent bar */}
   358→      <div
   359→        className="absolute inset-y-0 left-0 w-1.5"
   360→        style={{ background: campaign.color }}
   361→      />
   362→      <div
   363→        className="pointer-events-none absolute -right-8 -top-8 h-28 w-28 rounded-full opacity-15 blur-2xl"
   364→        style={{ background: campaign.color }}
   365→      />
   366→      <div className="relative ml-2 flex min-w-0 flex-1 flex-col">
   367→        {/* Top: status badge */}
   368→        <div className="flex items-center justify-between gap-2">
   369→          <span
   370→            className={cn(
   371→              "inline-flex items-center gap-1.5 rounded-full px-2 py-0.5 text-[11px] font-semibold",
   372→              status.bg,
   373→              status.text
   374→            )}
   375→          >
   376→            <span className={cn("h-1.5 w-1.5 rounded-full", status.dot)} />
   377→            {status.label}
   378→          </span>
   379→          <DropdownMenu>
   380→            <DropdownMenuTrigger asChild>
   381→              <button
   382→                className="grid h-7 w-7 place-items-center rounded-lg text-muted-foreground transition hover:bg-muted hover:text-foreground"
   383→                aria-label="Campaign actions"
   384→              >
   385→                <MoreVertical className="h-4 w-4" />
   386→              </button>
   387→            </DropdownMenuTrigger>
   388→            <DropdownMenuContent align="end" className="w-44 rounded-xl">
   389→              <DropdownMenuItem onClick={onEdit}>
   390→                <Pencil className="mr-2 h-4 w-4" /> Edit
   391→              </DropdownMenuItem>
   392→              <DropdownMenuItem onClick={onDuplicate}>
   393→                <Copy className="mr-2 h-4 w-4" /> Duplicate
   394→              </DropdownMenuItem>
   395→              <DropdownMenuSeparator />
   396→              <DropdownMenuItem
   397→                className="text-rose-500 focus:text-rose-500"
   398→                onClick={onDelete}
   399→              >
   400→                <Trash2 className="mr-2 h-4 w-4" /> Delete
   401→              </DropdownMenuItem>
   402→            </DropdownMenuContent>
   403→          </DropdownMenu>
   404→        </div>
   405→
   406→        {/* Name + description */}
   407→        <h3 className="mt-2 truncate text-base font-bold leading-tight">{campaign.name}</h3>
   408→        {campaign.description ? (
   409→          <p className="mt-1 line-clamp-2 text-xs text-muted-foreground">
   410→            {campaign.description}
   411→          </p>
   412→        ) : (
   413→          <p className="mt-1 text-xs italic text-muted-foreground/60">No description</p>
   414→        )}
   415→
   416→        {/* Objective row */}
   417→        <div className="mt-3 flex flex-wrap items-center gap-2">
   418→          {objective && (
   419→            <span className="inline-flex items-center gap-1 rounded-lg bg-muted/60 px-2 py-1 text-[11px] font-medium text-foreground/80">
   420→              <objective.icon className="h-3 w-3" />
   421→              {objective.label}
   422→            </span>
   423→          )}
   424→        </div>
   425→
   426→        {/* Date range */}
   427→        {range && (
   428→          <div className="mt-2 flex items-center gap-1.5 text-[11px] text-muted-foreground">
   429→            <CalendarRange className="h-3 w-3" />
   430→            <span className="truncate">{range}</span>
   431→          </div>
   432→        )}
   433→
   434→
   435→        {/* Footer */}
   436→        <div className="mt-4 flex items-center gap-2 border-t border-border/60 pt-3">
   437→          <Button
   438→            variant="ghost"
   439→            size="sm"
   440→            className="h-8 flex-1 rounded-xl text-xs"
   441→            onClick={onViewPosts}
   442→          >
   443→            View posts <ArrowRight className="ml-1 h-3.5 w-3.5" />
   444→          </Button>
   445→        </div>
   446→      </div>
   447→    </GlassCard>
   448→  );
   449→}

   451→function CreateCampaignDialog({
   452→  open,
   453→  onOpenChange,
   454→  onCreated,
   455→}: {
   456→  open: boolean;
   457→  onOpenChange: (v: boolean) => void;
   458→  onCreated: () => void;
   459→}) {
   460→  const { user } = useAuth();
   461→  const [name, setName] = React.useState("");
   462→  const [description, setDescription] = React.useState("");
   463→  const [color, setColor] = React.useState(COLOR_SWATCHES[0]);
   464→  const [objective, setObjective] = React.useState<string>("awareness");
   465→  const [startDate, setStartDate] = React.useState<string>("");
   466→  const [endDate, setEndDate] = React.useState<string>("");
   467→  const [submitting, setSubmitting] = React.useState(false);
   468→
   469→  const reset = () => {
   470→    setName("");
   471→    setDescription("");
   472→    setColor(COLOR_SWATCHES[0]);
   473→    setObjective("awareness");
   474→    setStartDate("");
   475→    setEndDate("");
   476→  };
   477→
   478→  const handleSubmit = async () => {
   479→    if (!name.trim()) {
   480→      toast.error("Campaign name is required");
   481→      return;
   482→    }
   483→    if (!user?.id) {
   484→      toast.error("You must be signed in to create a campaign");
   485→      return;
   486→    }
   487→    setSubmitting(true);
   488→    try {
   489→      const res = await fetch("/api/campaigns", {
   490→        method: "POST",
   491→        headers: { "Content-Type": "application/json" },
   492→        body: JSON.stringify({
   493→          name: name.trim(),
   494→          description: description.trim() || null,
   495→          color,
   496→          objective,
   497→          startDate: startDate || null,
   498→          endDate: endDate || null,
   499→          ownerId: user.id,
   500→        }),
   501→      });
   502→      if (!res.ok) throw new Error("Failed to create campaign");
   503→      toast.success(`Campaign "${name.trim()}" created`);
   504→      reset();
   505→      onCreated();
   506→    } catch {
   507→      toast.error("Could not create campaign");
   508→    } finally {
   509→      setSubmitting(false);
   510→    }
   511→  };
   512→
   513→  return (
   514→    <Dialog open={open} onOpenChange={onOpenChange}>
   515→      <DialogContent className="sm:max-w-lg rounded-3xl">
   516→        <DialogHeader>
   517→          <DialogTitle className="flex items-center gap-2">
   518→            <span
   519→              className="grid h-7 w-7 place-items-center rounded-lg bg-primary text-primary-foreground"
   520→            >
   521→              <Megaphone className="h-4 w-4" />
   522→            </span>
   523→            New campaign
   524→          </DialogTitle>
   525→          <DialogDescription>
   526→            Define the goals and timeline for your next campaign.
   527→          </DialogDescription>
   528→        </DialogHeader>
   529→
   530→        <div className="space-y-4">
   531→          {/* Name */}
   532→          <div className="space-y-2">
   533→            <Label htmlFor="cmp-name">Campaign name</Label>
   534→            <Input
   535→              id="cmp-name"
   536→              placeholder="Summer Launch 2025"
   537→              value={name}
   538→              onChange={(e) => setName(e.target.value)}
   539→            />
   540→          </div>
   541→
   542→          {/* Description */}
   543→          <div className="space-y-2">
   544→            <Label htmlFor="cmp-desc">Description</Label>
   545→            <Input
   546→              id="cmp-desc"
   547→              placeholder="A short summary of the campaign"
   548→              value={description}
   549→              onChange={(e) => setDescription(e.target.value)}
   550→            />
   551→          </div>
   552→
   553→          {/* Color picker */}
   554→          <div className="space-y-2">
   555→            <Label>Accent color</Label>
   556→            <div className="flex flex-wrap items-center gap-2">
   557→              {COLOR_SWATCHES.map((c) => (
   558→                <button
   559→                  key={c}
   560→                  type="button"
   561→                  onClick={() => setColor(c)}
   562→                  className={cn(
   563→                    "h-7 w-7 rounded-full border-2 transition",
   564→                    color === c
   565→                      ? "border-foreground scale-110"
   566→                      : "border-transparent hover:scale-110"
   567→                  )}
   568→                  style={{ background: c }}
   569→                  aria-label={`Select color ${c}`}
   570→                />
   571→              ))}
   572→            </div>
   573→          </div>
   574→
   575→          {/* Objective */}
   576→          <div className="space-y-2">
   577→            <Label>Objective</Label>
   578→            <Select value={objective} onValueChange={setObjective}>
   579→              <SelectTrigger className="w-full">
   580→                <SelectValue />
   581→              </SelectTrigger>
   582→              <SelectContent>

   636→function isoToDateInput(iso: string | null): string {
   637→  if (!iso) return "";
   638→  const d = parseISO(iso);
   639→  if (!isValid(d)) return "";
   640→  return format(d, "yyyy-MM-dd");
   641→}

   643→function EditCampaignDialog({
   644→  campaign,
   645→  onOpenChange,
   646→  onSaved,
   647→}: {
   648→  campaign: Campaign | null;
   649→  onOpenChange: (open: boolean) => void;
   650→  onSaved: () => void;
   651→}) {
   652→  const open = campaign !== null;
   653→  const [name, setName] = React.useState("");
   654→  const [description, setDescription] = React.useState("");
   655→  const [color, setColor] = React.useState(COLOR_SWATCHES[0]);
   656→  const [objective, setObjective] = React.useState<string>("awareness");
   657→  const [status, setStatus] = React.useState<Campaign["status"]>("draft");
   658→  const [startDate, setStartDate] = React.useState<string>("");
   659→  const [endDate, setEndDate] = React.useState<string>("");
   660→  const [submitting, setSubmitting] = React.useState(false);
   661→
   662→  // Prefill when a campaign is selected
   663→  React.useEffect(() => {
   664→    if (!campaign) return;
   665→    setName(campaign.name);
   666→    setDescription(campaign.description ?? "");
   667→    setColor(campaign.color || COLOR_SWATCHES[0]);
   668→    setObjective(campaign.objective ?? "awareness");
   669→    setStatus(campaign.status);
   670→    setStartDate(isoToDateInput(campaign.startDate));
   671→    setEndDate(isoToDateInput(campaign.endDate));
   672→  }, [campaign]);
   673→
   674→  async function handleSubmit() {
   675→    if (!campaign) return;
   676→    if (!name.trim()) {
   677→      toast.error("Campaign name is required");
   678→      return;
   679→    }
   680→    setSubmitting(true);
   681→    try {
   682→      const res = await fetch(`/api/campaigns/${campaign.id}`, {
   683→        method: "PATCH",
   684→        headers: { "Content-Type": "application/json" },
   685→        body: JSON.stringify({
   686→          name: name.trim(),
   687→          description: description.trim() || null,
   688→          color,
   689→          objective,
   690→          status,
   691→          startDate: startDate || null,
   692→          endDate: endDate || null,
   693→        }),
   694→      });
   695→      if (!res.ok) throw new Error("Failed to update campaign");
   696→      toast.success(`Campaign "${name.trim()}" updated`);
   697→      onSaved();
   698→    } catch {
   699→      toast.error("Could not update campaign");
   700→    } finally {
   701→      setSubmitting(false);
   702→    }
   703→  }
   704→
   705→  return (
   706→    <Dialog open={open} onOpenChange={onOpenChange}>
   707→      <DialogContent className="sm:max-w-lg rounded-3xl">
   708→        <DialogHeader>
   709→          <DialogTitle className="flex items-center gap-2">
   710→            <span
   711→              className="grid h-7 w-7 place-items-center rounded-lg bg-primary text-primary-foreground"
   712→            >
   713→              <Pencil className="h-4 w-4" />
   714→            </span>
   715→            Edit campaign
   716→          </DialogTitle>
   717→          <DialogDescription>
   718→            Update the details, status, or timeline of this campaign.
   719→          </DialogDescription>
   720→        </DialogHeader>
   721→
   722→        <div className="space-y-4">
   723→          <div className="space-y-2">
   724→            <Label htmlFor="edt-name">Campaign name</Label>
   725→            <Input
   726→              id="edt-name"
   727→              value={name}
   728→              onChange={(e) => setName(e.target.value)}
   729→            />
   730→          </div>
   731→
   732→          <div className="space-y-2">
   733→            <Label htmlFor="edt-desc">Description</Label>
   734→            <Input
   735→              id="edt-desc"
   736→              value={description}
   737→              onChange={(e) => setDescription(e.target.value)}
   738→            />
   739→          </div>
   740→
   741→          <div className="space-y-2">
   742→            <Label>Status</Label>
   743→            <Select
   744→              value={status}
   745→              onValueChange={(v) => setStatus(v as Campaign["status"])}
   746→            >
   747→              <SelectTrigger className="w-full">
   748→                <SelectValue />
   749→              </SelectTrigger>
   750→              <SelectContent>
   751→                <SelectItem value="draft">Draft</SelectItem>
   752→                <SelectItem value="active">Active</SelectItem>
   753→                <SelectItem value="paused">Paused</SelectItem>
   754→                <SelectItem value="completed">Completed</SelectItem>
   755→              </SelectContent>
   756→            </Select>
   757→          </div>
   758→
   759→          <div className="space-y-2">
   760→            <Label>Accent color</Label>
   761→            <div className="flex flex-wrap items-center gap-2">
   762→              {COLOR_SWATCHES.map((c) => (
   763→                <button
   764→                  key={c}
   765→                  type="button"
   766→                  onClick={() => setColor(c)}
   767→                  className={cn(
   768→                    "h-7 w-7 rounded-full border-2 transition",
   769→                    color === c
   770→                      ? "border-foreground scale-110"
   771→                      : "border-transparent hover:scale-110",
   772→                  )}
   773→                  style={{ background: c }}
   774→                  aria-label={`Select color ${c}`}
   775→                />
   776→              ))}
   777→            </div>
   778→          </div>
   779→
   780→          <div className="space-y-2">
   781→            <Label>Objective</Label>
   782→            <Select value={objective} onValueChange={setObjective}>
   783→              <SelectTrigger className="w-full">
   784→                <SelectValue />
   785→              </SelectTrigger>
   786→              <SelectContent>
   787→                {OBJECTIVES.map((o) => (
   788→                  <SelectItem key={o.value} value={o.value}>
   789→                    <span className="flex items-center gap-2">
   790→                      <o.icon className="h-3.5 w-3.5" />
   791→                      {o.label}
   792→                    </span>
   793→                  </SelectItem>
   794→                ))}
   795→              </SelectContent>
   796→            </Select>
   797→          </div>
   798→
   799→          <div className="grid grid-cols-2 gap-3">
   800→            <div className="space-y-2">
   801→              <Label htmlFor="edt-start">Start date</Label>
   802→              <Input
   803→                id="edt-start"
   804→                type="date"
   805→                value={startDate}
   806→                onChange={(e) => setStartDate(e.target.value)}
   807→              />
   808→            </div>

   810→              <Label htmlFor="edt-end">End date</Label>

   816→              />
   817→            </div>
   818→          </div>
   819→        </div>
   820→
   821→        <DialogFooter>
   822→          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={submitting}>
   823→            Cancel
   824→          </Button>
   825→          <button
   826→            onClick={handleSubmit}
   827→            disabled={submitting}
