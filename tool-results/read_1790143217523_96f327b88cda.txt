   110→function fmtSize(bytes: number) {
   111→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
   112→  if (bytes >= 1_048_576) return `${(bytes / 1_048_576).toFixed(1)} MB`;
   113→  if (bytes >= 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   114→  return `${bytes} B`;
   115→}
   116→
   117→function fmtBytesBig(bytes: number) {
   118→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
   119→  return `${(bytes / 1_048_576).toFixed(1)} MB`;
   120→}

   143→function parseTags(tags: string | null): string[] {
   144→  if (!tags) return [];

   600→function UploadDialog({
   601→  open,
   602→  onOpenChange,
   603→  folders,
   604→  onUploaded,
   605→}: {
   606→  open: boolean;
   607→  onOpenChange: (v: boolean) => void;
   608→  folders: string[];
   609→  onUploaded: () => void;
   610→}) {
   611→  const { user } = useAuth();
   612→  const [url, setUrl] = React.useState("");
   613→  const [name, setName] = React.useState("");
   614→  const [tags, setTags] = React.useState("");
   615→  const [folder, setFolder] = React.useState<string>("root");
   616→  const [type, setType] = React.useState<"image" | "video" | "audio">("image");
   617→  const [submitting, setSubmitting] = React.useState(false);
   618→
   619→  const reset = () => {
   620→    setUrl("");
   621→    setName("");
   622→    setTags("");
   623→    setFolder("root");
   624→    setType("image");
   625→  };
   626→
   627→  const handleSubmit = async (override?: Partial<typeof PRESET_LIBRARY[number]> & { url?: string; name?: string }) => {
   628→    const finalUrl = override?.url ?? url;
   629→    const finalName = override?.name ?? name;
   630→    if (!finalUrl || !finalName) {
   631→      toast.error("URL and name are required");
   632→      return;
   633→    }
   634→    setSubmitting(true);
   635→    try {
   636→      const res = await fetch("/api/media", {
   637→        method: "POST",
   638→        headers: { "Content-Type": "application/json" },
   639→        body: JSON.stringify({
   640→          name: finalName,
   641→          url: finalUrl,
   642→          type,
   643→          tags: (override?.tags ?? tags) || null,
   644→          folder,
   645→          uploadedById: user?.id ?? "u-admin",
   646→        }),
   647→      });
   648→      if (!res.ok) throw new Error("Upload failed");
   649→      toast.success(`"${finalName}" added`);
   650→      reset();
   651→      onUploaded();
   652→    } catch {
   653→      toast.error("Could not upload asset");
   654→    } finally {
   655→      setSubmitting(false);
   656→    }
   657→  };
   658→
   659→  return (
   660→    <Dialog open={open} onOpenChange={onOpenChange}>
   661→      <DialogContent className="sm:max-w-lg rounded-3xl">
   662→        <DialogHeader>
   663→          <DialogTitle className="flex items-center gap-2">
   664→            <span className="grid h-7 w-7 place-items-center rounded-lg bg-primary text-primary-foreground">
   665→              <Upload className="h-4 w-4" />
   666→            </span>
   667→            Add media
   668→          </DialogTitle>
   669→          <DialogDescription>
   670→            Paste a URL or pick from the preset library below.
   671→          </DialogDescription>
   672→        </DialogHeader>
   673→
   674→        <div className="space-y-4">
   675→          <div className="space-y-2">
   676→            <Label htmlFor="m-url">Asset URL</Label>
   677→            <Input
   678→              id="m-url"
   679→              placeholder="https://example.com/image.jpg"
   680→              value={url}
   681→              onChange={(e) => setUrl(e.target.value)}
   682→            />
   683→          </div>
   684→          <div className="grid grid-cols-2 gap-3">
   685→            <div className="space-y-2">
   686→              <Label htmlFor="m-name">Name</Label>
   687→              <Input
   688→                id="m-name"
   689→                placeholder="Photo name"
   690→                value={name}
   691→                onChange={(e) => setName(e.target.value)}
   692→              />
   693→            </div>
   694→            <div className="space-y-2">
   695→              <Label htmlFor="m-tags">Tags</Label>
   696→              <Input
   697→                id="m-tags"
   698→                placeholder="brand, product"
   699→                value={tags}
   700→                onChange={(e) => setTags(e.target.value)}
   701→              />
   702→            </div>
   703→          </div>
   704→          <div className="grid grid-cols-2 gap-3">
   705→            <div className="space-y-2">
   706→              <Label>Type</Label>
   707→              <Select
   708→                value={type}
   709→                onValueChange={(v) => setType(v as "image" | "video" | "audio")}
   710→              >
   711→                <SelectTrigger className="w-full">
   712→                  <SelectValue />
   713→                </SelectTrigger>
   714→                <SelectContent>
   715→                  <SelectItem value="image">Image</SelectItem>
   716→                  <SelectItem value="video">Video</SelectItem>
   717→                  <SelectItem value="audio">Audio</SelectItem>
   718→                </SelectContent>
   719→              </Select>
   720→            </div>
   721→            <div className="space-y-2">
   722→              <Label>Folder</Label>
   723→              <Select value={folder} onValueChange={setFolder}>
   724→                <SelectTrigger className="w-full">
   725→                  <SelectValue />
   726→                </SelectTrigger>
   727→                <SelectContent>
   728→                  <SelectItem value="root">root</SelectItem>
   729→                  {folders
   730→                    .filter((f) => f !== "root")
   731→                    .map((f) => (
   732→                      <SelectItem key={f} value={f}>
   733→                        {f}
   734→                      </SelectItem>
   735→                    ))}
   736→                </SelectContent>
   737→              </Select>
   738→            </div>
   739→          </div>
   740→
   741→          {/* Preset library */}
   742→          <div className="space-y-2">
   743→            <Label>Or pick from library</Label>
   744→            <div className="grid grid-cols-3 gap-2">
   745→              {PRESET_LIBRARY.map((p) => (
   746→                <button
   747→                  key={p.url}
   748→                  type="button"
   749→                  onClick={() => handleSubmit(p)}
   750→                  disabled={submitting}
   751→                  className="group relative aspect-square overflow-hidden rounded-xl border border-border/40 transition hover:-translate-y-0.5 hover:border-primary/40"
   752→                  title={p.name}
   753→                >
   754→                  <img
   755→                    src={p.url}
   756→                    alt={p.name}
   757→                    className="h-full w-full object-cover transition group-hover:scale-105"
   758→                  />
   759→                  <span className="absolute inset-0 bg-black/0 transition group-hover:bg-black/30" />
   760→                  <span className="absolute bottom-1 left-1 right-1 truncate rounded-md bg-white/90 px-1.5 py-0.5 text-[9px] font-medium opacity-0 transition group-hover:opacity-100">
   761→                    {p.name}
   762→                  </span>
   763→                </button>
   764→              ))}
   765→            </div>
   766→          </div>
   767→        </div>
   768→
   769→        <DialogFooter>
   770→          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={submitting}>
   771→            Cancel
   772→          </Button>
   773→          <button
   774→            onClick={() => handleSubmit()}
   775→            disabled={submitting}
   776→            className="liquid-pill flex items-center gap-2 rounded-xl px-4 py-2 text-sm font-semibold disabled:opacity-60"
   777→          >
   778→            <Plus className="h-4 w-4" />
   779→            {submitting ? "Adding…" : "Add asset"}
   780→          </button>
   781→        </DialogFooter>
   782→      </DialogContent>
   783→    </Dialog>
   784→  );
   785→}
