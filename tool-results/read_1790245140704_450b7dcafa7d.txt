    44→  qr: string | null;

    60→  new: { label: "New", color: "text-emerald-500", bg: "bg-emerald-500/15", icon: InboxIcon },

    65→// ============ Main view ============
    66→
    67→export function WhatsAppView() {
    68→  const { data, loading, reload } = useApi<WhatsAppMessage[]>("/api/whatsapp");
    69→  const { openComposer } = useNav();
    70→  const { user, token } = useAuth();
    71→  const ownerId = user?.id ?? "";
    72→
    73→  // Helper: build auth headers (Bearer token from auth store)
    74→  const authHeaders = React.useMemo(() => {
    75→    const headers: Record<string, string> = { "Content-Type": "application/json" };
    76→    if (token) headers["Authorization"] = `Bearer ${token}`;
    77→    return headers;
    78→  }, [token]);
    79→
    80→  const [showPasteForm, setShowPasteForm] = React.useState(false);
    81→  const [statusFilter, setStatusFilter] = React.useState<StatusFilter>("all");
    82→  const [search, setSearch] = React.useState("");
    83→  const [copiedId, setCopiedId] = React.useState<string | null>(null);
    84→
    85→  // === WhatsApp live connection state ===
    86→  const [connection, setConnection] = React.useState<ConnectionStatus | null>(null);
    87→  const [showQR, setShowQR] = React.useState(false);
    88→  const [connecting, setConnecting] = React.useState(false);
    89→
    90→  // Poll connection status every 2s when modal is open OR when not connected
    91→  React.useEffect(() => {
    92→    if (!ownerId) return;
    93→    let cancelled = false;
    94→
    95→    async function poll() {
    96→      try {
    97→        const res = await fetch(`/api/whatsapp/connection`, {
    98→          headers: token ? { Authorization: `Bearer ${token}` } : {},
    99→        });
   100→        if (!res.ok) {
   101→          if (!cancelled) setConnection({ connected: false, qr: null, phone: null, serviceOffline: true });
   102→          return;
   103→        }
   104→        const json: ConnectionStatus = await res.json();
   105→        if (!cancelled) {
   106→          setConnection(json);
   107→          // Auto-close QR modal when connected
   108→          if (json.connected && showQR) {
   109→            setShowQR(false);
   110→            toast.success("WhatsApp connected!", {
   111→              description: `Now receiving live messages from ${json.phone ?? "your phone"}.`,
   112→            });
   113→            reload();
   114→          }
   115→        }
   116→      } catch {
   117→        if (!cancelled) setConnection({ connected: false, qr: null, phone: null, serviceOffline: true });
   118→      }
   119→    }
   120→
   121→    poll();
   122→    // Poll every 2s while modal open OR not connected, else every 10s
   123→    const interval = setInterval(
   124→      poll,
   125→      showQR || !connection?.connected ? 2000 : 10000
   126→    );
   127→    return () => {
   128→      cancelled = true;
   129→      clearInterval(interval);
   130→    };
   131→  }, [ownerId, showQR, connection?.connected, token, reload]);
   132→
   133→  async function handleStartConnection() {
   134→    if (!ownerId) {
   135→      toast.error("Sign in to connect WhatsApp");
   136→      return;
   137→    }
   138→    setConnecting(true);
   139→    try {
   140→      const res = await fetch("/api/whatsapp/connection", {
   141→        method: "POST",
   142→        headers: authHeaders,
   143→        body: JSON.stringify({ action: "start" }),
   144→      });
   145→      const json = await res.json();
   146→      if (!res.ok) {
   147→        toast.error("WhatsApp service unavailable", {
   148→          description: json.error ?? "Ask the admin to start the WhatsApp mini-service.",
   149→          duration: 8000,
   150→        });
   151→      } else {
   152→        setShowQR(true);
   153→        toast.info("Generating QR code...", {
   154→          description: "Open WhatsApp on your phone → Settings → Linked Devices → Link a Device → scan the QR.",
   155→        });
   156→      }
   157→    } catch {
   158→      toast.error("Network error");
   159→    } finally {
   160→      setConnecting(false);
   161→    }
   162→  }
   163→
   164→  async function handleLogout() {
   165→    if (!ownerId) return;
   166→    try {
   167→      await fetch("/api/whatsapp/connection", {
   168→        method: "POST",
   169→        headers: authHeaders,
   170→        body: JSON.stringify({ action: "logout" }),
   171→      });
   172→      toast.success("WhatsApp disconnected");
   173→      setConnection({ connected: false, qr: null, phone: null });
   174→      reload();
   175→    } catch {
   176→      toast.error("Failed to disconnect");
   177→    }
   178→  }
   179→
   180→  const allMessages = data ?? [];
   181→
   182→  const filtered = React.useMemo(() => {
   183→    let list = allMessages;
   184→    if (statusFilter !== "all") {
   185→      list = list.filter((m) => m.status === statusFilter);
   186→    }
   187→    if (search.trim()) {
   188→      const q = search.toLowerCase();
   189→      list = list.filter(
   190→        (m) =>
   191→          m.content.toLowerCase().includes(q) ||
   192→          m.senderName.toLowerCase().includes(q) ||
   193→          (m.groupName ?? "").toLowerCase().includes(q)
   194→      );
   195→    }
   196→    return list;
   197→  }, [allMessages, statusFilter, search]);
   198→
   199→  const stats = React.useMemo(() => {
   200→    return {
   201→      total: allMessages.length,
   202→      new: allMessages.filter((m) => m.status === "new").length,
   203→      used: allMessages.filter((m) => m.status === "used").length,
   204→      archived: allMessages.filter((m) => m.status === "archived").length,
   205→    };
   206→  }, [allMessages]);
   207→
   208→  async function handleCopy(msg: WhatsAppMessage) {
   209→    try {
   210→      await navigator.clipboard.writeText(msg.content);
   211→      setCopiedId(msg.id);
   212→      setTimeout(() => setCopiedId(null), 2000);
   213→      toast.success("Copied to clipboard");
   214→    } catch {
   215→      toast.error("Failed to copy");
   216→    }
   217→  }
   218→
   219→  function handleSendToComposer(msg: WhatsAppMessage) {
   220→    // Pre-fill the composer with the WhatsApp message content
   221→    openComposer(msg.content);
   222→    // Mark as used
   223→    fetch(`/api/whatsapp/${msg.id}`, {
   224→      method: "PATCH",
   225→      headers: authHeaders,
   226→      body: JSON.stringify({ status: "used" }),
   227→    }).then(() => reload());
   228→    toast.success("Sent to Composer", {
   229→      description: "The message is now in your composer — generate AI variants & publish.",
   230→    });
   231→  }
   232→
   233→  async function handleArchive(msg: WhatsAppMessage) {
   234→    await fetch(`/api/whatsapp/${msg.id}`, {
   235→      method: "PATCH",
   236→      headers: authHeaders,
   237→      body: JSON.stringify({ status: "archived" }),
   238→    });
   239→    reload();
   240→    toast.success("Archived");
   241→  }
   242→
   243→  async function handleDelete(msg: WhatsAppMessage) {
   244→    await fetch(`/api/whatsapp/${msg.id}`, {
   245→      method: "DELETE",
   246→      headers: authHeaders,
   247→    });
   248→    reload();
   249→    toast.success("Deleted");
   250→  }
   251→
   252→  if (loading) return <WhatsAppSkeleton />;
   253→
   254→  return (
   255→    <div className="space-y-5">
   256→      <HelpText module="whatsapp">
   257→        <strong className="text-foreground">WhatsApp Inbox</strong> — connect your WhatsApp (scan QR, just like WhatsApp Web)
   258→        to auto-receive messages from groups & chats. Or paste manually. Then send any message to the AI Composer.
   259→      </HelpText>
   260→
   261→      {/* === CONNECTION BANNER (NEW) — live WhatsApp connection === */}
   262→      <ConnectionBanner
   263→        connection={connection}
   264→        connecting={connecting}
   265→        onConnect={handleStartConnection}
   266→        onDisconnect={handleLogout}
   267→        onShowQR={() => setShowQR(true)}
   268→      />
   269→
   270→      {/* === QR MODAL === */}
   271→      <Dialog open={showQR} onOpenChange={setShowQR}>
   272→        <DialogContent className="max-w-md">
   273→          <DialogHeader>
   274→            <DialogTitle className="flex items-center gap-2">
   275→              <QrCode className="h-5 w-5 text-primary" />
   276→              Scan WhatsApp QR
   277→            </DialogTitle>
   278→            <DialogClose asChild>
   279→              <button className="absolute right-4 top-4 rounded-lg p-1 text-muted-foreground transition hover:bg-muted">
   280→                <X className="h-4 w-4" />
   281→              </button>
   282→            </DialogClose>
   283→          </DialogHeader>
   284→          <QRCodeDisplay connection={connection} />
   285→        </DialogContent>
   286→      </Dialog>
   287→
   288→      {/* === KPI strip === */}
   289→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   290→        <KpiCard
   291→          label="Total Messages"
   292→          value={stats.total}
   293→          icon={<MessageCircle className="h-5 w-5" />}
   294→          accent="oklch(0.78 0.18 145)"
   295→          hint="pasted from WhatsApp"
   296→          delay={50}
   297→        />
   298→        <KpiCard
   299→          label="New"
   300→          value={stats.new}
   301→          icon={<InboxIcon className="h-5 w-5" />}
   302→          accent="oklch(0.74 0.18 165)"
   303→          hint="waiting to be published"
   304→          delay={120}
   305→        />
   306→        <KpiCard
   307→          label="Used"
   308→          value={stats.used}
   309→          icon={<CheckCheck className="h-5 w-5" />}
   310→          accent="oklch(0.62 0.22 255)"
   311→          hint="sent to composer"
   312→          delay={190}
   313→        />
   314→        <KpiCard
   315→          label="Archived"
   316→          value={stats.archived}
   317→          icon={<Archive className="h-5 w-5" />}
   318→          accent="oklch(0.66 0.22 305)"
   319→          hint="no longer needed"
   320→          delay={260}
   321→        />
   322→      </div>
   323→
   324→      {/* === Paste form === */}
   325→      <GlassCard className="crystal-in p-5" style={{ animationDelay: "0.30s" }}>
   326→        <SectionTitle
   327→          title="Paste from WhatsApp"
   328→          desc="Copy a message from WhatsApp, then paste it here to publish on social media"
   329→          action={
   330→            <button
   331→              onClick={() => setShowPasteForm(!showPasteForm)}
   332→              className="liquid-pill flex items-center gap-2 rounded-xl px-4 py-2 text-xs font-semibold"
   333→            >
   334→              {showPasteForm ? <Archive className="h-3.5 w-3.5" /> : <Plus className="h-3.5 w-3.5" />}
   335→              {showPasteForm ? "Close" : "Paste message"}
   336→            </button>
   337→          }
   338→        />
   339→        {showPasteForm && (
   340→          <PasteForm
   341→            userId={user?.id ?? null}
   342→            authHeaders={authHeaders}
   343→            onSaved={() => {
   344→              setShowPasteForm(false);
   345→              reload();
   346→            }}
   347→          />
   348→        )}
   349→      </GlassCard>
   350→
   351→      {/* === Messages list === */}
   352→      <GlassCard className="crystal-in p-5" style={{ animationDelay: "0.38s" }}>
   353→        <SectionTitle
   354→          title="Saved messages"
   355→          desc={`${filtered.length} of ${stats.total} messages`}
   356→          action={
   357→            <div className="flex items-center gap-1.5">
   358→              {/* Search */}
   359→              <div className="relative">
   360→                <Search className="pointer-events-none absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   361→                <input
   362→                  type="text"
   363→                  placeholder="Search..."
   364→                  value={search}
   365→                  onChange={(e) => setSearch(e.target.value)}
   366→                  className="w-32 rounded-lg border border-border/50 bg-background/60 py-1.5 pl-7 pr-2 text-xs outline-none transition focus:border-primary/50 sm:w-48"
   367→                />
   368→              </div>
   369→            </div>
   370→          }
   371→        />
   372→
   373→        {/* Status filter pills */}
   374→        <div className="mb-3 flex flex-wrap items-center gap-1.5">
   375→          {(["all", "new", "used", "archived"] as StatusFilter[]).map((f) => (
   376→            <button
   377→              key={f}
   378→              onClick={() => setStatusFilter(f)}
   379→              className={cn(
   380→                "rounded-full px-3 py-1.5 text-xs font-medium transition",
   381→                statusFilter === f
   382→                  ? "bg-primary text-primary-foreground shadow-sm"
   383→                  : "bg-muted/60 text-muted-foreground hover:bg-muted"
   384→              )}
   385→            >
   386→              {f === "all" ? "All" : STATUS_META[f]?.label ?? f}
   387→              {f !== "all" && stats[f] > 0 && (
   388→                <span className="ml-1.5 text-[10px] opacity-70">({stats[f]})</span>
   389→              )}
   390→            </button>
   391→          ))}
   392→        </div>
   393→
   394→        {/* Messages */}
   395→        <div className="scroll-liquid max-h-[600px] space-y-2 overflow-y-auto pr-1">
   396→          {filtered.length === 0 ? (
   397→            <div className="flex flex-col items-center justify-center gap-3 py-12 text-center">
   398→              <div className="grid h-12 w-12 place-items-center rounded-2xl bg-muted text-muted-foreground">
   399→                <MessageCircle className="h-6 w-6" />
   400→              </div>
   401→              <div>
   402→                <p className="text-sm font-semibold">
   403→                  {stats.total === 0 ? "No messages yet" : "No messages match your filter"}
   404→                </p>
   405→                <p className="mt-0.5 text-[11px] text-muted-foreground">
   406→                  {stats.total === 0
   407→                    ? "Paste a message from WhatsApp to get started"
   408→                    : "Try a different filter or search term"}
   409→                </p>
   410→              </div>
   411→              {stats.total === 0 && (
   412→                <button
   413→                  onClick={() => setShowPasteForm(true)}
   414→                  className="rounded-lg bg-primary/10 px-3 py-1.5 text-[11px] font-semibold text-primary transition hover:bg-primary/20"
   415→                >
   416→                  Paste message
   417→                </button>
   418→              )}
   419→            </div>
   420→          ) : (
   421→            filtered.map((msg, i) => (
   422→              <MessageCard
   423→                key={msg.id}
   424→                msg={msg}
   425→                copied={copiedId === msg.id}
   426→                onCopy={() => handleCopy(msg)}
   427→                onSendToComposer={() => handleSendToComposer(msg)}
   428→                onArchive={() => handleArchive(msg)}
   429→                onDelete={() => handleDelete(msg)}
   430→                delay={i * 0.03}
   431→              />
   432→            ))
   433→          )}
   434→        </div>
   435→      </GlassCard>
   436→    </div>
   437→  );
   438→}

   440→// ============ Paste Form ============
   441→function PasteForm({
   442→  userId,
   443→  authHeaders,
   444→  onSaved,
   445→}: {
   446→  userId: string | null;
   447→  authHeaders: Record<string, string>;
   448→  onSaved: () => void;
   449→}) {
   450→  const [source, setSource] = React.useState("group");
   451→  const [senderName, setSenderName] = React.useState("");
   452→  const [groupName, setGroupName] = React.useState("");
   453→  const [content, setContent] = React.useState("");
   454→  const [mediaUrl, setMediaUrl] = React.useState("");
   455→  const [saving, setSaving] = React.useState(false);
   456→
   457→  async function handleSubmit() {
   458→    if (!content.trim()) {
   459→      toast.error("Paste some content first");
   460→      return;
   461→    }
   462→    if (!senderName.trim()) {
   463→      toast.error("Enter the sender's name");
   464→      return;
   465→    }
   466→    setSaving(true);
   467→    try {
   468→      const res = await fetch("/api/whatsapp", {
   469→        method: "POST",
   470→        headers: authHeaders,
   471→        body: JSON.stringify({
   472→          source,
   473→          senderName: senderName.trim(),
   474→          groupName: groupName.trim() || null,
   475→          content: content.trim(),
   476→          mediaUrl: mediaUrl.trim() || null,
   477→          savedById: userId,
   478→        }),
   479→      });
   480→      if (res.ok) {
   481→        toast.success("Message saved");
   482→        setSenderName("");
   483→        setGroupName("");
   484→        setContent("");
   485→        setMediaUrl("");
   486→        onSaved();
   487→      } else {
   488→        const json = await res.json().catch(() => ({}));
   489→        toast.error(json.error ?? "Failed to save");
   490→      }
   491→    } catch {
   492→      toast.error("Network error");
   493→    } finally {
   494→      setSaving(false);
   495→    }
   496→  }
   497→
   498→  async function handlePasteFromClipboard() {
   499→    try {
   500→      const text = await navigator.clipboard.readText();
   501→      if (text) {
   502→        setContent(text);
   503→        toast.success("Pasted from clipboard");
   504→      } else {
   505→        toast.info("Clipboard is empty");
   506→      }
   507→    } catch {
   508→      toast.error("Couldn't access clipboard — paste manually with Ctrl+V");
   509→    }
   510→  }
   511→
   512→  return (
   513→    <div className="mt-4 space-y-3 rounded-2xl border border-border/40 bg-muted/20 p-4">
   514→      {/* Source + sender */}
   515→      <div className="grid gap-3 sm:grid-cols-2">
   516→        <div>
   517→          <label className="mb-1 block text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   518→            Source type
   519→          </label>
   520→          <div className="flex flex-wrap gap-1.5">
   521→            {Object.entries(SOURCE_META).map(([key, meta]) => (
   522→              <button
   523→                key={key}
   524→                onClick={() => setSource(key)}
   525→                className={cn(
   526→                  "flex items-center gap-1 rounded-lg px-2.5 py-1.5 text-[11px] font-semibold transition",
   527→                  source === key ? cn(meta.bg, meta.color) : "bg-muted/60 text-muted-foreground hover:bg-muted"
   528→                )}
   529→              >
   530→                <meta.icon className="h-3 w-3" />
   531→                {meta.label}
   532→              </button>
   533→            ))}
   534→          </div>
   535→        </div>
   536→        <div>
   537→          <label className="mb-1 block text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   538→            Sender name *
   539→          </label>
   540→          <input
   541→            type="text"
   542→            value={senderName}
   543→            onChange={(e) => setSenderName(e.target.value)}
   544→            placeholder="e.g. Sarah Khan"
   545→            className="w-full rounded-lg border border-border/50 bg-background/60 px-3 py-1.5 text-sm outline-none transition focus:border-primary/50"
   546→          />
   547→        </div>
   548→      </div>
   549→
   550→      {/* Group name */}
   551→      <div>
   552→        <label className="mb-1 block text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   553→          {source === "group" ? "Group name" : source === "channel" ? "Channel name" : "Chat name (optional)"}
   554→        </label>
   555→        <input
   556→          type="text"
   557→          value={groupName}
   558→          onChange={(e) => setGroupName(e.target.value)}
   559→          placeholder={source === "group" ? "e.g. Marketing Team" : source === "channel" ? "e.g. Company Updates" : "e.g. John Doe"}
   560→          className="w-full rounded-lg border border-border/50 bg-background/60 px-3 py-1.5 text-sm outline-none transition focus:border-primary/50"
   561→        />
   562→      </div>
   563→
   564→      {/* Content */}
   565→      <div>
   566→        <div className="mb-1 flex items-center justify-between">
   567→          <label className="text-[10px] font-bold uppercase tracking-wide text-muted-foreground">
   568→            Message content *
   569→          </label>
   570→          <button
   571→            onClick={handlePasteFromClipboard}

   574→            <Copy className="h-3 w-3" /> Paste from clipboard

   600→        </label>

   602→          type="url"

   604→          onChange={(e) => setMediaUrl(e.target.value)}

   608→      </div>
   609→
   610→      {/* Actions */}
   611→      <div className="flex items-center justify-end gap-2">
   612→        <button
   613→          onClick={() => {
   614→            setSenderName("");
   615→            setGroupName("");
   616→            setContent("");
   617→            setMediaUrl("");
   618→          }}
   619→          className="rounded-lg border border-border/50 px-3 py-1.5 text-xs font-medium text-muted-foreground transition hover:bg-muted"
   620→        >
   621→          Clear
   622→        </button>
   623→        <button
   624→          onClick={handleSubmit}
   625→          disabled={saving || !content.trim() || !senderName.trim()}
   626→          className="liquid-pill flex items-center gap-2 rounded-lg px-4 py-1.5 text-xs font-semibold disabled:opacity-50"
   627→        >
   628→          {saving ? <Clock className="h-3.5 w-3.5 animate-spin" /> : <Check className="h-3.5 w-3.5" />}
   629→          {saving ? "Saving..." : "Save message"}
   630→        </button>
   631→      </div>
   632→    </div>
   633→  );
