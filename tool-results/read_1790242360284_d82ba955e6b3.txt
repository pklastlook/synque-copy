    76→    76→  function getToken(): string | null {
    77→    77→    try { return JSON.parse(localStorage.getItem("socialflow-auth") ?? "{}")?.state?.token ?? null; }

    81→    81→  async function generateReport(templateId: string) {
    82→    82→    const template = TEMPLATES.find(t => t.id === templateId);
    83→    83→    if (!template) return;
    84→    84→    setGenerating(true);
    85→    85→    setReportType(templateId);
    86→    86→    const title = `${template.name} — ${new Date().toLocaleDateString(undefined, { month: "short", day: "numeric", year: "numeric" })}`;
    87→    87→    setReportTitle(title);
    88→    88→
    89→    89→    try {
    90→    90→      const token = getToken();
    91→    91→      const res = await fetch("/api/reports", {
    92→    92→        method: "POST",
    93→    93→        headers: {
    94→    94→          "Content-Type": "application/json",
    95→    95→          ...(token ? { "Authorization": `Bearer ${token}` } : {}),
    96→    96→        },
    97→    97→        body: JSON.stringify({
    98→    98→          name: title,
    99→    99→          type: templateId,
   100→   100→          description: `${DATE_RANGES.find(d => d.key === dateRange)?.label} · Last ${days} days`,
   101→   101→          config: JSON.stringify({ days, range: dateRange }),
   102→   102→          createdBy: null,
   103→   103→        }),
   104→   104→      });
   105→   105→      if (res.ok) {
   106→   106→        toast.success("Report generated", { description: "Scroll down to view the full report" });
   107→   107→        setShowReport(true);
   108→   108→        reload();
   109→   109→      } else {
   110→   110→        toast.error("Failed to generate report");
   111→   111→      }
   112→   112→    } catch {
   113→   113→      toast.error("Network error");
   114→   114→    } finally {
   115→   115→      setGenerating(false);
   116→   116→    }
   117→   117→  }
   118→   118→
   119→   119→  async function downloadPDF(r: ReportData) {
   120→   120→    try {
   121→   121→      const token = getToken();
   122→   122→      const res = await fetch(`/api/reports/${r.id}/pdf`, {
   123→   123→        headers: token ? { "Authorization": `Bearer ${token}` } : {},
   124→   124→      });
   125→   125→      if (!res.ok) throw new Error("PDF generation failed");
   126→   126→      const blob = await res.blob();
   127→   127→      const url = URL.createObjectURL(blob);
   128→   128→      const a = document.createElement("a");
   129→   129→      a.href = url;
   130→   130→      a.download = `${r.name.replace(/[^a-z0-9]/gi, "_")}.pdf`;
   131→   131→      a.click();
   132→   132→      URL.revokeObjectURL(url);
   133→   133→      toast.success("PDF downloaded");
   134→   134→    } catch { toast.error("Failed to generate PDF"); }
   135→   135→  }

   137→   137→  async function deleteReport(id: string) {

   141→   141→      method: "DELETE",
   142→   142→      headers: token ? { "Authorization": `Bearer ${token}` } : {},
   143→   143→    });
   144→   144→    toast.success("Report deleted");
   145→   145→    reload();
   146→   146→  }
