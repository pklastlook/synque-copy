   110→function platformProfileUrl(platform: PlatformId, handle: string): string | null {
   111→  const h = handle.replace(/^@/, "");
   112→  switch (platform) {
   113→    case "instagram":
   114→      return `https://instagram.com/${h}`;
   115→    case "facebook":
   116→      return `https://facebook.com/${h}`;
   117→    case "x":
   118→      return `https://x.com/${h}`;
   119→    case "linkedin":
   120→      return `https://linkedin.com/company/${h}`;
   121→    case "tiktok":
   122→      return `https://tiktok.com/@${h}`;
   123→    case "youtube":
   124→      return `https://youtube.com/@${h}`;
   125→    case "threads":
   126→      return `https://threads.net/@${h}`;
   127→    default:
   128→      return null;
   129→  }
   130→}

   534→
   535→async function patchCompetitor(
   536→  id: string,
   537→  patch: Partial<Pick<Competitor, "name" | "handle" | "platform" | "notes" | "status">>,
   538→  reload: () => void,
   539→) {
   540→  try {
   541→    const res = await fetch(`/api/competitors/${id}`, {
   542→      method: "PATCH",
   543→      headers: { "Content-Type": "application/json" },
   544→      body: JSON.stringify(patch),
   545→    });
   546→    if (!res.ok) throw new Error("Failed");
   547→    toast.success("Competitor updated");
   548→    reload();
   549→  } catch {
   550→    toast.error("Could not update competitor");
   551→  }
   552→}
   553→
   554→async function deleteCompetitor(
   555→  id: string,
   556→  name: string,
   557→  reload: () => void,
   558→) {
   559→  try {
   560→    const res = await fetch(`/api/competitors/${id}`, { method: "DELETE" });
   561→    if (!res.ok) throw new Error("Failed");
   562→    toast.success("Competitor removed", { description: name });
   563→    reload();
   564→  } catch {
   565→    toast.error("Could not delete competitor");
   566→  }
   567→}

   569→function AddCompetitorForm({
   570→  onCancel,
   571→  onDone,
   572→}: {
   573→  onCancel: () => void;
   574→  onDone: () => void;
   575→}) {
   576→  const [name, setName] = React.useState("");
   577→  const [handle, setHandle] = React.useState("");
   578→  const [platform, setPlatform] = React.useState<PlatformId>("instagram");
   579→  const [notes, setNotes] = React.useState("");
   580→  const [busy, setBusy] = React.useState(false);
   581→
   582→  const submit = async () => {
   583→    if (!name.trim() || !handle.trim()) {
   584→      toast.error("Name and handle are required");
   585→      return;
   586→    }
   587→    setBusy(true);
   588→    try {
   589→      const res = await fetch("/api/competitors", {
   590→        method: "POST",
   591→        headers: { "Content-Type": "application/json" },
   592→        body: JSON.stringify({
   593→          name: name.trim(),
   594→          handle: handle.trim().replace(/^@/, ""),
   595→          platform,
   596→          notes: notes.trim() || null,
   597→        }),
   598→      });
   599→      if (!res.ok) throw new Error("Create failed");
   600→      toast.success("Competitor added");
   601→      setName("");
   602→      setHandle("");
   603→      setNotes("");
   604→      onDone();

   955→function EditCompetitorDialog({
   956→  competitor,
   957→  onClose,
   958→  onSaved,
   959→}: {
   960→  competitor: Competitor | null;
   961→  onClose: () => void;
   962→  onSaved: () => void;
   963→}) {
   964→  const [name, setName] = React.useState("");
   965→  const [handle, setHandle] = React.useState("");
   966→  const [platform, setPlatform] = React.useState<PlatformId>("instagram");
   967→  const [notes, setNotes] = React.useState("");
   968→  const [busy, setBusy] = React.useState(false);
   969→
   970→  React.useEffect(() => {
   971→    if (competitor) {
   972→      setName(competitor.name);
   973→      setHandle(competitor.handle);
   974→      setPlatform(competitor.platform);
   975→      setNotes(competitor.notes ?? "");
   976→    }
   977→  }, [competitor]);
   978→
   979→  const submit = async () => {
   980→    if (!competitor) return;
   981→    if (!name.trim() || !handle.trim()) {
   982→      toast.error("Name and handle are required");
   983→      return;
   984→    }
   985→    setBusy(true);
   986→    try {
   987→      const res = await fetch(`/api/competitors/${competitor.id}`, {
   988→        method: "PATCH",
   989→        headers: { "Content-Type": "application/json" },
   990→        body: JSON.stringify({
   991→          name: name.trim(),
   992→          handle: handle.trim().replace(/^@/, ""),
   993→          platform,
   994→          notes: notes.trim() || null,
   995→        }),
   996→      });
   997→      if (!res.ok) throw new Error("Update failed");
   998→      toast.success("Competitor updated");
   999→      onSaved();
  1000→    } catch {
  1001→      toast.error("Could not update competitor");
  1002→    } finally {
  1003→      setBusy(false);
  1004→    }
  1005→  };
  1006→
  1007→  return (
  1008→    <Dialog open={!!competitor} onOpenChange={(o) => !o && onClose()}>
  1009→      <DialogContent className="rounded-3xl sm:max-w-md">
  1010→        <DialogHeader>
  1011→          <DialogTitle className="flex items-center gap-2">
  1012→            <Pencil className="h-5 w-5 text-primary" />
  1013→            Edit competitor
  1014→          </DialogTitle>
  1015→          <DialogDescription>
  1016→            Update the rival&apos;s profile details. Tracking status and notes can be changed here.
  1017→          </DialogDescription>
  1018→        </DialogHeader>
  1019→        <div className="space-y-4 py-2">
  1020→          <div className="space-y-2">
  1021→            <Label htmlFor="ce-name">Name</Label>
  1022→            <Input
  1023→              id="ce-name"
  1024→              value={name}
  1025→              onChange={(e) => setName(e.target.value)}
  1026→            />
  1027→          </div>
  1028→          <div className="grid grid-cols-2 gap-3">
  1029→            <div className="space-y-2">
  1030→              <Label htmlFor="ce-handle">Handle</Label>
  1031→              <Input
  1032→                id="ce-handle"
  1033→                value={handle}
  1034→                onChange={(e) => setHandle(e.target.value)}
  1035→              />
  1036→            </div>
  1037→            <div className="space-y-2">
  1038→              <Label>Platform</Label>
  1039→              <Select
  1040→                value={platform}
  1041→                onValueChange={(v) => setPlatform(v as PlatformId)}
  1042→              >
  1043→                <SelectTrigger className="w-full">
  1044→                  <SelectValue />
  1045→                </SelectTrigger>
  1046→                <SelectContent>
  1047→                  {PLATFORM_LIST.map((p) => (
  1048→                    <SelectItem key={p.id} value={p.id}>
  1049→                      {p.name}
  1050→                    </SelectItem>
  1051→                  ))}
  1052→                </SelectContent>
  1053→              </Select>
  1054→            </div>
  1055→          </div>
  1056→          <div className="space-y-2">
  1057→            <Label htmlFor="ce-notes">Notes</Label>
  1058→            <Textarea
  1059→              id="ce-notes"
  1060→              value={notes}
  1061→              onChange={(e) => setNotes(e.target.value)}
  1062→              className="min-h-[80px] resize-none"
  1063→            />
  1064→          </div>
  1065→        </div>
  1066→        <DialogFooter>
  1067→          <Button variant="ghost" onClick={onClose} disabled={busy}>
  1068→            Cancel
  1069→          </Button>
  1070→          <button
  1071→            onClick={submit}
