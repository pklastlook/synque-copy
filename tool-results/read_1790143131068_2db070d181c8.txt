   299→function ProfileSection() {
   300→  const { user } = useAuth();
   301→  const { reload } = useApi<ApiUser[]>("/api/users");
   302→  const [name, setName] = React.useState("");
   303→  const [email, setEmail] = React.useState("");
   304→  const [title, setTitle] = React.useState("");
   305→  const [department, setDepartment] = React.useState<Department>("Leadership");
   306→  const [bio, setBio] = React.useState("");
   307→  const [timezone, setTimezone] = React.useState("America/Los_Angeles");
   308→  const [language, setLanguage] = React.useState("en-US");
   309→  const [saving, setSaving] = React.useState(false);
   310→
   311→  // Load real user data
   312→  React.useEffect(() => {
   313→    if (!user) return;
   314→    setName(user.name);
   315→    setEmail(user.email);
   316→    setTitle(user.title ?? "");
   317→    setDepartment((user.title as Department) ?? "General");
   318→  }, [user]);
   319→
   320→  const save = async () => {
   321→    if (!user) return;
   322→    setSaving(true);
   323→    try {
   324→      await fetch("/api/users", {
   325→        method: "PATCH",
   326→        headers: { "Content-Type": "application/json" },
   327→        body: JSON.stringify({ id: user.id, name, email, title, department }),
   328→      });
   329→      toast.success("Profile saved", { description: "Your changes have been synced." });
   330→      reload();
   331→    } catch {
   332→      toast.error("Failed to save profile");
   333→    } finally {
   334→      setSaving(false);
   335→    }
   336→  };
   337→
   338→  return (
   339→    <GlassCard className="crystal-in p-5">
   340→      <SectionTitle
   341→        title="Profile"
   342→        desc="Your personal information visible to your team."
   343→        action={
   344→          <button
   345→            onClick={save}
   346→            disabled={saving}
   347→            className="liquid-pill flex items-center gap-1.5 rounded-xl px-3.5 py-1.5 text-xs font-semibold disabled:opacity-60"
   348→          >
   349→            <Save className="h-3.5 w-3.5" /> {saving ? "Saving…" : "Save changes"}

   351→        }

   357→        <div className="flex flex-col gap-2">
   358→          <div className="text-sm font-semibold">{name}</div>
   359→          <div className="text-xs text-muted-foreground">PNG / JPG up to 4MB · square recommended</div>
   360→          <div className="flex gap-2">

   770→  const [invitePassword, setInvitePassword] = React.useState("");
   771→  const [inviteLoading, setInviteLoading] = React.useState(false);
   772→  const [saving, setSaving] = React.useState(false);
   773→

   776→    if (!data) return;

   778→      id: u.id,

   817→          fetch("/api/users", {
   818→            method: "PATCH",
   819→            headers: { "Content-Type": "application/json" },

   821→              id: m.id,
   822→              role: m.role,
   823→              title: m.title,

   861→            body: JSON.stringify({ id: data.user.id, role: inviteRole, title: inviteRole === "admin" ? "Administrator" : "Team Member" }),
   862→          }).catch(() => {});
