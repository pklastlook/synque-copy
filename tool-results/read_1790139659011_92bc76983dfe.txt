   476→   476→/* ============================== 3. Team performance table ============================== */
   477→   477→
   478→   478→type SortKey = "engagement" | "posts" | "followers" | "accounts" | "tasks";
   479→   479→type SortDir = "asc" | "desc";
   480→   480→
   481→   481→function TeamPerformanceSection({ members }: { members: TeamMember[] }) {
   482→   482→  const [sortKey, setSortKey] = React.useState<SortKey>("engagement");
   483→   483→  const [sortDir, setSortDir] = React.useState<SortDir>("desc");
   484→   484→  const [selected, setSelected] = React.useState<TeamMember | null>(null);
   485→   485→
   486→   486→  const sorted = React.useMemo(() => {
   487→   487→    const copy = [...members];
   488→   488→    copy.sort((a, b) => {
   489→   489→      const va =
   490→   490→        sortKey === "tasks"
   491→   491→          ? a.tasksDone
   492→   492→          : sortKey === "engagement"
   493→   493→            ? a.engagement
   494→   494→            : sortKey === "posts"
   495→   495→              ? a.postsCount
   496→   496→              : sortKey === "followers"
   497→   497→                ? a.followers
   498→   498→                : a.accountsCount;
   499→   499→      const vb =
   500→   500→        sortKey === "tasks"
   501→   501→          ? b.tasksDone
   502→   502→          : sortKey === "engagement"
   503→   503→            ? b.engagement
   504→   504→            : sortKey === "posts"
   505→   505→              ? b.postsCount
   506→   506→              : sortKey === "followers"
   507→   507→                ? b.followers
   508→   508→                : b.accountsCount;
   509→   509→      return sortDir === "asc" ? va - vb : vb - va;
   510→   510→    });
   511→   511→    return copy;
   512→   512→  }, [members, sortKey, sortDir]);
