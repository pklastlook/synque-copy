   228→  const { data: accounts } = useApi<AccountOption[]>("/api/accounts");

   232→  const [editing, setEditing] = React.useState<Report | null>(null);
   233→  const [selectedId, setSelectedId] = React.useState<string | null>(null);
   234→
   235→  const reports = data ?? [];
   236→  const selected = reports.find((r) => r.id === selectedId) ?? null;
   237→
   238→  const totals = React.useMemo(() => {
   239→    const scheduled = reports.filter((r) => r.schedule).length;
   240→    const types = new Set(reports.map((r) => r.type)).size;
   241→    const now = new Date();
   242→    const dayMs = 86_400_000;
   243→    const weekAgo = new Date(now.getTime() - 7 * dayMs);
   244→    const generatedThisWeek = reports.filter(
   245→      (r) => r.lastGenerated && new Date(r.lastGenerated) >= weekAgo,
   246→    ).length;
   247→    return {
   248→      total: reports.length,
   249→      scheduled,
   250→      types,
   251→      generatedThisWeek,
   252→    };
   253→  }, [reports]);
   254→
   255→  const handleCreateFromTemplate = async (tpl: (typeof TEMPLATES)[number]) => {
   256→    const config: ReportConfig = {
   257→      metrics: tpl.metrics,
   258→      ...(tpl.dateRange ? { dateRange: tpl.dateRange } : {}),
   259→      accounts: "all",
   260→      ...(tpl.groupBy ? { groupBy: tpl.groupBy } : {}),
   261→    };
   262→    try {
   263→      const res = await fetch("/api/reports", {
   264→        method: "POST",
   265→        headers: { "Content-Type": "application/json" },
   266→        body: JSON.stringify({
   267→          name: tpl.name,
   268→          type: tpl.type,
   269→          description: tpl.desc,
   270→          config: JSON.stringify(config),
   271→          schedule: tpl.schedule ?? null,
   272→          createdBy: user?.id ?? null,
   273→        }),
   274→      });
   275→      if (!res.ok) throw new Error("Failed");
   276→      toast.success(`Created report from "${tpl.name}"`);
   277→      reload();
   278→    } catch {
   279→      toast.error("Could not create report");
   280→    }
   281→  };
   282→
   283→  const handleGenerate = async (report: Report) => {
   284→    try {
   285→      const res = await fetch(`/api/reports/${report.id}`, {
   286→        method: "PATCH",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({ lastGenerated: new Date().toISOString() }),
   289→      });
   290→      if (!res.ok) throw new Error("Failed");
   291→      toast.success("Report generated", { description: report.name });
   292→      reload();
   293→    } catch {
   294→      toast.error("Could not generate report");
   295→    }
   296→  };
   297→
   298→  const handleDelete = async (report: Report) => {

   300→      const res = await fetch(`/api/reports/${report.id}`, { method: "DELETE" });
   301→      if (!res.ok) throw new Error("Failed");
   302→      toast.success("Report deleted", { description: report.name });
   303→      if (selectedId === report.id) setSelectedId(null);
   304→      if (editing?.id === report.id) setEditing(null);
   305→      reload();
   306→    } catch {
   307→      toast.error("Could not delete report");
   308→    }
   309→  };

   316→      config = {};

   319→      const res = await fetch("/api/reports", {
   320→        method: "POST",

   323→          name: `${report.name} (copy)`,
   324→          type: report.type,
   325→          description: report.description,
   326→          config: JSON.stringify(config),
   327→          schedule: report.schedule,
   328→          createdBy: user?.id ?? null,
   329→        }),
   330→      });
   331→      if (!res.ok) throw new Error("Failed");
   332→      toast.success("Report duplicated", { description: report.name });
   333→      reload();
   334→    } catch {
   335→      toast.error("Could not duplicate report");
   336→    }
   337→  };

   356→              <span className="font-semibold text-foreground">{totals.total}</span> reports ·{" "}
   357→              <span className="font-semibold text-foreground">{totals.scheduled}</span> scheduled ·{" "}

   359→            </p>

   386→        />

   412→            const Icon = tpl.icon;

   414→              <button

   427→                  >
   428→                    <Icon className="h-5 w-5" />
   429→                  </div>
   430→                  <div className="min-w-0">
   431→                    <div className="truncate text-sm font-semibold">{tpl.name}</div>
   432→                    <div className="text-[11px] uppercase tracking-wide text-muted-foreground">

   522→  onGenerate,

   525→  onEdit,

   558→      />
   559→      <div className="relative flex items-start justify-between gap-2">
   560→        <button onClick={onSelect} className="flex min-w-0 items-center gap-3 text-left">
   561→          <div
   562→            className="grid h-10 w-10 shrink-0 place-items-center rounded-2xl text-white shadow-md"
   563→            style={{ background: meta.color }}
   564→          >

   588→              <Copy className="mr-2 h-3.5 w-3.5" /> Duplicate
   589→            </DropdownMenuItem>
   590→            <DropdownMenuSeparator />
   591→            <DropdownMenuItem
   592→              onSelect={onDelete}
   593→              className="text-rose-500 focus:text-rose-500"
   594→            >
   595→              <Trash2 className="mr-2 h-3.5 w-3.5" /> Delete
   596→            </DropdownMenuItem>
   597→          </DropdownMenuContent>
   598→        </DropdownMenu>
   599→      </div>

   665→function ReportPreview({
   666→  report,
   667→  lastGenerated,
   668→}: {
   669→  report: Report;
   670→  lastGenerated: string | null;
   671→}) {
   672→  const meta = TYPE_META[report.type];
   673→  const Icon = meta.icon;
   674→
   675→  let config: ReportConfig = {};
   676→  try {
   677→    config = JSON.parse(report.config) as ReportConfig;
   678→  } catch {
   679→    config = {};
   680→  }
   681→
   682→  const days = DATE_RANGE_DAYS[config.dateRange ?? "30d"] ?? 30;
   683→  const accountId = config.accounts && config.accounts !== "all" ? config.accounts : "all";
   684→  const analyticsUrl = `/api/analytics?days=${days}&accountId=${accountId}`;
   685→  const { data: analytics, loading } = useApi<AnalyticsResponse>(analyticsUrl);
   686→
   687→  const metricRows = (config.metrics && config.metrics.length > 0
   688→    ? config.metrics
   689→    : ["impressions", "reach", "engagement", "followers"]
   690→  ).slice(0, 5);
   691→
   692→  // Pick the first metric that has a matching series field for the trend chart.
   693→  const trendMetric =
   694→    metricRows.find((m) => (SERIES_METRICS as readonly string[]).includes(m)) ??
   695→    "impressions";
   696→
   697→  // Compute trend series mapped to {date, value} from the analytics daily series.
   698→  const series = (() => {
   699→    if (!analytics?.series) return [];
   700→    return analytics.series.map((p) => {
   701→      const row = p as unknown as Record<string, number>;
   702→      return {
   703→        date: p.date,
   704→        value: row[trendMetric] ?? 0,
   705→      };
   706→    });
   707→  })();
   708→
   709→  // Compute metric snapshots from perAccount sums.
   710→  const snapshot: Record<string, number> = (() => {
   711→    const sums: Record<string, number> = {
   712→      impressions: 0,
   713→      reach: 0,
   714→      engagement: 0,
   715→      followers: 0,
   716→      posts: 0,
   717→      likes: 0,
   718→      comments: 0,
   719→      shares: 0,
   720→      saves: 0,
   721→    };
   722→    if (!analytics?.perAccount) return sums;
   723→    for (const a of analytics.perAccount) {
   724→      sums.impressions += a.impressions;
   725→      sums.reach += a.reach;
   726→      sums.engagement += a.engagement;
   727→      sums.followers += a.followers;
   728→      sums.posts += a.posts;
   729→      sums.likes += a.likes;
   730→      sums.comments += a.comments;
   731→      sums.shares += a.shares;
   732→      sums.saves += a.saves;
   733→    }
   734→    return sums;
   735→  })();
   736→
   737→  // Compute the period-over-period delta for trendMetric using first vs last.
   738→  const first = series[0]?.value ?? 0;
   739→  const last = series[series.length - 1]?.value ?? 0;
   740→  const deltaPct = series.length >= 2 && first !== 0 ? Math.round(((last - first) / first) * 100) : null;
   741→
   742→  return (
   743→    <GlassCard className="animate-rise overflow-hidden p-5">
   744→      <SectionTitle
   745→        title={report.name}
   746→        desc={`${meta.label} report · ${config.dateRange ?? "30d"} range · ${lastGenerated ? `Generated ${timeAgo(lastGenerated)}` : "Not generated yet"}`}
   747→        action={
   748→          <div
   749→            className="flex items-center gap-2 rounded-full px-2.5 py-1 text-[11px] font-semibold text-white shadow"
   750→            style={{ background: meta.color }}
   751→          >
   752→            <Icon className="h-3.5 w-3.5" />
   753→            {meta.label}
   754→          </div>
   755→        }
   756→      />
   757→
   758→      {loading || !analytics ? (
   759→        <div className="grid gap-4 lg:grid-cols-3">
   760→          <div className="shimmer h-[260px] rounded-2xl lg:col-span-2" />
   761→          <div className="shimmer h-[260px] rounded-2xl" />
   762→        </div>
   763→      ) : analytics.series.length === 0 ? (
   764→        <div className="rounded-2xl border border-border/40 bg-background/40 p-6 text-center text-sm text-muted-foreground">
   765→          No analytics data in this range yet. Generate the report after publishing content.
   766→        </div>
   767→      ) : (
   768→        <div className="grid gap-4 lg:grid-cols-3">
   769→          <div className="lg:col-span-2">
   770→            <div className="rounded-2xl border border-border/40 bg-background/40 p-4">
   771→              <div className="mb-2 flex items-center justify-between">
   772→                <span className="text-xs font-medium text-muted-foreground capitalize">
   773→                  {trendMetric} trend
   774→                </span>
   775→                {deltaPct !== null && (
   776→                  <span
   777→                    className={cn(
   778→                      "text-[11px] font-semibold",
   779→                      deltaPct >= 0 ? "text-emerald-500" : "text-rose-500",
   780→                    )}
   781→                  >
   782→                    {deltaPct >= 0 ? "+" : ""}
   783→                    {deltaPct}%
   784→                  </span>
   785→                )}
   786→              </div>
   787→              <div className="h-[200px] w-full">
   788→                <ResponsiveContainer width="100%" height="100%">
   789→                  <AreaChart data={series} margin={{ left: -16, right: 4, top: 4, bottom: 0 }}>
   790→                    <defs>
   791→                      <linearGradient id="prevGrad" x1="0" y1="0" x2="0" y2="1">
   792→                        <stop offset="0%" stopColor={meta.color} stopOpacity={0.45} />
   793→                        <stop offset="100%" stopColor={meta.color} stopOpacity={0} />
   794→                      </linearGradient>
   795→                    </defs>
   796→                    <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   797→                    <XAxis
   798→                      dataKey="date"
   799→                      tickFormatter={(d) =>
   800→                        new Date(d).toLocaleDateString(undefined, { day: "numeric", month: "short" })
   801→                      }
   802→                      tick={{ fontSize: 10, fill: "var(--muted-foreground)" }}
   803→                      axisLine={false}
   804→                      tickLine={false}
   805→                      interval={Math.max(0, Math.floor(series.length / 6))}
   806→                    />
   807→                    <YAxis
   808→                      tickFormatter={(v) => fmt(Number(v))}
   809→                      tick={{ fontSize: 10, fill: "var(--muted-foreground)" }}
   810→                      axisLine={false}
   811→                      tickLine={false}
   812→                      width={44}
   813→                    />
   814→                    <Tooltip
   815→                      contentStyle={TOOLTIP_STYLE}
   816→                      labelFormatter={(d) =>
   817→                        new Date(d).toLocaleDateString(undefined, {
   818→                          weekday: "short",
   819→                          month: "short",
   820→                          day: "numeric",
   821→                        })
   822→                      }
   823→                      formatter={(v: number) => [fmt(v), trendMetric]}
   824→                    />
   825→                    <Area
   826→                      type="monotone"
   827→                      dataKey="value"
   828→                      stroke={meta.color}
   829→                      strokeWidth={2.5}
   830→                      fill="url(#prevGrad)"
   831→                    />
   832→                  </AreaChart>
   833→                </ResponsiveContainer>
   834→              </div>
   835→            </div>
   836→          </div>
   837→
   838→          <div className="rounded-2xl border border-border/40 bg-background/40 p-4">
   839→            <div className="mb-2 flex items-center gap-2 text-xs font-medium text-muted-foreground">
   840→              <CheckCircle2 className="h-3.5 w-3.5 text-primary" /> Metrics snapshot
   841→            </div>
   842→            <div className="space-y-1.5">
   843→              {metricRows.map((m) => {
   844→                const value = snapshot[m] ?? 0;
   845→                return (
   846→                  <div
   847→                    key={m}
   848→                    className="flex items-center justify-between rounded-xl bg-muted/30 px-3 py-2 text-xs"
   849→                  >
   850→                    <span className="capitalize text-muted-foreground">{m}</span>

   852→                  </div>
   853→                );
   854→              })}
   855→            </div>
   856→          </div>
   857→        </div>
   858→      )}
   859→    </GlassCard>
