   130→export function ComposerView() {
   131→  const { composerSeed, clearComposerSeed } = useNav();
   132→  const { user } = useAuth();
   133→  const currentUserId = user?.id ?? "";
   134→  const { data: accounts } = useApi<AccountData[]>("/api/accounts");
   135→  const { data: mediaData } = useApi<MediaLibraryResponse>("/api/media");
   136→  const mediaAssets = React.useMemo(
   137→    () => (mediaData?.assets ?? []).filter((a) => a.type === "image"),
   138→    [mediaData]
   139→  );
   140→
   141→  const [state, setState] = React.useState<ComposerState>(DEFAULT_STATE);
   142→  const [generating, setGenerating] = React.useState(false);
   143→  const [publishing, setPublishing] = React.useState(false);
   144→  const [search, setSearch] = React.useState("");
   145→  const [uploadOpen, setUploadOpen] = React.useState(false);
   146→  const [urlInput, setUrlInput] = React.useState("");
   147→  const [cropAspect, setCropAspect] = React.useState("1:1");
   148→
   149→  const initRef = React.useRef(false);
   150→
   151→  // Preselect the top 4 accounts by followers (the /api/accounts route
   152→  // already sorts by followers desc) so the composer opens with a sensible
   153→  // multi-platform default — without relying on hardcoded seeded IDs.
   154→  React.useEffect(() => {
   155→    if (initRef.current || !accounts || accounts.length === 0) return;
   156→    initRef.current = true;
   157→    const presets = accounts.slice(0, 4).map((a) => a.id);
   158→    if (presets.length) {
   159→      setState((s) => ({ ...s, selectedAccountIds: presets }));
   160→    }
   161→  }, [accounts]);
   162→
   163→  // Composer seed prefill — set caption + jump to step 2 (keeps existing behavior)
   164→  React.useEffect(() => {
   165→    if (composerSeed) {
   166→      setState((s) => ({ ...s, caption: composerSeed, step: 2 }));
   167→      clearComposerSeed();
   168→    }
   169→  }, [composerSeed, clearComposerSeed]);
   170→
   171→  const update = React.useCallback((patch: Partial<ComposerState>) => {
   172→    setState((s) => ({ ...s, ...patch }));
   173→  }, []);
   174→
   175→  const accountsList = accounts ?? [];
   176→  const selectedAccounts = accountsList.filter((a) =>
   177→    state.selectedAccountIds.includes(a.id)
   178→  );
   179→  const filteredAccounts = accountsList.filter((a) => {
   180→    const q = search.trim().toLowerCase();
   181→    if (!q) return true;
   182→    return (
   183→      a.displayName.toLowerCase().includes(q) ||
   184→      a.handle.toLowerCase().includes(q) ||
   185→      PLATFORMS[a.platform].name.toLowerCase().includes(q)
   186→    );
   187→  });
   188→
   189→  function toggleAccount(id: string) {
   190→    setState((s) => {
   191→      const next = new Set(s.selectedAccountIds);
   192→      if (next.has(id)) next.delete(id);
   193→      else next.add(id);
   194→      return { ...s, selectedAccountIds: Array.from(next) };
   195→    });
   196→  }
   197→
   198→  const selectAll = () =>
   199→    setState((s) => ({ ...s, selectedAccountIds: accountsList.map((a) => a.id) }));
   200→  const clearAll = () => setState((s) => ({ ...s, selectedAccountIds: [] }));
   201→
   202→  const goToStep = (step: Step) => setState((s) => ({ ...s, step }));
   203→  const next = () =>
   204→    setState((s) => ({ ...s, step: Math.min(5, (s.step as number) + 1) as Step }));
   205→  const back = () =>
   206→    setState((s) => ({ ...s, step: Math.max(1, (s.step as number) - 1) as Step }));
   207→
   208→  async function generate(regenerateOnly?: string) {
   209→    if (!state.caption.trim()) {
   210→      toast.error("Write a caption first");
   211→      return;
   212→    }
   213→    if (selectedAccounts.length === 0) {
   214→      toast.error("Select at least one account");
   215→      return;
   216→    }
   217→    if (!user) {
   218→      toast.error("Your session is loading — please try again in a moment");
   219→      return;
   220→    }
   221→    setGenerating(true);
   222→    const targets = selectedAccounts.map((a) => ({
   223→      accountId: a.id,
   224→      platform: a.platform,
   225→      handle: a.handle,
   226→    }));
   227→    const bodyTargets = regenerateOnly
   228→      ? targets.filter((t) => t.accountId === regenerateOnly)
   229→      : targets;
   230→    try {
   231→      const res = await fetch("/api/ai/generate", {
   232→        method: "POST",
   233→        headers: { "Content-Type": "application/json" },
   234→        body: JSON.stringify({
   235→          caption: state.caption,
   236→          tone: state.tone,
   237→          targets: bodyTargets,
   238→          mediaHint: state.mediaUrl ? "attached visual" : "no media",
   239→          userId: currentUserId,
   240→        }),
   241→      });
   242→      const json = await res.json().catch(() => ({}));
   243→      const incoming = json.variants ?? json.fallback?.variants;
   244→      if (incoming && typeof incoming === "object") {
   245→        const normalized: Record<string, Variant> = {};
   246→        for (const k of Object.keys(incoming)) {
   247→          normalized[k] = {
   248→            content: incoming[k]?.content ?? "",
   249→            hashtags: incoming[k]?.hashtags ?? "",
   250→          };
   251→        }
   252→        if (regenerateOnly) {
   253→          const nv = normalized[regenerateOnly];
   254→          if (nv) {
   255→            setState((s) => ({
   256→              ...s,
   257→              variants: { ...s.variants, [regenerateOnly]: nv },
   258→            }));
   259→            toast.success("Regenerated variant");
   260→          } else {
   261→            toast.error("Variant not found in AI response");
   262→          }
   263→        } else {
   264→          setState((s) => ({ ...s, variants: normalized }));
   265→          toast.success(
   266→            `Generated ${Object.keys(normalized).length} platform-native captions`
   267→          );
   268→        }
   269→      } else {
   270→        toast.error(json.error ?? "Generation failed");
   271→      }
   272→    } catch {
   273→      toast.error("Network error generating captions");
   274→    } finally {
   275→      setGenerating(false);
   276→    }
   277→  }
   278→
   279→  async function publish() {
   280→    if (!user) {
   281→      toast.error("Your session is loading — please try again in a moment");
   282→      return;
   283→    }
   284→    if (selectedAccounts.length === 0) {
   285→      toast.error("Select accounts");
   286→      return;
   287→    }
   288→    setPublishing(true);
   289→    try {
   290→      // 1. Save the post to the database
   291→      const v: Record<string, { content: string; hashtags?: string }> = {};
   292→      for (const a of selectedAccounts) {
   293→        const ex = state.variants[a.id];
   294→        v[a.id] = ex ?? { content: state.caption, hashtags: "" };
   295→      }
   296→      const scheduledAt =
   297→        state.scheduleEnabled && state.scheduledAt
   298→          ? new Date(state.scheduledAt).toISOString()
   299→          : null;
   300→
   301→      const res = await fetch("/api/posts", {
   302→        method: "POST",
   303→        headers: { "Content-Type": "application/json" },
   304→        body: JSON.stringify({
   305→          title: state.caption.slice(0, 50).trim() || "Untitled",
   306→          caption: state.caption,
   307→          mediaUrl: state.mediaUrl || null,
   308→          tone: state.tone,
   309→          scheduledAt,
   310→          accountIds: selectedAccounts.map((a) => a.id),
   311→          variants: v,
   312→          authorId: currentUserId,
   313→        }),
   314→      });
   315→
   316→      if (!res.ok) {
   317→        toast.error("Failed to save post");
   318→        return;
   319→      }
   320→
   321→      // 2. If scheduled, skip real posting (will be posted later)
   322→      if (state.scheduleEnabled && scheduledAt) {
   323→        toast.success("Post scheduled", {
   324→          description: `Will be published on ${new Date(scheduledAt).toLocaleString()}`,
   325→        });
   326→        setState(DEFAULT_STATE);
   327→        return;
   328→      }
   329→
   330→      // 3. Attempt REAL platform publishing for each selected account
   331→      const results: { account: string; platform: string; success: boolean; message: string; url?: string }[] = [];
   332→
   333→      for (const a of selectedAccounts) {
   334→        const variant = v[a.id];
   335→        const content = variant.hashtags
   336→          ? `${variant.content}\n\n${variant.hashtags.split(",").map(t => `#${t.trim()}`).join(" ")}`
   337→          : variant.content;
   338→
   339→        // Map platform id to API platform name
   340→        const apiPlatform = a.platform === "x" ? "twitter" : a.platform;
   341→
   342→        try {
   343→          const pubRes = await fetch(`/api/publish/${apiPlatform}`, {
   344→            method: "POST",
   345→            headers: { "Content-Type": "application/json" },
   346→            body: JSON.stringify({
   347→              userId: currentUserId,
   348→              accountId: a.id,
   349→              content,
   350→            }),
   351→          });
   352→          const pubJson = await pubRes.json();
   353→
   354→          if (pubRes.ok && pubJson.ok) {
   355→            results.push({
   356→              account: a.displayName,
   357→              platform: a.platform,
   358→              success: true,
   359→              message: "Published successfully",
   360→              url: pubJson.url,
   361→            });
   362→          } else {
   363→            results.push({
   364→              account: a.displayName,
   365→              platform: a.platform,
   366→              success: false,
   367→              message: pubJson.error ?? "Failed to publish",
   368→            });
   369→          }
   370→        } catch {
   371→          results.push({
   372→            account: a.displayName,
   373→            platform: a.platform,
   374→            success: false,
   375→            message: "Network error during publish",
   376→          });
   377→        }
   378→      }
   379→
   380→      // 4. Show results with iOS-style popups
   381→      const succeeded = results.filter((r) => r.success);
   382→      const failed = results.filter((r) => !r.success);
   383→
   384→      if (succeeded.length > 0 && failed.length === 0) {
   385→        toast.success("Published!", {
   386→          description: `Posted to ${succeeded.length} account${succeeded.length > 1 ? "s" : ""}.`,
   387→        });
   388→        succeeded.forEach((r) => {
   389→          if (r.url) {
   390→            toast.success(r.account, {
   391→              description: `View tweet: ${r.url}`,
   392→              duration: 6000,
   393→            });
   394→          }
   395→        });
   396→      } else if (succeeded.length > 0 && failed.length > 0) {
   397→        toast.warning("Partial publish", {
   398→          description: `Published to ${succeeded.length}, failed ${failed.length}.`,
   399→          duration: 6000,
   400→        });
   401→        failed.forEach((r) => {
   402→          toast.error(`Failed: ${r.account}`, {
   403→            description: r.message.slice(0, 120),
   404→            duration: 8000,
   405→          });
   406→        });
   407→      } else {
   408→        // All failed — check if it's a 402 (Free tier paywall)
   409→        const firstError = failed[0];
   410→        const is402 = firstError?.message.includes("Free tier") || firstError?.message.includes("doesn't allow posting");
   411→
   412→        if (is402) {
   413→          // Copy content to clipboard for manual posting
   414→          const contentToCopy = selectedAccounts.map(a => {
   415→            const variant = v[a.id];
   416→            return variant.hashtags
   417→              ? `${variant.content}\n\n${variant.hashtags.split(",").map(t => `#${t.trim()}`).join(" ")}`
   418→              : variant.content;
   419→          }).join("\n\n---\n\n");
   420→
   421→          try {
   422→            await navigator.clipboard.writeText(contentToCopy);
   423→            toast.warning("Post saved — copy to clipboard ready", {
   424→              description: "Twitter Free tier doesn't allow API posting. Content copied! Click 'Open X' to paste manually.",
   425→              duration: 10000,
   426→              action: {
   427→                label: "Open X",
   428→                onClick: () => window.open("https://x.com/compose/post", "_blank"),
   429→              },
   430→            });
   431→          } catch {
   432→            toast.warning("Post saved — manual posting required", {
   433→              description: "Twitter Free tier doesn't allow API posting. Upgrade to Basic ($100/mo) or post manually at x.com/compose/post",
   434→              duration: 10000,
   435→              action: {
   436→                label: "Open X",
   437→                onClick: () => window.open("https://x.com/compose/post", "_blank"),
   438→              },
   439→            });
   440→          }
   441→        } else {
   442→          toast.error("Post saved but not published", {
   443→            description: firstError?.message?.slice(0, 150) ?? "Unknown error.",
   444→            duration: 8000,
   445→          });
   446→        }
   447→      }
   448→
   449→      setState(DEFAULT_STATE);
   450→    } catch {
   451→      toast.error("Network error");
   452→    } finally {
   453→      setPublishing(false);
   454→    }
   455→  }
   456→
   457→  function submitUrl() {
   458→    if (!urlInput.trim()) {
   459→      toast.error("Enter an image URL");
   460→      return;
   461→    }
   462→    update({ mediaUrl: urlInput.trim() });
   463→    setUrlInput("");
   464→    setUploadOpen(false);
   465→    toast.success("Media URL added");
   466→  }
   467→
   468→  const canContinue =
   469→    state.step === 1
   470→      ? selectedAccounts.length > 0
   471→      : state.step === 2
   472→      ? state.caption.trim().length > 0
   473→      : state.step === 3
   474→      ? !!state.mediaUrl
   475→      : state.step === 4
   476→      ? true
   477→      : false;
   478→
   479→  return (
   480→    <div className="space-y-5 pb-28">
   481→      <StepIndicator step={state.step} onStep={goToStep} />
   482→
   483→      <div className="mx-auto w-full max-w-3xl space-y-5">
   484→        {state.step === 1 && (
   485→          <Step1
   486→            accounts={filteredAccounts}
   487→            selectedIds={state.selectedAccountIds}
   488→            onToggle={toggleAccount}
   489→            onSelectAll={selectAll}
   490→            onClear={clearAll}
   491→            search={search}
   492→            onSearch={setSearch}
   493→            selectedAccounts={selectedAccounts}
   494→          />
   495→        )}
   496→        {state.step === 2 && (
   497→          <Step2
   498→            caption={state.caption}
   499→            onCaption={(caption) => update({ caption })}
   500→            tone={state.tone}
   501→            onTone={(tone) => update({ tone })}
   502→            onGenerate={() => generate()}
   503→            generating={generating}
   504→            hasVariants={Object.keys(state.variants).length > 0}
   505→            selectedCount={selectedAccounts.length}
   506→            variants={state.variants}
   507→            selectedAccounts={selectedAccounts}
   508→          />
   509→        )}
   510→        {state.step === 3 && (
   511→          <Step3
   512→            mediaUrl={state.mediaUrl}
   513→            mediaType={state.mediaType}
   514→            altText={state.altText}
   515→            onMedia={(mediaUrl) => update({ mediaUrl })}
   516→            onMediaType={(mediaType) => update({ mediaType })}
   517→            onAltText={(altText) => update({ altText })}
   518→            uploadOpen={uploadOpen}
   519→            onUploadOpen={setUploadOpen}
   520→            urlInput={urlInput}
   521→            onUrlInput={setUrlInput}
   522→            onUrlSubmit={submitUrl}
   523→            mediaAssets={mediaAssets}
   524→            user={user}
   525→            cropAspect={cropAspect}
   526→            onCropAspect={setCropAspect}
   527→          />
   528→        )}
   529→        {state.step === 4 && (
   530→          <Step4
   531→            accounts={selectedAccounts}
   532→            variants={state.variants}
   533→            onVariantChange={(id, v) =>
   534→              setState((s) => ({ ...s, variants: { ...s.variants, [id]: v } }))
   535→            }
   536→            mediaUrl={state.mediaUrl}
   537→            caption={state.caption}
   538→            onRegenerate={(id) => generate(id)}
   539→            onRegenerateAll={() => generate()}
   540→            generating={generating}
   541→          />
   542→        )}
   543→        {state.step === 5 && (
   544→          <Step5
   545→            accounts={selectedAccounts}
   546→            mediaUrl={state.mediaUrl}
   547→            mediaType={state.mediaType}
   548→            tone={state.tone}
   549→            caption={state.caption}
   550→            scheduleEnabled={state.scheduleEnabled}
   551→            onScheduleEnabled={(scheduleEnabled) => update({ scheduleEnabled })}
   552→            scheduledAt={state.scheduledAt}
   553→            onScheduledAt={(scheduledAt) => update({ scheduledAt })}
   554→            hasVariants={Object.keys(state.variants).length > 0}
   555→          />
   556→        )}
   557→      </div>
   558→
   559→      <StepFooter
   560→        step={state.step}
   561→        onBack={back}
   562→        onNext={next}
   563→        canContinue={canContinue}
   564→        onPublish={publish}
   565→        publishing={publishing}
   566→        scheduleEnabled={state.scheduleEnabled}
   567→      />
   568→    </div>
   569→  );
   570→}

   572→/* ============================================================
   573→   Step indicator — sticky, responsive (collapses on mobile)
   574→   ============================================================ */
   575→function StepIndicator({
   576→  step,
   577→  onStep,
   578→}: {
   579→  step: Step;
   580→  onStep: (s: Step) => void;
   581→}) {
   582→  const current = STEPS.find((s) => s.id === step) ?? STEPS[0];
   583→  return (
   584→    <div className="sticky top-0 z-30 -mx-4 px-4 py-2 sm:-mx-6 sm:px-6 lg:-mx-8 lg:px-8">
   585→      <div className="glass-strong glass-sheen rounded-2xl px-4 py-3 sm:px-5">
   586→        {/* Mobile — compact */}
   587→        <div className="flex items-center justify-between gap-3 sm:hidden">
   588→          <div className="flex items-center gap-2.5">
   589→            <span className="grid h-8 w-8 place-items-center rounded-full bg-primary text-xs font-bold text-primary-foreground glow-pulse">
   590→              {step}
   591→            </span>
   592→            <div className="min-w-0">
   593→              <div className="text-[10px] uppercase tracking-wider text-muted-foreground">
   594→                Step {step} of {STEPS.length}
   595→              </div>
   596→              <div className="truncate text-sm font-semibold">{current.label}</div>
   597→            </div>
   598→          </div>
   599→          <div className="flex gap-1">
   600→            {STEPS.map((s) => (
   601→              <span
   602→                key={s.id}
   603→                className={cn(
   604→                  "h-1.5 rounded-full transition-all",
   605→                  s.id === step
   606→                    ? "w-5 bg-primary"
   607→                    : s.id < step
