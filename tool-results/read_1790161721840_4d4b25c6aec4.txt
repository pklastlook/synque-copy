   111→/** Deterministic color from a string for trending tag pills. */
   112→function colorFor(tag: string) {
   113→  let h = 0;
   114→  for (let i = 0; i < tag.length; i++) h = tag.charCodeAt(i) + ((h << 5) - h);
   115→  return PILL_PALETTE[Math.abs(h) % PILL_PALETTE.length];
   116→}
   117→
   118→/** Split tags string on commas or newlines, trim, strip leading '#', drop empties. */
   119→function parseTags(raw: string): string[] {
   120→  return raw
   121→    .split(/[,\n]/)
   122→    .map((s) => s.trim().replace(/^#/, ""))
   123→    .filter(Boolean);
   124→}

   376→function GroupCard({
   377→  group,
   378→  maxUsage,
   379→  onUseInComposer,
   380→  onEdit,
   381→  onDelete,
   382→}: {
   383→  group: HashtagGroup;
   384→  maxUsage: number;
   385→  onUseInComposer: () => void;
   386→  onEdit: () => void;
   387→  onDelete: () => void;
   388→}) {
   389→  const tags = parseTags(group.tags);
   390→  const usagePct = Math.round((group.usageCount / maxUsage) * 100);
   391→
   392→  function copyTag(tag: string) {
   393→    navigator.clipboard?.writeText(`#${tag}`).catch(() => {});
   394→    toast.success(`Copied #${tag}`);
   395→  }
   396→
   397→  function copyAll() {
   398→    const text = tags.map((t) => `#${t}`).join(" ");
   399→    navigator.clipboard?.writeText(text).catch(() => {});
   400→    toast.success(`Copied ${tags.length} hashtags`, {
   401→      description: "Ready to paste into your post",
   402→    });
   403→  }
   404→
   405→  return (
   406→    <GlassCard hover className="glass-sheen group relative flex flex-col overflow-hidden p-5">
   407→      {/* Top accent */}
   408→      <div
   409→        className="absolute inset-x-0 top-0 h-1.5"
   410→        style={{ background: group.color }}
   411→      />
   412→      <div
   413→        className="pointer-events-none absolute -right-8 -top-8 h-28 w-28 rounded-full opacity-20 blur-2xl transition group-hover:opacity-40"
   414→        style={{ background: group.color }}
   415→      />
   416→
   417→      {/* Header */}
   418→      <div className="relative mt-1 flex items-start justify-between gap-2">
   419→        <div className="min-w-0">
   420→          <h3 className="truncate text-base font-bold leading-tight">{group.name}</h3>

   422→            <PlatformCell platform={group.platform} />

   424→        </div>
   425→        <div className="flex shrink-0 items-center gap-1">

   427→            <DropdownMenuTrigger asChild>
   428→              <button
   429→                className="grid h-7 w-7 place-items-center rounded-lg text-muted-foreground transition hover:bg-muted hover:text-foreground"
   430→                aria-label={`Actions for ${group.name}`}
   431→              >

   437→                <Pencil className="mr-2 h-4 w-4" /> Edit

   444→                <Trash2 className="mr-2 h-4 w-4" /> Delete

   449→            className="grid h-9 w-9 shrink-0 place-items-center rounded-2xl text-white shadow-sm"
   450→            style={{ background: group.color }}
   451→          >
   452→            <Hash className="h-4 w-4" />
   453→          </span>
   454→        </div>

   520→          Use in composer

   522→      </div>
   523→    </GlassCard>
   524→  );
   525→}

   527→function CreateGroupDialog({ onCreated }: { onCreated: () => void }) {
   528→  const [open, setOpen] = React.useState(false);
   529→  const [name, setName] = React.useState("");
   530→  const [tags, setTags] = React.useState("");
   531→  const [platform, setPlatform] = React.useState<HashtagPlatform>("all");
   532→  const [color, setColor] = React.useState(SWATCHES[0]);
   533→  const [saving, setSaving] = React.useState(false);
   534→
   535→  function reset() {
   536→    setName("");
   537→    setTags("");
   538→    setPlatform("all");
   539→    setColor(SWATCHES[0]);
   540→  }
   541→
   542→  async function handleSubmit() {
   543→    if (!name.trim() || !tags.trim()) {
   544→      toast.error("Name and tags are required");
   545→      return;
   546→    }
   547→    setSaving(true);
   548→    try {
   549→      const res = await fetch("/api/hashtags", {
   550→        method: "POST",
   551→        headers: { "Content-Type": "application/json" },
   552→        body: JSON.stringify({
   553→          name: name.trim(),
   554→          tags: tags.trim(),
   555→          platform,
   556→          color,
   557→        }),
   558→      });
   559→      const json = await res.json();
   560→      if (json.ok) {
   561→        toast.success("Hashtag group created", { description: name.trim() });
   562→        setOpen(false);
   563→        reset();
   564→        onCreated();
   565→      } else {
   566→        toast.error(json.error ?? "Failed to create group");
   567→      }
   568→    } catch {
   569→      toast.error("Network error");
   570→    } finally {
   571→      setSaving(false);
   572→    }
   573→  }
   574→
   575→  const previewCount = parseTags(tags).length;
   576→
   577→  return (
   578→    <>
   579→      <button
   580→        onClick={() => setOpen(true)}
   581→        className="liquid-pill glow-ring flex items-center gap-2 rounded-2xl px-4 py-2.5 text-sm font-semibold"
   582→      >
   583→        <Plus className="h-4 w-4" />
   584→        Create group
   585→      </button>
   586→      <Dialog
   587→        open={open}

   685→function EditGroupDialog({
   686→  group,
   687→  onOpenChange,
   688→  onSaved,
   689→}: {
   690→  group: HashtagGroup | null;
   691→  onOpenChange: (open: boolean) => void;
   692→  onSaved: () => void;
   693→}) {
   694→  const open = group !== null;
   695→  const [name, setName] = React.useState("");
   696→  const [tags, setTags] = React.useState("");
   697→  const [platform, setPlatform] = React.useState<HashtagPlatform>("all");
   698→  const [color, setColor] = React.useState(SWATCHES[0]);
   699→  const [saving, setSaving] = React.useState(false);
   700→
   701→  React.useEffect(() => {
   702→    if (!group) return;
   703→    setName(group.name);
   704→    setTags(group.tags);
   705→    setPlatform((group.platform as HashtagPlatform) ?? "all");
   706→    setColor(group.color || SWATCHES[0]);
   707→  }, [group]);
   708→
   709→  async function handleSubmit() {
   710→    if (!group) return;
   711→    if (!name.trim() || !tags.trim()) {
   712→      toast.error("Name and tags are required");
   713→      return;
   714→    }
   715→    setSaving(true);
   716→    try {
   717→      const res = await fetch(`/api/hashtags/${group.id}`, {
   718→        method: "PATCH",
   719→        headers: { "Content-Type": "application/json" },
   720→        body: JSON.stringify({
   721→          name: name.trim(),
   722→          tags: tags.trim(),
   723→          platform,
   724→          color,
   725→        }),
   726→      });
   727→      const json = await res.json();
   728→      if (!res.ok || !json.ok) {
   729→        throw new Error(json.error ?? "Failed to update group");
   730→      }
   731→      toast.success("Hashtag group updated", { description: name.trim() });
   732→      onSaved();
   733→    } catch (e) {
   734→      toast.error(e instanceof Error ? e.message : "Network error");
   735→    } finally {
   736→      setSaving(false);
   737→    }
   738→  }
   739→
   740→  const previewCount = parseTags(tags).length;
   741→
   742→  return (
   743→    <Dialog open={open} onOpenChange={onOpenChange}>
   744→      <DialogContent className="rounded-3xl sm:max-w-md">
   745→        <DialogHeader>
   746→          <DialogTitle className="flex items-center gap-2">
   747→            <Pencil className="h-5 w-5 text-primary" />
   748→            Edit hashtag group
   749→          </DialogTitle>
   750→          <DialogDescription>
   751→            Update the name, tags, platform, or color of this group.
   752→          </DialogDescription>
   753→        </DialogHeader>
   754→        <div className="space-y-4 py-2">
   755→          <div className="space-y-2">
   756→            <Label htmlFor="edt-hg-name">Group name</Label>
   757→            <Input
   758→              id="edt-hg-name"
   759→              value={name}
   760→              onChange={(e) => setName(e.target.value)}
   761→            />
   762→          </div>
   763→          <div className="space-y-2">
   764→            <Label htmlFor="edt-hg-tags">
   765→              Tags{" "}
   766→              <span className="text-[11px] font-normal text-muted-foreground">
   767→                ({previewCount} parsed)
   768→              </span>
   769→            </Label>
   770→            <Textarea
   771→              id="edt-hg-tags"
   772→              value={tags}
   773→              onChange={(e) => setTags(e.target.value)}
   774→              rows={4}
   775→            />
   776→            <p className="text-[11px] text-muted-foreground">
   777→              Comma or newline separated. The <code>#</code> prefix is added automatically.
   778→            </p>
   779→          </div>
   780→          <div className="space-y-2">
   781→            <Label>Platform</Label>
   782→            <Select
   783→              value={platform}
   784→              onValueChange={(v) => setPlatform(v as HashtagPlatform)}
   785→            >
   786→              <SelectTrigger className="w-full">
   787→                <SelectValue />
   788→              </SelectTrigger>
   789→              <SelectContent>
   790→                {PLATFORM_OPTIONS.map((p) => (
   791→                  <SelectItem key={p.value} value={p.value}>
   792→                    {p.label}
   793→                  </SelectItem>
   794→                ))}
   795→              </SelectContent>
   796→            </Select>
   797→          </div>
   798→          <div className="space-y-2">
   799→            <Label>Color</Label>
   800→            <div className="flex flex-wrap gap-2">
   801→              {SWATCHES.map((sw) => (
   802→                <button
   803→                  key={sw}
   804→                  type="button"
   805→                  onClick={() => setColor(sw)}
   806→                  className={cn(
   807→                    "h-8 w-8 rounded-full transition",
   808→                    color === sw
   809→                      ? "ring-2 ring-offset-2 ring-offset-background"
   810→                      : "opacity-80 hover:opacity-100",
   811→                  )}
   812→                  style={{ background: sw, boxShadow: color === sw ? `0 0 0 2px ${sw}` : undefined }}
   813→                  aria-label={`Color ${sw}`}
   814→                />
   815→              ))}
   816→            </div>
   817→          </div>

   821→            Cancel
   822→          </Button>

   824→            {saving ? "Saving…" : "Save changes"}
   825→            {!saving && <CheckCircle2 className="h-3.5 w-3.5" />}
