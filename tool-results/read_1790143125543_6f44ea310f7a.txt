   457→function AddCompetitorForm({
   458→  onCancel,
   459→  onDone,
   460→}: {
   461→  onCancel: () => void;
   462→  onDone: () => void;
   463→}) {
   464→  const [name, setName] = React.useState("");
   465→  const [handle, setHandle] = React.useState("");
   466→  const [platform, setPlatform] = React.useState<PlatformId>("instagram");
   467→  const [notes, setNotes] = React.useState("");
   468→  const [busy, setBusy] = React.useState(false);
   469→
   470→  const submit = async () => {
   471→    if (!name.trim() || !handle.trim()) {
   472→      toast.error("Name and handle are required");
   473→      return;
   474→    }
   475→    setBusy(true);
   476→    try {
   477→      const res = await fetch("/api/competitors", {
   478→        method: "POST",
   479→        headers: { "Content-Type": "application/json" },
   480→        body: JSON.stringify({
   481→          name: name.trim(),
   482→          handle: handle.trim().replace(/^@/, ""),
   483→          platform,
   484→          notes: notes.trim() || null,
   485→        }),
   486→      });
   487→      if (!res.ok) throw new Error("Create failed");
   488→      toast.success("Competitor added");
   489→      setName("");
   490→      setHandle("");
   491→      setNotes("");
   492→      onDone();
   493→    } catch {
   494→      toast.error("Could not add competitor");
   495→    } finally {
   496→      setBusy(false);
   497→    }
   498→  };
   499→
   500→  return (
   501→    <div className="space-y-4 py-2">
   502→      <div className="space-y-2">
   503→        <Label htmlFor="c-name">Name</Label>
   504→        <Input
   505→          id="c-name"
   506→          placeholder="e.g. Acme Studio"
   507→          value={name}
   508→          onChange={(e) => setName(e.target.value)}
   509→        />
   510→      </div>
   511→      <div className="grid grid-cols-2 gap-3">
   512→        <div className="space-y-2">
   513→          <Label htmlFor="c-handle">Handle</Label>
   514→          <Input
   515→            id="c-handle"
   516→            placeholder="@acme"
   517→            value={handle}
   518→            onChange={(e) => setHandle(e.target.value)}
   519→          />
   520→        </div>
   521→        <div className="space-y-2">
   522→          <Label>Platform</Label>
   523→          <Select
   524→            value={platform}
   525→            onValueChange={(v) => setPlatform(v as PlatformId)}
   526→          >
   527→            <SelectTrigger className="w-full">
   528→              <SelectValue />
   529→            </SelectTrigger>
   530→            <SelectContent>
   531→              {PLATFORM_LIST.map((p) => (
   532→                <SelectItem key={p.id} value={p.id}>
   533→                  {p.name}
   534→                </SelectItem>
   535→              ))}
   536→            </SelectContent>
   537→          </Select>
   538→        </div>
   539→      </div>
   540→      <div className="space-y-2">
   541→        <Label htmlFor="c-notes">Notes</Label>
   542→        <Textarea
   543→          id="c-notes"
   544→          placeholder="What should the team watch for?"
   545→          value={notes}
   546→          onChange={(e) => setNotes(e.target.value)}
   547→          className="min-h-[80px] resize-none"
   548→        />
   549→      </div>
   550→      <DialogFooter>
   551→        <Button variant="ghost" onClick={onCancel}>
   552→          Cancel
   553→        </Button>
   554→        <button
   555→          onClick={submit}
   556→          disabled={busy}
   557→          className="liquid-pill flex items-center gap-2 rounded-xl px-4 py-2 text-sm font-semibold disabled:opacity-60"
   558→        >
   559→          <Plus className="h-4 w-4" />
   560→          {busy ? "Adding…" : "Add competitor"}
   561→        </button>
   562→      </DialogFooter>
   563→    </div>
   564→  );
   565→}

   567→function CompetitorCard({ competitor }: { competitor: Competitor }) {
   568→  const growthUp = competitor.growthRate >= 0;
   569→  const sparkColor = growthUp ? "oklch(0.76 0.18 145)" : "var(--primary)";
   570→  const series = competitor.series.length
   571→    ? competitor.series
   572→    : [
   573→        { date: new Date().toISOString().slice(0, 10), followers: competitor.followers, engagement: 0, posts: 0 },
   574→      ];
   575→  const benchmarkPct =
   576→    competitor.followers > 0
   577→      ? Math.min(100, Math.round((competitor.followers / OUR_FOLLOWERS) * 100))
   578→      : 0;
   579→  const ahead = competitor.followers >= OUR_FOLLOWERS;
   580→
   581→  return (
   582→    <GlassCard hover className="flex flex-col p-5">
   583→      <div className="flex items-start gap-3">
   584→        <PlatformBadge platform={competitor.platform} size={44} />
   585→        <div className="min-w-0 flex-1">
   586→          <div className="flex flex-wrap items-center gap-2">
   587→            <h3 className="truncate text-base font-bold leading-tight">
   588→              {competitor.name}
   589→            </h3>
   590→            <span
   591→              className={cn(
   592→                "inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-semibold capitalize",
   593→                competitor.status === "tracking"
   594→                  ? "bg-emerald-500/15 text-emerald-600 dark:text-emerald-400"
   595→                  : "bg-muted text-muted-foreground",
   596→              )}
   597→            >
   598→              {competitor.status === "tracking" ? (
   599→                <Play className="h-2.5 w-2.5" />
   600→              ) : (
   601→                <Pause className="h-2.5 w-2.5" />
   602→              )}
   603→              {competitor.status}
   604→            </span>
   605→          </div>
   606→          <PlatformPill
   607→            platform={competitor.platform}
   608→            handle={competitor.handle}
   609→            className="mt-1"
   610→          />
   611→        </div>
   612→        <div className="text-right">
   613→          <div className="text-xl font-bold tabular-nums leading-none">
   614→            {fmt(competitor.followers)}
   615→          </div>
   616→          <div className="text-[10px] text-muted-foreground">followers</div>
   617→        </div>
   618→      </div>
   619→
   620→      {/* Sparkline */}
   621→      <div className="mt-4 h-16 w-full">
   622→        <ResponsiveContainer width="100%" height="100%">
   623→          <LineChart
   624→            data={series}
   625→            margin={{ left: 0, right: 0, top: 4, bottom: 0 }}
   626→          >
   627→            <Line
   628→              type="monotone"
   629→              dataKey="followers"
   630→              stroke={sparkColor}
   631→              strokeWidth={2}
   632→              dot={false}
   633→              isAnimationActive={false}
   634→            />
   635→          </LineChart>
   636→        </ResponsiveContainer>
   637→      </div>
   638→
   639→      {/* Stats grid */}
   640→      <div className="mt-3 grid grid-cols-3 gap-2">
   641→        <Stat
   642→          label="Engagement"
   643→          value={`${competitor.engagementRate.toFixed(1)}%`}
   644→        />
   645→        <Stat
   646→          label="Growth"
   647→          value={`${growthUp ? "+" : ""}${competitor.growthRate.toFixed(1)}%`}
   648→          tone={growthUp ? "up" : "down"}
   649→        />
   650→        <Stat
   651→          label="Posts/wk"
   652→          value={competitor.postingFreq.toFixed(1)}
   653→        />
   654→      </div>
   655→
   656→      {/* Benchmark vs us */}
   657→      <div className="mt-4 rounded-xl bg-muted/40 p-3">
   658→        <div className="mb-1.5 flex items-center justify-between text-[11px]">
   659→          <span className="text-muted-foreground">vs. our top account</span>
   660→          <span
   661→            className={cn(
   662→              "font-semibold",
   663→              ahead ? "text-rose-500" : "text-emerald-500",
   664→            )}
   665→          >
   666→            {ahead ? "+" : "−"}
   667→            {fmt(Math.abs(competitor.followers - OUR_FOLLOWERS))}
   668→          </span>
   669→        </div>
   670→        <div className="relative h-2 w-full overflow-hidden rounded-full bg-muted">
   671→          <div
   672→            className="absolute inset-y-0 left-0 rounded-full bg-gradient-to-r from-primary to-[var(--gold)]"
   673→            style={{ width: `${benchmarkPct}%` }}
   674→          />
   675→          <div
   676→            className="absolute inset-y-0 w-px bg-foreground/60"
   677→            style={{ left: "100%", transform: "translateX(-100%)" }}
   678→            aria-hidden
   679→          />
   680→        </div>
   681→        <div className="mt-1 flex items-center justify-between text-[10px] text-muted-foreground">
   682→          <span>{fmt(competitor.followers)} rival</span>
   683→          <span>{fmt(OUR_FOLLOWERS)} us</span>
   684→        </div>
   685→      </div>
   686→
   687→      {/* Notes */}
   688→      {competitor.notes && (
   689→        <div className="mt-3 rounded-xl bg-muted/30 px-3 py-2 text-[12px] text-foreground/80">
   690→          <span className="mr-1 text-muted-foreground">“</span>
   691→          {competitor.notes}
   692→          <span className="ml-0.5 text-muted-foreground">”</span>
   693→        </div>
   694→      )}
   695→
   696→      <div className="mt-4 flex items-center gap-2 border-t border-border/60 pt-3">
   697→        <Button
   698→          variant="ghost"
   699→          size="sm"
   700→          className="h-8 rounded-xl"
   701→          onClick={() => toast.info(`Opening ${competitor.name} profile`)}
   702→        >
   703→          View profile <ArrowRight className="ml-1 h-3.5 w-3.5" />
   704→        </Button>
   705→        <Sparkles className="ml-auto h-3.5 w-3.5 text-muted-foreground/40" />
   706→      </div>
   707→    </GlassCard>
   708→  );
   709→}
